## File 2 — `cmd/api/main.go`

Purpose: Backend application entry point with structured logging, environment-based configuration, HTTP server timeouts, graceful shutdown, and startup error handling.

```

package main

import (
	"context"
	"errors"
	"fmt"
	"net/http"
	"os"
	"os/signal"
	"syscall"
	"time"

	"github.com/gin-gonic/gin"
	"go.uber.org/zap"
)

const (
	defaultHTTPAddr       = ":8080"
	defaultShutdownTimeout = 15 * time.Second
	defaultReadTimeout     = 10 * time.Second
	defaultReadHeaderTimeout = 5 * time.Second
	defaultWriteTimeout    = 30 * time.Second
	defaultIdleTimeout     = 60 * time.Second
)

func main() {
	if err := run(); err != nil {
		_, _ = fmt.Fprintf(os.Stderr, "kdxlabsllm: %v\n", err)
		os.Exit(1)
	}
}

func run() error {
	logger, err := newLogger()
	if err != nil {
		return fmt.Errorf("initialize logger: %w", err)
	}
	defer func() {
		_ = logger.Sync()
	}()

	addr := os.Getenv("HTTP_ADDR")
	if addr == "" {
		addr = defaultHTTPAddr
	}

	if os.Getenv("APP_ENV") == "production" {
		gin.SetMode(gin.ReleaseMode)
	} else {
		gin.SetMode(gin.DebugMode)
	}

	router := gin.New()
	router.Use(gin.Recovery())

	// Health endpoints are intentionally lightweight. Dependency-aware
	// readiness checks will be registered when database and cache clients
	// are initialized in subsequent files.
	router.GET("/health/live", func(c *gin.Context) {
		c.JSON(http.StatusOK, gin.H{
			"status":  "ok",
			"service": "kdxlabsllm",
		})
	})

	router.GET("/health/ready", func(c *gin.Context) {
		c.JSON(http.StatusOK, gin.H{
			"status":  "ok",
			"service": "kdxlabsllm",
		})
	})

	server := &http.Server{
		Addr:              addr,
		Handler:           router,
		ReadTimeout:       defaultReadTimeout,
		ReadHeaderTimeout: defaultReadHeaderTimeout,
		WriteTimeout:      defaultWriteTimeout,
		IdleTimeout:       defaultIdleTimeout,
	}

	serverErrors := make(chan error, 1)
	go func() {
		logger.Info("starting HTTP server",
			zap.String("service", "kdxlabsllm"),
			zap.String("address", addr),
			zap.String("environment", appEnvironment()),
		)

		if err := server.ListenAndServe(); err != nil &&
			!errors.Is(err, http.ErrServerClosed) {
			serverErrors <- err
		}
	}()

	ctx, stop := signal.NotifyContext(
		context.Background(),
		os.Interrupt,
		syscall.SIGTERM,
	)
	defer stop()

	select {
	case err := <-serverErrors:
		return fmt.Errorf("HTTP server failed: %w", err)

	case <-ctx.Done():
		logger.Info("shutdown signal received")
	}

	shutdownCtx, cancel := context.WithTimeout(
		context.Background(),
		defaultShutdownTimeout,
	)
	defer cancel()

	if err := server.Shutdown(shutdownCtx); err != nil {
		logger.Error("graceful shutdown failed", zap.Error(err))

		if closeErr := server.Close(); closeErr != nil {
			return fmt.Errorf(
				"shutdown server: %w; force close: %v",
				err,
				closeErr,
			)
		}

		return fmt.Errorf("shutdown server: %w", err)
	}

	logger.Info("HTTP server stopped")
	return nil
}

func newLogger() (*zap.Logger, error) {
	if appEnvironment() == "production" {
		return zap.NewProduction()
	}

	return zap.NewDevelopment()
}

func appEnvironment() string {
	if env := os.Getenv("APP_ENV"); env != "" {
		return env
	}

	return "development"
}
```
