## File 21 — `internal/platform/httpx/recovery.go`

This middleware provides panic recovery for KDXLabsLLM. If a handler panics, it logs the panic internally with the request ID and returns a generic HTTP 500 response instead of exposing implementation details.

```

package httpx

import (
	"net/http"

	"github.com/gin-gonic/gin"
	"go.uber.org/zap"
)

// Recovery returns middleware that recovers from panics in request handlers.
//
// Register RequestID before Recovery so recovered requests can be correlated
// with access logs and other diagnostics.
func Recovery(logger *zap.Logger) gin.HandlerFunc {
	if logger == nil {
		logger = zap.NewNop()
	}

	return func(c *gin.Context) {
		defer func() {
			if recovered := recover(); recovered != nil {
				logger.Error(
					"panic recovered while handling HTTP request",
					zap.Any("panic", recovered),
					zap.String("request_id", GetRequestID(c)),
					zap.String("method", c.Request.Method),
					zap.String("path", c.Request.URL.Path),
				)

				// If the response has already started, its status and headers
				// cannot safely be replaced with a new JSON response.
				if !c.Writer.Written() {
					c.AbortWithStatusJSON(http.StatusInternalServerError, Envelope{
						Error: &ErrorDetail{
							Code:    "INTERNAL_ERROR",
							Message: "An internal server error occurred.",
						},
						RequestID: GetRequestID(c),
					})
				} else {
					// A partially written response cannot be replaced.
					// Abort remaining handlers and let the server finish
					// the response that has already started.
					c.Abort()
				}
			}
		}()

		c.Next()
	}
}
```
