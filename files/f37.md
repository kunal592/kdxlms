	"time"

func (s *AssessmentAuditDeadLetterMemoryIdempotencyStore) CleanupExpired(
	ctx context.Context,
	before time.Time,
) (int, error) {
	if s == nil {
		return 0, errors.New("idempotency store is not initialized")
	}
	if err := ctx.Err(); err != nil {
		return 0, err
	}

	s.mu.Lock()
	defer s.mu.Unlock()

	removed := 0
	for key, record := range s.records {
		if !record.ExpiresAt.After(before) {
			delete(s.records, key)
			removed++
		}
	}
	return removed, nil
}

		snapshot.LastSuccess = time.Unix(0, value).UTC()

	headerName string,
	token string,
) *AssessmentAuditDeadLetterInternalSecretAuthorizer {
	headerName = strings.TrimSpace(headerName)
	if headerName == "" {
		headerName = "Authorization"
	}
	return &AssessmentAuditDeadLetterInternalSecretAuthorizer{
		headerName: headerName,
		token:      []byte(token),
	}
}

func (a *AssessmentAuditDeadLetterInternalSecretAuthorizer) AuthorizeInternalRequest(
	r *http.Request,
) bool {
	if a == nil || r == nil || len(a.token) == 0 {
		return false
	}

	provided := r.Header.Get(a.headerName)

		provided = strings.TrimSpace(strings.TrimPrefix(provided, "Bearer "))
	}
	if len(provided) != len(a.token) {

	}
