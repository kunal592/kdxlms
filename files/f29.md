import (
	"context"
	"errors"
	"net/http"

	"github.com/gin-gonic/gin"
	"github.com/kdxlabs/kdxlabsllm/internal/identity"
	"github.com/kdxlabs/kdxlabsllm/internal/platform/httpx"
)

type TenantAuthorizer interface {
	Authorize(context.Context, string, string, Permission) error
}

func RequireTenantPermission(
	authorizer TenantAuthorizer,
	permission Permission,
) gin.HandlerFunc {
	return func(c *gin.Context) {
		if authorizer == nil || !validPermission(permission) {
			httpx.InternalServerError(c)
			return
		}

		user, ok := identity.AuthenticatedUser(c)
		if !ok || user.ID == "" {
			httpx.Unauthorized(
				c,
				"AUTHENTICATION_REQUIRED",
				"Authentication is required.",
			)
			return
		}

		tenantID := c.Param("tenant_id")
		if tenantID == "" {
			httpx.BadRequest(c, "INVALID_TENANT_ID", "A tenant identifier is required.")
			return
		}

		if err := authorizer.Authorize(
			c.Request.Context(),
			tenantID,
			user.ID,
			permission,
		); err != nil {
			if errors.Is(err, ErrAccessDenied) {
				httpx.Fail(
					c,
					http.StatusForbidden,
					"TENANT_ACCESS_DENIED",
					"You do not have permission to perform this action.",
				)
				return
			}

			httpx.InternalServerError(c)
			return
		}

		c.Next()
	}
}

package course

import (
	"errors"
	"regexp"
	"strings"
	"time"
)

type Status string

const (
	StatusDraft     Status = "draft"
	StatusPublished Status = "published"
	StatusArchived  Status = "archived"
)

var courseSlugPattern = regexp.MustCompile(`^[a-z0-9]+(?:-[a-z0-9]+)*$`)

type Course struct {
	ID          string    `json:"id"`
	TenantID    string    `json:"tenant_id"`
	Title       string    `json:"title"`
	Slug        string    `json:"slug"`
	Description string    `json:"description"`
	InstructorID string   `json:"instructor_id"`
	Status      Status    `json:"status"`
	PublishedAt *time.Time `json:"published_at,omitempty"`
	CreatedAt   time.Time `json:"created_at"`
	UpdatedAt   time.Time `json:"updated_at"`
}

func NormalizeSlug(value string) (string, error) {
	slug := strings.ToLower(strings.TrimSpace(value))
	if len(slug) < 3 || len(slug) > 100 || !courseSlugPattern.MatchString(slug) {
		return "", ErrInvalidSlug
	}
	return slug, nil
}

func (c Course) Validate() error {
	if strings.TrimSpace(c.ID) == "" || strings.TrimSpace(c.TenantID) == "" {
		return ErrInvalidID
	}
	title := strings.TrimSpace(c.Title)
	if title == "" || len(title) > 200 {
		return ErrInvalidTitle
	}
	if _, err := NormalizeSlug(c.Slug); err != nil {
		return err
	}
	if len(c.Description) > 10000 {
		return ErrInvalidDescription
	}
	if strings.TrimSpace(c.InstructorID) == "" {
		return ErrInvalidInstructor
	}
	switch c.Status {
	case StatusDraft, StatusPublished, StatusArchived:
	default:
		return ErrInvalidStatus
	}
	if c.Status == StatusPublished && c.PublishedAt == nil {
		return errors.New("published courses require a publication timestamp")
	}
	if c.CreatedAt.IsZero() || c.UpdatedAt.Before(c.CreatedAt) {
		return errors.New("course timestamps are invalid")
	}
	return nil
}

type Clock interface {
	Now() time.Time
}

type IDGenerator func(prefix string) string

type Service struct {
	repository Repository
	clock      Clock
	newID      IDGenerator
}

func NewService(repository Repository, clock Clock, newID IDGenerator) (*Service, error) {
	if repository == nil || clock == nil || newID == nil {
		return nil, errors.New("course service: repository, clock, and ID generator are required")
	}
	return &Service{repository: repository, clock: clock, newID: newID}, nil
}

type CreateInput struct {
	TenantID     string
	Title        string
	Slug         string
	Description  string
	InstructorID string
}

func (s *Service) Create(ctx context.Context, input CreateInput) (Course, error) {
	if ctx == nil {
		return Course{}, errors.New("course service: context is required")
	}

	tenantID := strings.TrimSpace(input.TenantID)
	instructorID := strings.TrimSpace(input.InstructorID)
	title := strings.TrimSpace(input.Title)

	if tenantID == "" {
		return Course{}, ErrInvalidID
	}
	if instructorID == "" {
		return Course{}, ErrInvalidInstructor
	}
	if title == "" || len(title) > 200 {
		return Course{}, ErrInvalidTitle
	}

	slug, err := NormalizeSlug(input.Slug)
	if err != nil {
		return Course{}, err
	}

	now := s.clock.Now().UTC()
	record := Course{
		ID:           s.newID("crs"),
		TenantID:     tenantID,
		Title:        title,
		Slug:         slug,
		Description:  strings.TrimSpace(input.Description),
		InstructorID: instructorID,
		Status:       StatusDraft,
		CreatedAt:    now,
		UpdatedAt:    now,
	}
	if err := record.Validate(); err != nil {
		return Course{}, err
	}
	if err := s.repository.Create(ctx, record); err != nil {
		return Course{}, err
	}
	return record, nil
}

func (s *Service) Get(ctx context.Context, tenantID, courseID string) (Course, error) {
	if ctx == nil {
		return Course{}, errors.New("course service: context is required")
	}
	if strings.TrimSpace(tenantID) == "" || strings.TrimSpace(courseID) == "" {
		return Course{}, ErrInvalidID
	}
	return s.repository.GetByID(ctx, tenantID, courseID)
}

func (s *Service) List(
	ctx context.Context,
	tenantID string,
	statuses []Status,
	page int,
	pageSize int,
) ([]Course, int64, error) {
	if ctx == nil {
		return nil, 0, errors.New("course service: context is required")
	}
	if strings.TrimSpace(tenantID) == "" {
		return nil, 0, ErrInvalidID
	}
	if page < 1 || page > 1_000_000 || pageSize < 1 || pageSize > 100 {
		return nil, 0, errors.New("course service: invalid pagination")
	}
	for _, status := range statuses {
		if !validStatus(status) {
			return nil, 0, ErrInvalidStatus
		}
	}
	return s.repository.ListByTenant(ctx, tenantID, statuses, pageSize, (page-1)*pageSize)
}

	switch status {
	case StatusDraft, StatusPublished, StatusArchived:
		return true
	default:
		return false
	}

	_ context.Context,
	tenantID string,
	statuses []Status,
	limit, offset int,
) ([]Course, int64, error) {
	var records []Course
	for _, record := range r.records {
		if record.TenantID != tenantID {
			continue
		}
		if len(statuses) > 0 {
			matched := false
			for _, status := range statuses {
				if record.Status == status {
					matched = true
					break
				}
			}
			if !matched {
				continue
			}
		}
		records = append(records, record)
	}

	total := int64(len(records))
	if offset >= len(records) {
		return []Course{}, total, nil
	}
	end := offset + limit
	if end > len(records) {
		end = len(records)
	}
	return records[offset:end], total, nil

func (r *PostgresRepository) ListByTenant(
	ctx context.Context,
	tenantID string,
	statuses []Status,
	limit, offset int,
) ([]Course, int64, error) {
	if ctx == nil {
		return nil, 0, errors.New("course repository: context is required")
	}
	if strings.TrimSpace(tenantID) == "" || limit < 1 || limit > 100 || offset < 0 {
		return nil, 0, errors.New("course repository: invalid list parameters")
	}

	var total int64
	if err := r.db.QueryRow(ctx, `
		SELECT COUNT(*)
		FROM courses
		WHERE tenant_id = $1
		  AND (cardinality($2::text[]) = 0 OR status = ANY($2::text[]))
	`, tenantID, statusStrings(statuses)).Scan(&total); err != nil {
		return nil, 0, fmt.Errorf("count courses: %w", err)
	}

	rows, err := r.db.Query(ctx, `
		SELECT `+courseColumns+`
		FROM courses
		WHERE tenant_id = $1
		  AND (cardinality($2::text[]) = 0 OR status = ANY($2::text[]))
		ORDER BY created_at DESC, id ASC
		LIMIT $3 OFFSET $4
	`, tenantID, statusStrings(statuses), limit, offset)
	if err != nil {
		return nil, 0, fmt.Errorf("list courses: %w", err)
	}
	defer rows.Close()

	records := make([]Course, 0)
	for rows.Next() {
		record, scanErr := scanCourse(rows)
		if scanErr != nil {
			return nil, 0, scanErr
		}
		records = append(records, record)
	}
	if err := rows.Err(); err != nil {
		return nil, 0, fmt.Errorf("iterate courses: %w", err)
	}
	return records, total, nil
}

func (r *PostgresRepository) UpdateDetails(
	ctx context.Context,
	tenantID, courseID, title, description string,
	updatedAt time.Time,
) error {
	if ctx == nil {
		return errors.New("course repository: context is required")
	}
	title = strings.TrimSpace(title)
	if title == "" || len(title) > 200 || len(description) > 10000 {
		return ErrInvalidTitle
	}

	tag, err := r.db.Exec(ctx, `
		UPDATE courses
		SET title = $3, description = $4, updated_at = $5
		WHERE tenant_id = $1 AND id = $2 AND status <> 'archived'
	`, tenantID, courseID, title, strings.TrimSpace(description), updatedAt.UTC())
	if err != nil {
		return mapCourseWriteError(err)
	}
	if tag.RowsAffected() == 0 {
		return r.classifyMissingOrArchived(ctx, tenantID, courseID)
	}
	return nil
}

func (r *PostgresRepository) SetStatus(
	ctx context.Context,
	tenantID, courseID string,
	status Status,
	publishedAt *time.Time,
	updatedAt time.Time,
) error {
	if ctx == nil {
		return errors.New("course repository: context is required")
	}
	if !validStatus(status) {
		return ErrInvalidStatus
	}
	if status == StatusPublished && publishedAt == nil {
		return errors.New("published courses require a publication timestamp")
	}
	if status != StatusPublished {
		publishedAt = nil
	}

	tag, err := r.db.Exec(ctx, `
		UPDATE courses
		SET status = $3, published_at = $4, updated_at = $5
		WHERE tenant_id = $1 AND id = $2
	`, tenantID, courseID, string(status), publishedAt, updatedAt.UTC())
	if err != nil {
		return mapCourseWriteError(err)
	}
	if tag.RowsAffected() == 0 {
		return ErrNotFound
	}
	return nil
}

func (r *PostgresRepository) classifyMissingOrArchived(
	ctx context.Context,
	tenantID, courseID string,
) error {
	var status string
	err := r.db.QueryRow(ctx, `
		SELECT status FROM courses WHERE tenant_id = $1 AND id = $2
	`, tenantID, courseID).Scan(&status)
	if errors.Is(err, pgx.ErrNoRows) {
		return ErrNotFound
	}
	if err != nil {
		return fmt.Errorf("classify course update: %w", err)
	}
	if status == string(StatusArchived) {
		return ErrAccessDenied
	}
	return ErrNotFound
}

func statusStrings(statuses []Status) []string {
	result := make([]string, len(statuses))
	for i, status := range statuses {
		result[i] = string(status)
	}
	return result
}

func mapCourseWriteError(err error) error {
	var pgErr *pgconn.PgError
	if errors.As(err, &pgErr) {
		switch {
		case pgErr.Code == "23505" && pgErr.ConstraintName == "courses_tenant_slug_unique":
			return ErrSlugTaken
		case pgErr.Code == "23503" && pgErr.ConstraintName == "courses_tenant_instructor_membership_fk":
			return ErrInvalidInstructor
		case pgErr.Code == "23514" && strings.HasPrefix(pgErr.ConstraintName, "courses_"):
			return ErrInvalidStatus
		}
	}
	return fmt.Errorf("course database operation: %w", err)
}

		return
	}

import (
	"errors"
	"strings"
	"time"
)

type LessonType string

const (
	LessonTypeText  LessonType = "text"
	LessonTypeVideo LessonType = "video"
	LessonTypeLink  LessonType = "link"
)

type LessonStatus string

const (
	LessonStatusDraft     LessonStatus = "draft"
	LessonStatusPublished LessonStatus = "published"
)

type Lesson struct {
	ID           string       `json:"id"`
	TenantID     string       `json:"tenant_id"`
	CourseID     string       `json:"course_id"`
	Title        string       `json:"title"`
	Description  string       `json:"description"`
	Type         LessonType   `json:"type"`
	Content      string       `json:"content"`
	ContentURL   string       `json:"content_url,omitempty"`
	Position     int          `json:"position"`
	Status       LessonStatus `json:"status"`
	PublishedAt  *time.Time   `json:"published_at,omitempty"`
	CreatedAt    time.Time    `json:"created_at"`
	UpdatedAt    time.Time    `json:"updated_at"`
}

var (
	ErrLessonNotFound     = errors.New("lesson not found")
	ErrLessonInvalidID    = errors.New("lesson identifier is invalid")
	ErrLessonInvalidTitle = errors.New("lesson title is invalid")
	ErrLessonInvalidType  = errors.New("lesson type is invalid")
	ErrLessonInvalidState = errors.New("lesson status is invalid")
	ErrLessonInvalidOrder = errors.New("lesson position is invalid")
)

func (l Lesson) Validate() error {
	if strings.TrimSpace(l.ID) == "" ||
		strings.TrimSpace(l.TenantID) == "" ||
		strings.TrimSpace(l.CourseID) == "" {
		return ErrLessonInvalidID
	}
	title := strings.TrimSpace(l.Title)
	if title == "" || len(title) > 200 {
		return ErrLessonInvalidTitle
	}
	switch l.Type {
	case LessonTypeText:
		if strings.TrimSpace(l.Content) == "" || len(l.Content) > 200000 {
			return errors.New("text lesson content is invalid")
		}
	case LessonTypeVideo, LessonTypeLink:
		if strings.TrimSpace(l.ContentURL) == "" || len(l.ContentURL) > 2048 {
			return errors.New("lesson content URL is invalid")
		}
	default:
		return ErrLessonInvalidType
	}
	if l.Position < 1 || l.Position > 10000 {
		return ErrLessonInvalidOrder
	}
	switch l.Status {
	case LessonStatusDraft:
		if l.PublishedAt != nil {
			return errors.New("draft lesson cannot have a publication timestamp")
		}
	case LessonStatusPublished:
		if l.PublishedAt == nil {
			return errors.New("published lesson requires a publication timestamp")
		}
	default:
		return ErrLessonInvalidState
	}
	if l.CreatedAt.IsZero() || l.UpdatedAt.Before(l.CreatedAt) {
		return errors.New("lesson timestamps are invalid")
	}
	return nil
}
