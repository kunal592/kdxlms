## File 11 — `internal/platform/httpx/request_id.go`

This middleware assigns a request ID to every HTTP request and returns it in the `X-Request-ID` response header. It also stores the ID in Gin's request context so logs and future error responses can reference the same request.

Security features:

- Generates cryptographically random IDs.
- Accepts an incoming request ID only if it matches a strict format.
- Rejects malformed or oversized incoming IDs by generating a new one.
- Avoids trusting arbitrary client-supplied header values.

```

package httpx

import (
	"crypto/rand"
	"encoding/hex"
	"net/http"
	"regexp"

	"github.com/gin-gonic/gin"
)

const (
	// RequestIDHeader is the HTTP header used to propagate request IDs.
	RequestIDHeader = "X-Request-ID"

	// RequestIDContextKey is the Gin context key for the request ID.
	RequestIDContextKey = "request_id"

	requestIDBytes = 16
)

var requestIDPattern = regexp.MustCompile(`^[a-zA-Z0-9][a-zA-Z0-9._-]{0,63}$`)

// RequestID returns Gin middleware that validates or generates a request ID.
//
// A valid incoming ID is preserved for distributed tracing. Invalid IDs are
// replaced with a cryptographically random identifier. The selected ID is
// stored in the Gin context and returned in the response header.
func RequestID() gin.HandlerFunc {
	return func(c *gin.Context) {
		requestID := c.GetHeader(RequestIDHeader)

		if !isValidRequestID(requestID) {
			generatedID, err := generateRequestID()
			if err != nil {
				// Fail closed rather than handling a request without a
				// correlation identifier if the secure random source fails.
				c.AbortWithStatusJSON(http.StatusInternalServerError, gin.H{
					"error": gin.H{
						"code":    "INTERNAL_ERROR",
						"message": "An internal server error occurred.",
					},
				})
				return
			}

			requestID = generatedID
		}

		c.Set(RequestIDContextKey, requestID)
		c.Header(RequestIDHeader, requestID)

		c.Next()
	}
}

// GetRequestID retrieves the request ID from Gin's context.
func GetRequestID(c *gin.Context) string {
	if c == nil {
		return ""
	}

	value, exists := c.Get(RequestIDContextKey)
	if !exists {
		return ""
	}

	requestID, ok := value.(string)
	if !ok {
		return ""
	}

	return requestID
}

func isValidRequestID(value string) bool {
	return requestIDPattern.MatchString(value)
}

func generateRequestID() (string, error) {
	var randomBytes [requestIDBytes]byte

	if _, err := rand.Read(randomBytes[:]); err != nil {
		return "", err
	}

	return hex.EncodeToString(randomBytes[:]), nil
}
```
