```
}

func (c AssessmentAuditCursorProxyConfig) Validate() error {
	for _, raw := range c.TrustedProxyCIDRs {
		prefix, err := netip.ParsePrefix(raw)
		if err != nil || prefix != prefix.Masked() {
			return ErrAssessmentAuditCursorProxyConfig
		}
	}
	return nil
}

func (c AssessmentAuditCursorProxyConfig) IsTrustedProxy(address string) bool {
	ip, err := netip.ParseAddr(address)
	if err != nil {
		return false
	}
	for _, raw := range c.TrustedProxyCIDRs {
		prefix, err := netip.ParsePrefix(raw)
		if err == nil && prefix.Contains(ip) {
			return true
		}
	}
	return false
}
```

```
package assessment

import (
	"errors"
	"net"
	"net/http"
	"net/netip"
	"strings"
)

var ErrAssessmentAuditCursorProxyIP = errors.New("untrusted or invalid proxy address")

// AssessmentAuditCursorEffectiveClientIP trusts X-Forwarded-For only when
// RemoteAddr belongs to a configured trusted proxy. The chain is evaluated
// from right to left, skipping trusted proxies until the first untrusted hop.
func AssessmentAuditCursorEffectiveClientIP(
	r *http.Request,
	config AssessmentAuditCursorProxyConfig,
) (string, error) {
	if r == nil {
		return "", ErrAssessmentAuditCursorProxyIP
	}

	host, _, err := net.SplitHostPort(strings.TrimSpace(r.RemoteAddr))
	if err != nil {
		return "", ErrAssessmentAuditCursorProxyIP
	}
	remote, err := netip.ParseAddr(host)
	if err != nil {
		return "", ErrAssessmentAuditCursorProxyIP
	}
	if !config.IsTrustedProxy(remote.String()) {
		return remote.String(), nil
	}

	raw := r.Header.Get("X-Forwarded-For")
	if raw == "" {
		return "", ErrAssessmentAuditCursorProxyIP
	}

	parts := strings.Split(raw, ",")
	addresses := make([]netip.Addr, 0, len(parts))
	for _, part := range parts {
		ip, parseErr := netip.ParseAddr(strings.TrimSpace(part))
		if parseErr != nil {
			return "", ErrAssessmentAuditCursorProxyIP
		}
		addresses = append(addresses, ip)
	}

	for i := len(addresses) - 1; i >= 0; i-- {
		if !config.IsTrustedProxy(addresses[i].String()) {
			return addresses[i].String(), nil
		}
	}
	return "", ErrAssessmentAuditCursorProxyIP
}
```

```
				http.StatusTooManyRequests,
				"rate_limited",

			)
```

```
package assessment

import (
	"context"
	"errors"
)

var ErrAssessmentAuditCursorPolicyBundleManagerV3 = errors.New("policy bundle manager revision conflict")

type AssessmentAuditCursorPolicyBundleManagerV3 struct {
	Registry    *AssessmentAuditCursorRevisionedPolicyRegistry
	Audit       AssessmentAuditCursorPolicyBundleAuditSink
	ResultSink  AssessmentAuditCursorPolicyBundleResultSink
	Secret      []byte
}

func (m AssessmentAuditCursorPolicyBundleManagerV3) Apply(
	ctx context.Context,
	bundle AssessmentAuditCursorPolicyBundle,
	expectedRevision AssessmentAuditCursorPolicyRevision,
	actorID string,
) error {
	if ctx == nil || m.Registry == nil || m.Audit == nil ||
		m.ResultSink == nil || actorID == "" {
		return ErrAssessmentAuditCursorPolicyBundleManagerV3
	}
	if err := ctx.Err(); err != nil {
		return err
	}
	if err := bundle.ValidateCanonical(m.Secret); err != nil {
		return ErrAssessmentAuditCursorPolicyBundleManagerV3
	}

	if err := m.Audit.RecordPolicyBundleEvent(ctx, AssessmentAuditCursorPolicyBundleAuditEvent{
		Revision:    bundle.Revision.Revision,
		Action:      "apply_requested",
		ActorID:     actorID,
		PolicyCount: len(bundle.Policies),
	}); err != nil {
		return ErrAssessmentAuditCursorPolicyBundleManagerV3
	}

	applyErr := m.Registry.ReplaceIfRevision(
		expectedRevision,
		bundle.Revision,
		bundle.Policies,
	)
	result := AssessmentAuditCursorPolicyBundleResult{
		Revision: bundle.Revision.Revision,
		ActorID:  actorID,
		Applied:  applyErr == nil,
	}
	if applyErr != nil {
		result.FailureCode = "policy_revision_conflict"
	}

	if err := RecordAssessmentAuditCursorPolicyBundleResult(ctx, m.ResultSink, result); err != nil {
		return ErrAssessmentAuditCursorPolicyBundleManagerV3
	}
	if applyErr != nil {
		return ErrAssessmentAuditCursorPolicyBundleManagerV3
	}
	return nil
}
```
