## File 23 — `internal/platform/httpx/body_limit.go`

This middleware limits incoming HTTP request body sizes to protect the API from oversized JSON payloads and memory exhaustion. Individual endpoints can later apply stricter limits for uploads, authentication, and course content.

```

package httpx

import (
	"net/http"

	"github.com/gin-gonic/gin"
)

const (
	// DefaultMaxRequestBodyBytes is the default request body limit for
	// ordinary JSON API endpoints: 1 MiB.
	DefaultMaxRequestBodyBytes int64 = 1 << 20
)

// BodyLimit rejects requests whose declared Content-Length exceeds maxBytes
// and wraps the request body to enforce the same limit when Content-Length
// is missing or inaccurate.
//
// Register this middleware before handlers that decode request bodies.
// Multipart uploads should use a separately configured, larger limit.
func BodyLimit(maxBytes int64) gin.HandlerFunc {
	if maxBytes <= 0 {
		maxBytes = DefaultMaxRequestBodyBytes
	}

	return func(c *gin.Context) {
		if c.Request.ContentLength > maxBytes {
			c.AbortWithStatusJSON(http.StatusRequestEntityTooLarge, Envelope{
				Error: &ErrorDetail{
					Code:    "REQUEST_BODY_TOO_LARGE",
					Message: "The request body exceeds the permitted size.",
				},
				RequestID: GetRequestID(c),
			})
			return
		}

		c.Request.Body = http.MaxBytesReader(
			c.Writer,
			c.Request.Body,
			maxBytes,
		)

		c.Next()
	}
}
```
