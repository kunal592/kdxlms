package assessment

import (
	"context"
	"errors"
	"fmt"
	"strings"
	"time"
)

type AssessmentPublisherRepository interface {
	GetByID(ctx context.Context, tenantID, courseID, assessmentID string) (Assessment, error)
	Update(ctx context.Context, assessment Assessment) error
}

type AssessmentPublishClock interface {
	Now() time.Time
}

type AssessmentPublishService struct {
	assessments AssessmentPublisherRepository
	questions   QuestionRepository
	clock       AssessmentPublishClock
}

func NewAssessmentPublishService(
	assessments AssessmentPublisherRepository,
	questions QuestionRepository,
	clock AssessmentPublishClock,
) (*AssessmentPublishService, error) {
	if assessments == nil || questions == nil || clock == nil {
		return nil, errors.New("publish service dependencies are required")
	}
	return &AssessmentPublishService{
		assessments: assessments,
		questions:   questions,
		clock:       clock,
	}, nil
}

func (s *AssessmentPublishService) Publish(
	ctx context.Context,
	tenantID, courseID, assessmentID string,
) (Assessment, error) {
	if strings.TrimSpace(tenantID) == "" ||
		strings.TrimSpace(courseID) == "" ||
		strings.TrimSpace(assessmentID) == "" {
		return Assessment{}, ErrInvalidAssessment
	}

	item, err := s.assessments.GetByID(ctx, tenantID, courseID, assessmentID)
	if err != nil {
		return Assessment{}, err
	}
	if item.Status == StatusPublished {
		return item, nil
	}

	questions, err := s.questions.ListQuestions(
		ctx, tenantID, courseID, assessmentID, MaxQuestionsPerAssessment+1, 0,
	)
	if err != nil {
		return Assessment{}, err
	}
	if err := ValidateQuestionCount(len(questions)); err != nil {
		return Assessment{}, err
	}
	if err := ValidateAssessmentForPublishing(item, questions); err != nil {
		return Assessment{}, err
	}
	if err := ValidateAttemptLimits(item.AttemptLimit, item.TimeLimitSeconds); err != nil {
		return Assessment{}, fmt.Errorf("%w: %v", ErrAssessmentCannotPublish, err)
	}

	now := s.clock.Now().UTC()
	item.Status = StatusPublished
	item.PublishedAt = &now
	item.UpdatedAt = now

	if err := s.assessments.Update(ctx, item); err != nil {
		return Assessment{}, err
	}
	return item, nil
}

package assessment

import (
	"context"
	"encoding/json"
	"errors"
	"net/http"
	"strings"
)

type AssessmentPublisher interface {
	Publish(
		ctx context.Context,
		tenantID, courseID, assessmentID string,
	) (Assessment, error)
}

type AssessmentPublishHandler struct {
	service AssessmentPublisher
}

func NewAssessmentPublishHandler(service AssessmentPublisher) *AssessmentPublishHandler {
	return &AssessmentPublishHandler{service: service}
}

func (h *AssessmentPublishHandler) Publish(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodPost {
		writeAssessmentError(w, http.StatusMethodNotAllowed, "method not allowed")
		return
	}

	tenantID := strings.TrimSpace(r.Header.Get("X-Tenant-ID"))
	courseID := strings.TrimSpace(r.PathValue("courseID"))
	assessmentID := strings.TrimSpace(r.PathValue("assessmentID"))

	if tenantID == "" || courseID == "" || assessmentID == "" {
		writeAssessmentError(w, http.StatusBadRequest, "missing request context")
		return
	}

	item, err := h.service.Publish(r.Context(), tenantID, courseID, assessmentID)
	if err != nil {
		switch {
		case errors.Is(err, ErrNotFound):
			writeAssessmentError(w, http.StatusNotFound, "assessment not found")
		case errors.Is(err, ErrAssessmentCannotPublish),
			errors.Is(err, ErrInvalidAttemptLimits):
			writeAssessmentError(w, http.StatusUnprocessableEntity, "assessment does not meet publishing requirements")
		default:
			writeAssessmentError(w, http.StatusInternalServerError, "could not publish assessment")
		}
		return
	}

	w.Header().Set("Content-Type", "application/json")
	_ = json.NewEncoder(w).Encode(NewAssessmentResponse(item))
}

type AuditClock interface {
	Now() time.Time
}

type AuditedAssessmentPublisher struct {
	publisher AssessmentPublisher
	audit     AssessmentAuditWriter
	clock     AuditClock
}

func NewAuditedAssessmentPublisher(
	publisher AssessmentPublisher,
	audit AssessmentAuditWriter,
	clock AuditClock,
) (*AuditedAssessmentPublisher, error) {
	if publisher == nil || audit == nil || clock == nil {
		return nil, errors.New("publisher, audit writer and clock are required")
	}
	return &AuditedAssessmentPublisher{
		publisher: publisher,
		audit:     audit,
		clock:     clock,
	}, nil
}

// PublishForActor publishes an assessment and records the actor and action.
// The caller must authorize the actor before invoking this method.
func (p *AuditedAssessmentPublisher) PublishForActor(
	ctx context.Context,
	tenantID, actorUserID, courseID, assessmentID string,
) (Assessment, error) {
	item, err := p.publisher.Publish(ctx, tenantID, courseID, assessmentID)
	if err != nil {
		return Assessment{}, err
	}

	err = WriteAssessmentAudit(ctx, p.audit, AssessmentAuditEvent{
		TenantID:     tenantID,
		ActorUserID:  actorUserID,
		CourseID:     courseID,
		AssessmentID: assessmentID,
		Action:       "assessment.published",
		OccurredAt:   p.clock.Now(),
	})
	if err != nil {
		return Assessment{}, errors.Join(
			errors.New("assessment published but audit write failed"),
			err,
		)
	}

	return item, nil
}

import (
	"context"

var ErrInvalidAuditFilter = errors.New("invalid assessment audit filter")

const (
	DefaultAssessmentAuditLimit = 50
	MaxAssessmentAuditLimit     = 200
)

type AssessmentAuditQueryService struct {
	repository AssessmentAuditRepository
}

func NewAssessmentAuditQueryService(
	repository AssessmentAuditRepository,
) (*AssessmentAuditQueryService, error) {
	if repository == nil {
		return nil, errors.New("assessment audit repository is required")
	}
	return &AssessmentAuditQueryService{repository: repository}, nil
}

func (s *AssessmentAuditQueryService) List(
	ctx context.Context,
	filter AssessmentAuditFilter,
) ([]AssessmentAuditRecord, error) {
	filter.TenantID = strings.TrimSpace(filter.TenantID)
	filter.CourseID = strings.TrimSpace(filter.CourseID)
	filter.AssessmentID = strings.TrimSpace(filter.AssessmentID)
	filter.ActorUserID = strings.TrimSpace(filter.ActorUserID)
	filter.Action = strings.TrimSpace(filter.Action)

	if filter.TenantID == "" {
		return nil, ErrInvalidAuditFilter
	}
	if filter.Limit == 0 {
		filter.Limit = DefaultAssessmentAuditLimit
	}
	if filter.Limit < 1 || filter.Limit > MaxAssessmentAuditLimit || filter.Offset < 0 {
		return nil, ErrInvalidAuditFilter
	}
	if filter.From != nil && filter.To != nil && filter.From.After(*filter.To) {
		return nil, ErrInvalidAuditFilter
	}
	if filter.From != nil {
		from := filter.From.UTC()
		filter.From = &from
	}
	if filter.To != nil {
		to := filter.To.UTC()
		filter.To = &to
	}

	return s.repository.ListAssessmentAudit(ctx, filter)
}

func NewAssessmentAuditTimeRange(from, to *time.Time) (*time.Time, *time.Time, error) {
	if from != nil && to != nil && from.After(*to) {
		return nil, nil, ErrInvalidAuditFilter
	}

	var normalizedFrom, normalizedTo *time.Time
	if from != nil {
		value := from.UTC()
		normalizedFrom = &value
	}
	if to != nil {
		value := to.UTC()
		normalizedTo = &value

func (h *AssessmentAuditHandler) List(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodGet {
		writeAssessmentError(w, http.StatusMethodNotAllowed, "method not allowed")
		return
	}

	filter, err := parseAssessmentAuditFilter(r)
	if err != nil {
		writeAssessmentError(w, http.StatusBadRequest, "invalid audit filters")
		return
	}

	records, err := h.service.List(r.Context(), filter)
	if err != nil {
		if errors.Is(err, ErrInvalidAuditFilter) {
			writeAssessmentError(w, http.StatusBadRequest, "invalid audit filters")
			return
		}
		writeAssessmentError(w, http.StatusInternalServerError, "could not retrieve audit events")
		return
	}

	w.Header().Set("Content-Type", "application/json")
	w.Header().Set("Cache-Control", "no-store")
	_ = json.NewEncoder(w).Encode(struct {
		Items []AssessmentAuditRecord `json:"items"`
		Count int                     `json:"count"`
	}{
		Items: records,
		Count: len(records),
	})
}

func parseAssessmentAuditFilter(r *http.Request) (AssessmentAuditFilter, error) {
	query := r.URL.Query()
	filter := AssessmentAuditFilter{
		TenantID:     strings.TrimSpace(r.Header.Get("X-Tenant-ID")),
		CourseID:     strings.TrimSpace(query.Get("course_id")),
		AssessmentID: strings.TrimSpace(query.Get("assessment_id")),
		ActorUserID:  strings.TrimSpace(query.Get("actor_user_id")),
		Action:       strings.TrimSpace(query.Get("action")),
	}

	if filter.TenantID == "" {
		return AssessmentAuditFilter{}, ErrInvalidAuditFilter
	}

	if raw := query.Get("limit"); raw != "" {
		value, err := strconv.Atoi(raw)
		if err != nil {
			return AssessmentAuditFilter{}, ErrInvalidAuditFilter
		}
		filter.Limit = value
	}

	if raw := query.Get("offset"); raw != "" {
		value, err := strconv.Atoi(raw)
		if err != nil {
			return AssessmentAuditFilter{}, ErrInvalidAuditFilter
		}
		filter.Offset = value
	}

	if raw := query.Get("from"); raw != "" {
		value, err := time.Parse(time.RFC3339, raw)
		if err != nil {
			return AssessmentAuditFilter{}, ErrInvalidAuditFilter
		}
		filter.From = &value
	}

	if raw := query.Get("to"); raw != "" {
		value, err := time.Parse(time.RFC3339, raw)
		if err != nil {
			return AssessmentAuditFilter{}, ErrInvalidAuditFilter
		}
		filter.To = &value
	}

type AssessmentAuditOutboxWorker struct {
	queue  AssessmentAuditOutboxQueue
	writer AssessmentAuditWriter
	clock  AuditClock
	batch  int
}

func NewAssessmentAuditOutboxWorker(
	queue AssessmentAuditOutboxQueue,
	writer AssessmentAuditWriter,
	clock AuditClock,
	batch int,
) (*AssessmentAuditOutboxWorker, error) {
	if queue == nil || writer == nil || clock == nil || batch < 1 || batch > 500 {
		return nil, errors.New("invalid assessment audit outbox worker configuration")
	}
	return &AssessmentAuditOutboxWorker{
		queue: queue, writer: writer, clock: clock, batch: batch,
	}, nil
}

func (w *AssessmentAuditOutboxWorker) ProcessBatch(
	ctx context.Context,
) (int, error) {
	records, err := w.queue.ClaimPendingAssessmentAudit(ctx, w.batch)
	if err != nil {
		return 0, err
	}

	processed := 0
	var firstErr error

	for _, record := range records {
		if err := ctx.Err(); err != nil {
			return processed, err
		}

		event := AssessmentAuditEvent{
			TenantID:     record.TenantID,
			ActorUserID:  record.ActorUserID,
			CourseID:     record.CourseID,
			AssessmentID: record.AssessmentID,
			Action:       record.Action,
			OccurredAt:   record.OccurredAt,
		}

		if err := ValidateAssessmentAuditEvent(event); err != nil {
			_ = w.queue.MarkAssessmentAuditFailed(ctx, record.EventID, err.Error())
			if firstErr == nil {
				firstErr = err
			}
			continue
		}

		if err := WriteAssessmentAudit(ctx, w.writer, event); err != nil {
			_ = w.queue.MarkAssessmentAuditFailed(ctx, record.EventID, err.Error())
			if firstErr == nil {
				firstErr = err
			}
			continue
		}

		if err := w.queue.MarkAssessmentAuditProcessed(
			ctx, record.EventID, w.clock.Now().UTC(),
		); err != nil {
			if firstErr == nil {
				firstErr = err
			}
			continue
		}
		processed++
	}

	return processed, firstErr
}

func (q *PostgresAssessmentAuditOutboxQueue) ClaimPendingAssessmentAudit(
	ctx context.Context,
	limit int,
) ([]AssessmentAuditOutboxRecord, error) {
	if q == nil || q.db == nil {
		return nil, errors.New("outbox queue is not initialized")
	}
	if limit < 1 || limit > 500 {
		return nil, ErrInvalidAuditOutboxEvent
	}

	tx, err := q.db.BeginTx(ctx, pgx.TxOptions{})
	if err != nil {
		return nil, fmt.Errorf("begin outbox claim: %w", err)
	}
	defer func() { _ = tx.Rollback(ctx) }()

	const query = `
		SELECT event_id, tenant_id, actor_user_id, course_id,
		       assessment_id, action, occurred_at, created_at, attempts
		FROM assessment_audit_outbox
		WHERE processed_at IS NULL
		  AND attempts < 20
		ORDER BY created_at ASC, event_id ASC
		LIMIT $1
		FOR UPDATE SKIP LOCKED
	`

	rows, err := tx.Query(ctx, query, limit)
	if err != nil {
		return nil, fmt.Errorf("claim audit outbox records: %w", err)
	}

	records := make([]AssessmentAuditOutboxRecord, 0, limit)
	for rows.Next() {
		var record AssessmentAuditOutboxRecord
		if err := rows.Scan(
			&record.EventID,
			&record.TenantID,
			&record.ActorUserID,
			&record.CourseID,
			&record.AssessmentID,
			&record.Action,
			&record.OccurredAt,
			&record.CreatedAt,
			&record.Attempts,
		); err != nil {
			rows.Close()
			return nil, fmt.Errorf("scan audit outbox record: %w", err)
		}
		records = append(records, record)
	}
	if err := rows.Err(); err != nil {
		rows.Close()
		return nil, fmt.Errorf("iterate audit outbox records: %w", err)
	}
	rows.Close()

	// Holding row locks across external writes is avoided by this implementation's
	// transaction boundary. Records are returned after commit, so a production
	// queue should add a lease/claim token before running multiple workers.
	if err := tx.Commit(ctx); err != nil {
		return nil, fmt.Errorf("commit audit outbox claim: %w", err)
	}
	return records, nil
}

func (q *PostgresAssessmentAuditOutboxQueue) MarkAssessmentAuditProcessed(
	ctx context.Context,
	eventID string,
	processedAt time.Time,
) error {
	if q == nil || q.db == nil {
		return errors.New("outbox queue is not initialized")
	}
	eventID = strings.TrimSpace(eventID)
	if eventID == "" || processedAt.IsZero() {
		return ErrInvalidOutboxStatusUpdate
	}

	tag, err := q.db.Exec(ctx, `
		UPDATE assessment_audit_outbox
		SET processed_at = $2, last_error = NULL
		WHERE event_id = $1 AND processed_at IS NULL
	`, eventID, processedAt.UTC())
	if err != nil {
		return fmt.Errorf("mark audit outbox processed: %w", err)
	}
	if tag.RowsAffected() == 0 {
		return ErrInvalidOutboxStatusUpdate
	}
	return nil
}

func (q *PostgresAssessmentAuditOutboxQueue) MarkAssessmentAuditFailed(
	ctx context.Context,
	eventID, message string,
) error {
	if q == nil || q.db == nil {
		return errors.New("outbox queue is not initialized")
	}
	eventID = strings.TrimSpace(eventID)
	if eventID == "" {
		return ErrInvalidOutboxStatusUpdate
	}

		message = message[:1000]
	}

		SET attempts = attempts + 1, last_error = $2
		WHERE event_id = $1 AND processed_at IS NULL
	`, eventID, message)
	if err != nil {
		return fmt.Errorf("mark audit outbox failure: %w", err)
	}
	return nil

var ErrInvalidOutboxRetryPolicy = errors.New("invalid outbox retry policy")

type AssessmentAuditRetryPolicy struct {
	MaxAttempts int
	BaseDelay   time.Duration
	MaxDelay    time.Duration
}

func (p AssessmentAuditRetryPolicy) Validate() error {
	if p.MaxAttempts < 1 || p.MaxAttempts > 100 ||
		p.BaseDelay <= 0 ||
		p.MaxDelay < p.BaseDelay {
		return ErrInvalidOutboxRetryPolicy
	}
	return nil
}

func (p AssessmentAuditRetryPolicy) Delay(attempt int) (time.Duration, error) {
	if err := p.Validate(); err != nil {
		return 0, err
	}
	if attempt < 1 {
		return 0, ErrInvalidOutboxRetryPolicy
	}

	delay := p.BaseDelay
	for i := 1; i < attempt && delay < p.MaxDelay; i++ {
		if delay > p.MaxDelay/2 {
			delay = p.MaxDelay
			break
		}
		delay *= 2
	}
	if delay > p.MaxDelay {
		delay = p.MaxDelay
	}
	return delay, nil
}

// Protects spreadsheet consumers from interpreting untrusted text as a formula.

func safeAssessmentAuditCSVCell(value string) string {
	trimmed := strings.TrimLeftFunc(value, unicode.IsSpace)

	}

	case '=', '+', '-', '@':
		return "'" + value
	default:
		return value
	}

	ctx context.Context,

			SELECT event_id
			FROM assessment_audit_outbox
			WHERE processed_at IS NULL
			  AND attempts >= $1
			  AND (claimed_until IS NULL OR claimed_until < NOW())
			ORDER BY created_at ASC, event_id ASC
			LIMIT $2
