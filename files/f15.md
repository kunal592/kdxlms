## File 15 — `internal/platform/httpx/security_headers.go`

This middleware adds baseline HTTP security headers to KDXLabsLLM API responses. It also prevents MIME sniffing, restricts framing, disables caching of sensitive API responses by default, and controls referrer information.

```

package httpx

import "github.com/gin-gonic/gin"

// SecurityHeaders returns middleware that applies baseline security headers.
//
// These headers complement authentication, authorization, input validation,
// TLS configuration, and other security controls; they do not replace them.
func SecurityHeaders() gin.HandlerFunc {
	return func(c *gin.Context) {
		header := c.Writer.Header()

		header.Set("X-Content-Type-Options", "nosniff")
		header.Set("X-Frame-Options", "DENY")
		header.Set("Referrer-Policy", "no-referrer")
		header.Set("X-Permitted-Cross-Domain-Policies", "none")

		// This backend serves an API, not browser-rendered HTML.
		// Prevent browsers and intermediaries from caching responses unless
		// a specific endpoint deliberately implements a different policy.
		header.Set("Cache-Control", "no-store")
		header.Set("Pragma", "no-cache")

		// Prevent browser features that are not required by this API.
		header.Set(
			"Permissions-Policy",
			"camera=(), microphone=(), geolocation=()",
		)

		// HSTS is added only when the request is known to use HTTPS.
		// In production, TLS should normally terminate at a trusted proxy.
		if c.Request.TLS != nil {
			header.Set(
				"Strict-Transport-Security",
				"max-age=31536000",
			)
		}

		c.Next()
	}
}
```
