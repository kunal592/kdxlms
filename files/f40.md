}

	if m == nil {
		return
	}
	if err != nil {
		m.encodeFailure.Add(1)
		return
	}
	m.encodeSuccess.Add(1)

	if m == nil {
		return
	}

		m.decodeFailure.Add(1)

	if m == nil {
		return
	}

	m.lastRotation.Store(now.Unix())

	if m != nil {
		m.pruneCount.Add(1)
	}
}

	if m == nil {
		return AssessmentAuditCursorProviderMetricsSnapshot{}

var ErrAssessmentAuditCursorRateLimitConfig = errors.New("assessment audit cursor rate-limit configuration is invalid")

type AssessmentAuditCursorRateLimitConfig struct {
	Requests int
	Window   time.Duration
}

func (c AssessmentAuditCursorRateLimitConfig) Validate() error {
	if c.Requests < 1 || c.Requests > 100000 ||
		c.Window < time.Second || c.Window > 24*time.Hour {
		return ErrAssessmentAuditCursorRateLimitConfig
	}
	return nil
}

type assessmentAuditCursorRateEntry struct {
	start time.Time
	count int
}

type AssessmentAuditCursorRateLimiter struct {
	mu      sync.Mutex
	config  AssessmentAuditCursorRateLimitConfig
	now     func() time.Time
	entries map[string]assessmentAuditCursorRateEntry
}

func NewAssessmentAuditCursorRateLimiter(
	config AssessmentAuditCursorRateLimitConfig,
	now func() time.Time,
) (*AssessmentAuditCursorRateLimiter, error) {
	if err := config.Validate(); err != nil {
		return nil, err
	}
	if now == nil {
		now = time.Now
	}

	return &AssessmentAuditCursorRateLimiter{
		config:  config,
		now:     now,
		entries: make(map[string]assessmentAuditCursorRateEntry),
	}, nil
}

func (l *AssessmentAuditCursorRateLimiter) Allow(key string) bool {
	if l == nil || strings.TrimSpace(key) == "" {
		return false
	}

	now := l.now()

	l.mu.Lock()
	defer l.mu.Unlock()

	entry, exists := l.entries[key]
	if !exists || now.Sub(entry.start) >= l.config.Window || now.Before(entry.start) {
		l.entries[key] = assessmentAuditCursorRateEntry{start: now, count: 1}
		return true
	}

	if entry.count >= l.config.Requests {
		return false
	}

	entry.count++
	l.entries[key] = entry
	return true
}

func assessmentAuditCursorClientIP(r *http.Request) string {
	if r == nil {
		return ""
	}

	host, _, err := net.SplitHostPort(r.RemoteAddr)
	if err == nil {
		return host
	}

	// Do not trust forwarded headers here. They must be normalized by a
	// separately configured trusted-proxy layer, if one exists.
	return strings.TrimSpace(r.RemoteAddr)
}

import "time"

// Cleanup removes expired rate-limit entries to bound memory usage.
// Call periodically from a managed maintenance loop.
func (l *AssessmentAuditCursorRateLimiter) Cleanup() int {
	if l == nil || l.now == nil {
		return 0
	}

	now := l.now()
	l.mu.Lock()
	defer l.mu.Unlock()

	removed := 0
	for key, entry := range l.entries {
		if now.Before(entry.start) || now.Sub(entry.start) >= l.config.Window {
			delete(l.entries, key)
			removed++
		}
	}

	return removed
}

func (l *AssessmentAuditCursorRateLimiter) WindowDuration() time.Duration {
	if l == nil {
		return 0
	}
	return l.config.Window
}

// AssessmentAuditCursorBoundedRateLimiter adds a hard cap to distinct keys.
// It is suitable for a single process; shared deployments should enforce the
// same cardinality and rate policy in their shared limiter.
type AssessmentAuditCursorBoundedRateLimiter struct {
	mu        sync.Mutex
	limiter   *AssessmentAuditCursorRateLimiter
	maxKeys   int
}

func NewAssessmentAuditCursorBoundedRateLimiter(
	limiter *AssessmentAuditCursorRateLimiter,
	maxKeys int,
) (*AssessmentAuditCursorBoundedRateLimiter, error) {
	if limiter == nil || maxKeys < 1 || maxKeys > 10_000_000 {
		return nil, ErrAssessmentAuditCursorLimiterUnavailable
	}
	return &AssessmentAuditCursorBoundedRateLimiter{
		limiter: limiter,
		maxKeys: maxKeys,
	}, nil
}

func (b *AssessmentAuditCursorBoundedRateLimiter) Allow(key string) (bool, error) {
	if b == nil || b.limiter == nil || key == "" {
		return false, ErrAssessmentAuditCursorLimiterUnavailable
	}

	b.mu.Lock()
	defer b.mu.Unlock()

	b.limiter.mu.Lock()
	_, exists := b.limiter.entries[key]
	count := len(b.limiter.entries)
	b.limiter.mu.Unlock()

	if !exists && count >= b.maxKeys {
		return false, ErrAssessmentAuditCursorRateLimitCardinality
	}

	return b.limiter.Allow(key), nil
}

import (
	"context"
	"sync"
	"time"
)

type assessmentAuditCursorRateLimitBucket struct {
	start time.Time
	count int
}

type AssessmentAuditCursorMemoryStore struct {
	mu      sync.Mutex
	now     func() time.Time
	buckets map[string]assessmentAuditCursorRateLimitBucket
}

func NewAssessmentAuditCursorMemoryStore(
	now func() time.Time,
) *AssessmentAuditCursorMemoryStore {
	if now == nil {
		now = time.Now
	}
	return &AssessmentAuditCursorMemoryStore{
		now:     now,
		buckets: make(map[string]assessmentAuditCursorRateLimitBucket),
	}
}

func (s *AssessmentAuditCursorMemoryStore) Consume(
	ctx context.Context,
	key string,
	limit int,
	window time.Duration,
) (AssessmentAuditCursorRateLimitDecision, error) {
	if s == nil || ctx == nil || s.now == nil {
		return AssessmentAuditCursorRateLimitDecision{}, ErrAssessmentAuditCursorRateLimitStore
	}
	if err := ctx.Err(); err != nil {
		return AssessmentAuditCursorRateLimitDecision{}, err
	}
	if err := ValidateAssessmentAuditCursorRateLimitKey(key); err != nil {
		return AssessmentAuditCursorRateLimitDecision{}, err
	}
	if limit < 1 || window < time.Second {
		return AssessmentAuditCursorRateLimitDecision{}, ErrAssessmentAuditCursorRateLimitConfig
	}

	now := s.now()
	s.mu.Lock()
	defer s.mu.Unlock()

	bucket, exists := s.buckets[key]
	if !exists || now.Before(bucket.start) || now.Sub(bucket.start) >= window {
		bucket = assessmentAuditCursorRateLimitBucket{start: now}
	}

	allowed := bucket.count < limit
	if allowed {
		bucket.count++

package assessment

import "time"

func (s *AssessmentAuditCursorMemoryStore) Cleanup(window time.Duration) int {
	if s == nil || s.now == nil || window < time.Second {
		return 0
	}

	now := s.now()
	s.mu.Lock()
	defer s.mu.Unlock()

	removed := 0
	for key, bucket := range s.buckets {
		if now.Before(bucket.start) || now.Sub(bucket.start) >= window {
			delete(s.buckets, key)
			removed++
		}
	}
	return removed
}

package assessment

import (
	"context"
	"time"
)

type AssessmentAuditCursorBoundedMemoryStore struct {
	Store  *AssessmentAuditCursorMemoryStore
	Config AssessmentAuditCursorRateLimitStoreConfig
}

func (s AssessmentAuditCursorBoundedMemoryStore) Consume(
	ctx context.Context,
	key string,
	limit int,
	window time.Duration,
) (AssessmentAuditCursorRateLimitDecision, error) {
	if s.Store == nil || ctx == nil {
		return AssessmentAuditCursorRateLimitDecision{}, ErrAssessmentAuditCursorRateLimitStore
	}
	if err := s.Config.Validate(); err != nil {
		return AssessmentAuditCursorRateLimitDecision{}, err
	}
	if err := ctx.Err(); err != nil {
		return AssessmentAuditCursorRateLimitDecision{}, err
	}

	s.Store.mu.Lock()
	_, exists := s.Store.buckets[key]
	count := len(s.Store.buckets)
	s.Store.mu.Unlock()

	if !exists && count >= s.Config.MaxEntries {
		s.Store.Cleanup(s.Config.DefaultWindow)
		s.Store.mu.Lock()
		_, exists = s.Store.buckets[key]
		count = len(s.Store.buckets)
		s.Store.mu.Unlock()
		if !exists && count >= s.Config.MaxEntries {
			return AssessmentAuditCursorRateLimitDecision{}, ErrAssessmentAuditCursorRateLimitStore
		}
	}

	if window <= 0 {
		window = s.Config.DefaultWindow
	}
	return s.Store.Consume(ctx, key, limit, window)
}

var _ AssessmentAuditCursorRateLimitStore = AssessmentAuditCursorBoundedMemoryStore{}
