package assessment

import (
	"errors"
	"io"
	"net/http"
)

var (
	ErrAssessmentAuditRequestBodyTooLarge = errors.New("request body exceeds configured limit")
	ErrAssessmentAuditRequestBodyMissing  = errors.New("request body is required")
)

func ReadAssessmentAuditRequestBody(
	r *http.Request,
	maxBytes int64,
) ([]byte, error) {
	if r == nil {
		return nil, errors.New("request is required")
	}
	if maxBytes <= 0 {
		return nil, errors.New("maximum body size must be positive")
	}
	if r.Body == nil {
		return nil, ErrAssessmentAuditRequestBodyMissing
	}
	if r.ContentLength > maxBytes {
		return nil, ErrAssessmentAuditRequestBodyTooLarge
	}

	body, err := io.ReadAll(io.LimitReader(r.Body, maxBytes+1))
	if err != nil {
		return nil, err
	}
	if int64(len(body)) > maxBytes {
		return nil, ErrAssessmentAuditRequestBodyTooLarge
	}
	if len(body) == 0 {
		return nil, ErrAssessmentAuditRequestBodyMissing
	}
	return body, nil
}

import (
	"net/http"
)

func AssessmentAuditDeadLetterRequestBodyLimitMiddleware(
	maxBytes int64,
	next http.Handler,
) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if r == nil {
			http.Error(w, "bad request", http.StatusBadRequest)
			return
		}
		if maxBytes <= 0 {
			http.Error(w, "server configuration error", http.StatusInternalServerError)
			return
		}
		if next == nil {
			http.Error(w, "server configuration error", http.StatusInternalServerError)
			return
		}
		if r.ContentLength > maxBytes {
			http.Error(w, "request body too large", http.StatusRequestEntityTooLarge)
			return
		}
		if r.Body != nil {
			r.Body = http.MaxBytesReader(w, r.Body, maxBytes)
		}

		next.ServeHTTP(w, r)
	})
}

		return ErrAssessmentAuditUnsupportedContentType

	return nil

package assessment

import (
	"bytes"
	"encoding/json"
	"errors"
	"io"
)

var (
	ErrAssessmentAuditInvalidJSON = errors.New("invalid JSON request")
	ErrAssessmentAuditTrailingJSON = errors.New("request contains trailing JSON data")
)

func DecodeAssessmentAuditJSON[T any](body []byte, destination *T) error {
	if destination == nil {
		return errors.New("destination is required")
	}
	if len(bytes.TrimSpace(body)) == 0 {
		return ErrAssessmentAuditInvalidJSON
	}

	decoder := json.NewDecoder(bytes.NewReader(body))
	decoder.DisallowUnknownFields()

	if err := decoder.Decode(destination); err != nil {
		return ErrAssessmentAuditInvalidJSON
	}

	var trailing any
	if err := decoder.Decode(&trailing); !errors.Is(err, io.EOF) {
		if err == nil {
			return ErrAssessmentAuditTrailingJSON
		}
		return ErrAssessmentAuditTrailingJSON
	}

	return nil
}

import (
	"errors"
	"net/http"
	"strings"

var ErrAssessmentAuditRequestIDInvalid = errors.New("invalid request ID")

func ValidateAssessmentAuditRequestID(r *http.Request, maxLength int) (string, error) {
	if r == nil {
		return "", errors.New("request is required")
	}
	if maxLength <= 0 {
		return "", errors.New("maximum request ID length must be positive")
	}

	value := strings.TrimSpace(r.Header.Get("X-Request-ID"))
	if value == "" || len(value) > maxLength {
		return "", ErrAssessmentAuditRequestIDInvalid
	}

	for _, ch := range value {
		if ch < 0x21 || ch > 0x7e {
			return "", ErrAssessmentAuditRequestIDInvalid
		}
	}

	return value, nil
}

package assessment

import (
	"context"
	"net/http"
)

type assessmentAuditRequestIDContextKey struct{}

func AssessmentAuditRequestIDFromContext(ctx context.Context) (string, bool) {
	if ctx == nil {
		return "", false
	}
	value, ok := ctx.Value(assessmentAuditRequestIDContextKey{}).(string)
	return value, ok && value != ""
}

func AssessmentAuditRequestIDMiddleware(maxLength int, next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if r == nil {
			http.Error(w, "bad request", http.StatusBadRequest)
			return
		}
		if maxLength <= 0 || next == nil {
			http.Error(w, "server configuration error", http.StatusInternalServerError)
			return
		}

		requestID, err := ValidateAssessmentAuditRequestID(r, maxLength)
		if err != nil {
			http.Error(w, "valid X-Request-ID header is required", http.StatusBadRequest)
			return
		}

		ctx := context.WithValue(r.Context(), assessmentAuditRequestIDContextKey{}, requestID)
		w.Header().Set("X-Request-ID", requestID)
		next.ServeHTTP(w, r.WithContext(ctx))
	})
}
