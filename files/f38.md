import (
	"errors"
	"io"
	"net/http"
)

var (
	ErrAssessmentAuditRequestBodyTooLarge = errors.New("request body exceeds configured limit")
	ErrAssessmentAuditRequestBodyMissing  = errors.New("request body is required")
)

func ReadAssessmentAuditRequestBody(
	r *http.Request,
	maxBytes int64,
) ([]byte, error) {
	if r == nil {
		return nil, errors.New("request is required")
	}
	if maxBytes <= 0 {
		return nil, errors.New("maximum body size must be positive")
	}
	if r.Body == nil {
		return nil, ErrAssessmentAuditRequestBodyMissing
	}
	if r.ContentLength > maxBytes {
		return nil, ErrAssessmentAuditRequestBodyTooLarge
	}

	body, err := io.ReadAll(io.LimitReader(r.Body, maxBytes+1))
	if err != nil {
		return nil, err
	}
	if int64(len(body)) > maxBytes {
		return nil, ErrAssessmentAuditRequestBodyTooLarge
	}
	if len(body) == 0 {
		return nil, ErrAssessmentAuditRequestBodyMissing
	}
	return body, nil
}

	"io"

var (
	ErrAssessmentAuditInvalidJSON = errors.New("invalid JSON request")
	ErrAssessmentAuditTrailingJSON = errors.New("request contains trailing JSON data")
)

func DecodeAssessmentAuditJSON[T any](body []byte, destination *T) error {
	if destination == nil {
		return errors.New("destination is required")
	}
	if len(bytes.TrimSpace(body)) == 0 {
		return ErrAssessmentAuditInvalidJSON
	}

	decoder := json.NewDecoder(bytes.NewReader(body))
	decoder.DisallowUnknownFields()

	if err := decoder.Decode(destination); err != nil {
		return ErrAssessmentAuditInvalidJSON
	}

	var trailing any
	if err := decoder.Decode(&trailing); !errors.Is(err, io.EOF) {
		if err == nil {
			return ErrAssessmentAuditTrailingJSON
		}
		return ErrAssessmentAuditTrailingJSON
	}

	return nil
}

var ErrAssessmentAuditRequestIDGeneration = errors.New("assessment audit request ID generation failed")

// GenerateAssessmentAuditRequestID creates a cryptographically random request ID.
// It deliberately does not accept caller-supplied values.
func GenerateAssessmentAuditRequestID() (string, error) {
	var raw [16]byte
	if _, err := rand.Read(raw[:]); err != nil {
		return "", errors.Join(ErrAssessmentAuditRequestIDGeneration, err)
	}

	id := hex.EncodeToString(raw[:])
	if strings.TrimSpace(id) == "" {
		return "", ErrAssessmentAuditRequestIDGeneration
	}
	return id, nil
}

import (
	"net/http"
	"strings"

// AssessmentAuditRequestIDFallbackMiddleware preserves a valid incoming request
// ID, or generates one when the caller omitted it. Invalid supplied IDs are
// rejected rather than silently replaced.
func AssessmentAuditRequestIDFallbackMiddleware(
	next http.Handler,
	maxLength int,
) http.Handler {
	if next == nil {
		next = http.NotFoundHandler()
	}
	if maxLength <= 0 {
		maxLength = 128
	}

	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		provided := strings.TrimSpace(r.Header.Get("X-Request-ID"))
		if provided != "" {
			if err := ValidateAssessmentAuditRequestID(provided, maxLength); err != nil {
				http.Error(w, "invalid request ID", http.StatusBadRequest)
				return
			}
			r.Header.Set("X-Request-ID", provided)
			next.ServeHTTP(w, r)
			return
		}

		id, err := GenerateAssessmentAuditRequestID()
		if err != nil {
			http.Error(w, "request ID generation failed", http.StatusInternalServerError)
			return
		}

		r.Header.Set("X-Request-ID", id)
		next.ServeHTTP(w, r)
	})
}

// AssessmentAuditAllowedMethodsMiddleware restricts a handler to an explicit
// allowlist and advertises the same methods in the Allow header.
func AssessmentAuditAllowedMethodsMiddleware(
	next http.Handler,
	methods ...string,
) http.Handler {
	if next == nil {
		next = http.NotFoundHandler()
	}

	allowed := make(map[string]struct{}, len(methods))
	ordered := make([]string, 0, len(methods))
	for _, method := range methods {
		method = strings.TrimSpace(method)
		if method == "" {
			continue
		}
		if _, exists := allowed[method]; exists {
			continue
		}
		allowed[method] = struct{}{}
		ordered = append(ordered, method)
	}
	allowHeader := strings.Join(ordered, ", ")

	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if _, ok := allowed[r.Method]; !ok {
			if allowHeader != "" {
				w.Header().Set("Allow", allowHeader)
			}
			http.Error(w, "method not allowed", http.StatusMethodNotAllowed)
			return
		}
		next.ServeHTTP(w, r)
	})
}

// AssessmentAuditExactOriginMiddleware permits browser requests only from
// explicitly configured origins. It does not enable credentials or wildcard
// origins. Non-browser requests without Origin are passed through unchanged.
func AssessmentAuditExactOriginMiddleware(
	next http.Handler,
	origins []string,
) http.Handler {
	if next == nil {
		next = http.NotFoundHandler()
	}

	allowed := make(map[string]struct{}, len(origins))
	for _, origin := range origins {
		origin = strings.TrimSpace(origin)
		if origin == "" || ValidateAssessmentAuditOrigin(origin) != nil {
			continue
		}
		allowed[origin] = struct{}{}
	}

	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		origin := strings.TrimSpace(r.Header.Get("Origin"))
		if origin == "" {
			next.ServeHTTP(w, r)
			return
		}

		if _, ok := allowed[origin]; !ok {
			http.Error(w, "origin not allowed", http.StatusForbidden)
			return
		}

		w.Header().Set("Vary", "Origin")
		w.Header().Set("Access-Control-Allow-Origin", origin)

		if r.Method == http.MethodOptions {
			w.Header().Set("Access-Control-Allow-Methods", "GET, POST, PUT, PATCH, DELETE, OPTIONS")
			w.Header().Set("Access-Control-Allow-Headers", "Authorization, Content-Type, X-Request-ID, Idempotency-Key")
			w.WriteHeader(http.StatusNoContent)
			return
		}

		next.ServeHTTP(w, r)
	})
}

	if maxConcurrent < 1 {
		maxConcurrent = 1
	}
	return &AssessmentAuditConcurrencyLimiter{
		slots: make(chan struct{}, maxConcurrent),
	}
}

// Middleware rejects excess concurrent requests instead of allowing unbounded
// work to accumulate in the handler.
func (l *AssessmentAuditConcurrencyLimiter) Middleware(next http.Handler) http.Handler {
	if next == nil {
		next = http.NotFoundHandler()
	}
	if l == nil || l.slots == nil {
		return next
	}

	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		select {
		case l.slots <- struct{}{}:
			defer func() { <-l.slots }()
			next.ServeHTTP(w, r)
		default:
			w.Header().Set("Retry-After", "1")
			w.Header().Set("X-Assessment-Audit-Concurrency-Limited", strconv.FormatBool(true))
			http.Error(w, "server is busy", http.StatusServiceUnavailable)
		}
	})

	handler http.Handler,

	}

			http.MethodPost,
			http.MethodOptions,
		}
	}

	result = AssessmentAuditAllowedMethodsMiddleware(result, cfg.AllowedMethods...)
	result = AssessmentAuditExactOriginMiddleware(result, cfg.AllowedOrigins)

	}
	return parsed, nil

var ErrAssessmentAuditResponseLimitInvalid = errors.New("assessment audit response limit must be positive")

type assessmentAuditLimitedResponseWriter struct {
	http.ResponseWriter
	maxBytes int64
	written  int64
	exceeded bool
}

func (w *assessmentAuditLimitedResponseWriter) WriteHeader(status int) {
	if w.exceeded {
		return
	}
	w.ResponseWriter.WriteHeader(status)
}

func (w *assessmentAuditLimitedResponseWriter) Write(p []byte) (int, error) {
	remaining := w.maxBytes - w.written
	if remaining <= 0 {
		w.exceeded = true
		return 0, http.ErrHandlerTimeout
	}
	if int64(len(p)) > remaining {
		w.exceeded = true
		n, err := w.ResponseWriter.Write(p[:remaining])
		w.written += int64(n)
		if err != nil {
			return n, err
		}
		return n, http.ErrHandlerTimeout
	}

	n, err := w.ResponseWriter.Write(p)
	w.written += int64(n)
	return n, err
}

// AssessmentAuditResponseSizeMiddleware limits response bytes written by the
// wrapped handler. Prefer endpoint-specific pagination limits as the primary
// protection against oversized responses.
func AssessmentAuditResponseSizeMiddleware(
	next http.Handler,
	maxBytes int64,
) (http.Handler, error) {
	if maxBytes <= 0 {
		return nil, ErrAssessmentAuditResponseLimitInvalid
	}
	if next == nil {
		next = http.NotFoundHandler()
	}

	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		limited := &assessmentAuditLimitedResponseWriter{
			ResponseWriter: w,
			maxBytes:       maxBytes,
		}
		next.ServeHTTP(limited, r)
	}), nil
}

func ParseAssessmentAuditOrigins(raw string) ([]string, error) {

		}

			return nil, errors.Join(ErrAssessmentAuditOriginListInvalid, err)
		}
		if _, exists := seen[origin]; exists {
			continue
		}
		seen[origin] = struct{}{}
		origins = append(origins, origin)
	}

	return origins, nil

import (
	"net/http"
	"strings"

// AssessmentAuditAllowedHostMiddleware checks the request Host against an
// explicit allowlist. It does not trust forwarded-host headers.
func AssessmentAuditAllowedHostMiddleware(
	next http.Handler,
	allowedHosts []string,
) http.Handler {
	if next == nil {
		next = http.NotFoundHandler()
	}

	allowed := make(map[string]struct{}, len(allowedHosts))
	for _, host := range allowedHosts {
		host = strings.ToLower(strings.TrimSpace(host))
		host = strings.TrimSuffix(host, ".")
		if host != "" {
			allowed[host] = struct{}{}
		}
	}

	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		host := strings.ToLower(strings.TrimSpace(r.Host))
		host = strings.TrimSuffix(host, ".")

		if len(allowed) == 0 {
			http.Error(w, "host is not configured", http.StatusMisdirectedRequest)
			return
		}
		if _, ok := allowed[host]; !ok {
			http.Error(w, "host not allowed", http.StatusMisdirectedRequest)
			return
		}

		next.ServeHTTP(w, r)
	})
}

package assessment

import (
	"net/http"
)

// AssessmentAuditHTTPSMiddleware redirects plain HTTP requests to HTTPS when
// enabled. Deployments behind a reverse proxy should terminate TLS at the
// trusted proxy and enforce HTTPS there, rather than trusting arbitrary
// X-Forwarded-Proto headers.
func AssessmentAuditHTTPSMiddleware(
	next http.Handler,
	enabled bool,
) http.Handler {
	if next == nil {
		next = http.NotFoundHandler()
	}
	if !enabled {
		return next
	}

	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if r.TLS != nil {
			next.ServeHTTP(w, r)
			return
		}

		host := r.Host
		if host == "" {
			http.Error(w, "HTTPS required", http.StatusBadRequest)
			return
		}

		target := "https://" + host + r.URL.RequestURI()
		http.Redirect(w, r, target, http.StatusPermanentRedirect)
	})
}

// ValidateAssessmentAuditHeader rejects control characters and values exceeding
// the specified byte limit. Header names must be supplied by trusted code.
func ValidateAssessmentAuditHeader(r *http.Request, name string, maxBytes int) error {
	if r == nil || strings.TrimSpace(name) == "" || maxBytes < 1 {
		return ErrAssessmentAuditHeaderInvalid
	}

	value := r.Header.Get(name)
	if len(value) > maxBytes {
		return ErrAssessmentAuditHeaderInvalid
	}

	for _, ch := range value {
		if ch < 0x20 || ch == 0x7f {
			return ErrAssessmentAuditHeaderInvalid
		}
	}

	return nil
}

var ErrAssessmentAuditBodyDrainLimitInvalid = errors.New("assessment audit body drain limit must be non-negative")

// DrainAssessmentAuditRequestBody drains at most maxBytes bytes and closes the
// request body. It avoids unbounded draining of attacker-controlled input.
func DrainAssessmentAuditRequestBody(r *http.Request, maxBytes int64) error {
	if r == nil {
		return errors.New("request is nil")
	}
	if maxBytes < 0 {
		return ErrAssessmentAuditBodyDrainLimitInvalid
	}
	if r.Body == nil {
		return nil
	}

	_, drainErr := io.Copy(io.Discard, io.LimitReader(r.Body, maxBytes))
	closeErr := r.Body.Close()
	if drainErr != nil {
		return drainErr
	}
	return closeErr
}
