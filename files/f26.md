package identity

import (
	"context"
	"errors"
	"time"
)

var (
	ErrInvalidVerificationToken = errors.New("invalid verification token")
	ErrVerificationTokenExpired = errors.New("verification token expired")
	ErrVerificationTokenUsed    = errors.New("verification token already used")
)

type EmailVerificationToken struct {
	ID         string
	UserID     string
	TokenDigest string
	CreatedAt  time.Time
	ExpiresAt  time.Time
	ConsumedAt *time.Time
}

func (t EmailVerificationToken) Validate() error {
	if t.ID == "" || t.UserID == "" {
		return errors.New("verification token ID and user ID are required")
	}
	if len(t.TokenDigest) != 64 {
		return errors.New("verification token digest must be 64 characters")
	}
	if t.CreatedAt.IsZero() || !t.ExpiresAt.After(t.CreatedAt) {
		return errors.New("verification token expiry must follow creation")
	}
	if t.ConsumedAt != nil && t.ConsumedAt.Before(t.CreatedAt) {
		return errors.New("verification token consumption precedes creation")
	}
	return nil
}

func (t EmailVerificationToken) IsActive(now time.Time) bool {
	return t.ConsumedAt == nil && now.Before(t.ExpiresAt)
}

type VerificationTokenRepository interface {
	CreateVerificationToken(context.Context, EmailVerificationToken) error
	GetVerificationTokenByDigest(context.Context, string) (EmailVerificationToken, error)
	ConsumeVerificationToken(context.Context, string, time.Time) error
	InvalidateUserVerificationTokens(context.Context, string, time.Time) error
}

import (
	"context"
	"errors"
	"fmt"
	"time"

	"github.com/jackc/pgx/v5"
	"github.com/jackc/pgx/v5/pgxpool"

var ErrVerificationTokenNotFound = errors.New("verification token not found")

type PostgresVerificationTokenRepository struct {
	pool *pgxpool.Pool
}

var _ VerificationTokenRepository = (*PostgresVerificationTokenRepository)(nil)

func NewPostgresVerificationTokenRepository(
	pool *pgxpool.Pool,
) (*PostgresVerificationTokenRepository, error) {
	if pool == nil {
		return nil, errors.New("verification repository: pool is required")
	}
	return &PostgresVerificationTokenRepository{pool: pool}, nil
}

func (r *PostgresVerificationTokenRepository) CreateVerificationToken(
	ctx context.Context,
	token EmailVerificationToken,
) error {
	if ctx == nil {
		return errors.New("verification repository: context is required")
	}
	if err := token.Validate(); err != nil {
		return fmt.Errorf("validate verification token: %w", err)
	}

	_, err := r.pool.Exec(ctx, `
		INSERT INTO email_verification_tokens (
			id, user_id, token_digest, created_at, expires_at,
			consumed_at
		)
		VALUES ($1, $2, $3, $4, $5, $6)
	`,
		token.ID,
		token.UserID,
		token.TokenDigest,
		token.CreatedAt,
		token.ExpiresAt,
		token.ConsumedAt,
	)
	return err
}

func (r *PostgresVerificationTokenRepository) GetVerificationTokenByDigest(
	ctx context.Context,
	digest string,
) (EmailVerificationToken, error) {
	if ctx == nil {
		return EmailVerificationToken{}, errors.New("verification repository: context is required")
	}

	var token EmailVerificationToken
	err := r.pool.QueryRow(ctx, `
		SELECT id, user_id, token_digest, created_at, expires_at, consumed_at
		FROM email_verification_tokens
		WHERE token_digest = $1
	`, digest).Scan(
		&token.ID,
		&token.UserID,
		&token.TokenDigest,
		&token.CreatedAt,
		&token.ExpiresAt,
		&token.ConsumedAt,
	)

	if errors.Is(err, pgx.ErrNoRows) {
		return EmailVerificationToken{}, ErrVerificationTokenNotFound
	}
	return token, err
}

func (r *PostgresVerificationTokenRepository) ConsumeVerificationToken(
	ctx context.Context,
	digest string,
	consumedAt time.Time,
) error {
	if ctx == nil {
		return errors.New("verification repository: context is required")
	}

	tag, err := r.pool.Exec(ctx, `
		UPDATE email_verification_tokens
		SET consumed_at = $2
		WHERE token_digest = $1
		  AND consumed_at IS NULL
		  AND expires_at > $2
	`, digest, consumedAt)
	if err != nil {
		return err
	}
	if tag.RowsAffected() == 0 {
		return ErrInvalidVerificationToken
	}
	return nil
}

func (r *PostgresVerificationTokenRepository) InvalidateUserVerificationTokens(
	ctx context.Context,
	userID string,
	invalidatedAt time.Time,
) error {
	if ctx == nil {
		return errors.New("verification repository: context is required")
	}

	_, err := r.pool.Exec(ctx, `
		UPDATE email_verification_tokens
		SET consumed_at = $2
		WHERE user_id = $1
		  AND consumed_at IS NULL
	`, userID, invalidatedAt)
	return err
}

package identity

import (
	"context"
	"errors"
	"time"

	"github.com/kdxlabs/kdxlabsllm/internal/platform/security"
)

type EmailVerificationService struct {
	users      UserRepository
	tokens     VerificationTokenRepository
	clock      Clock
	tokenTTL   time.Duration
}

type IssueVerificationResult struct {
	UserID    string
	Token     string
	ExpiresAt time.Time
}

func NewEmailVerificationService(
	users UserRepository,
	tokens VerificationTokenRepository,
	clock Clock,
	tokenTTL time.Duration,
) (*EmailVerificationService, error) {
	if users == nil || tokens == nil || clock == nil {
		return nil, errors.New("verification service: users, tokens and clock are required")
	}
	if tokenTTL < time.Minute || tokenTTL > 24*time.Hour {
		return nil, errors.New("verification service: token TTL must be between 1 minute and 24 hours")
	}

	return &EmailVerificationService{
		users: users,
		tokens: tokens,
		clock: clock,
		tokenTTL: tokenTTL,
	}, nil
}

// Issue creates a one-time verification token. The caller must deliver the
// returned raw token through a trusted email-delivery provider.
func (s *EmailVerificationService) Issue(
	ctx context.Context,
	userID string,
) (IssueVerificationResult, error) {
	if ctx == nil {
		return IssueVerificationResult{}, errors.New("verification service: context is required")
	}
	if userID == "" {
		return IssueVerificationResult{}, ErrInvalidUserID
	}

	user, err := s.users.GetByID(ctx, userID)
	if err != nil {
		return IssueVerificationResult{}, err
	}
	if user.Status == UserStatusDisabled || user.Status == UserStatusSuspended {
		return IssueVerificationResult{}, ErrAccountDisabled
	}
	if user.EmailVerifiedAt != nil {
		return IssueVerificationResult{}, ErrInvalidUserState
	}

	now := s.clock.Now()
	rawToken, err := security.GenerateOpaqueToken(32)
	if err != nil {
		return IssueVerificationResult{}, err
	}
	digest, err := security.DigestToken(rawToken)
	if err != nil {
		return IssueVerificationResult{}, err
	}
	tokenID, err := NewID("evt")
	if err != nil {
		return IssueVerificationResult{}, err
	}

	if err := s.tokens.InvalidateUserVerificationTokens(ctx, userID, now); err != nil {
		return IssueVerificationResult{}, err
	}

	record := EmailVerificationToken{
		ID: tokenID,
		UserID: userID,
		TokenDigest: digest,
		CreatedAt: now,
		ExpiresAt: now.Add(s.tokenTTL),
	}
	if err := s.tokens.CreateVerificationToken(ctx, record); err != nil {
		return IssueVerificationResult{}, err
	}

	return IssueVerificationResult{
		UserID: userID,
		Token: rawToken,
		ExpiresAt: record.ExpiresAt,
	}, nil
}

func (s *EmailVerificationService) Verify(
	ctx context.Context,
	rawToken string,
) error {
	if ctx == nil {
		return errors.New("verification service: context is required")
	}
	if rawToken == "" {
		return ErrInvalidVerificationToken
	}

	digest, err := security.DigestToken(rawToken)
	if err != nil {
		return ErrInvalidVerificationToken
	}

	record, err := s.tokens.GetVerificationTokenByDigest(ctx, digest)
	if err != nil {
		if errors.Is(err, ErrVerificationTokenNotFound) {
			return ErrInvalidVerificationToken
		}
		return err
	}

	now := s.clock.Now()
	if record.ConsumedAt != nil {
		return ErrVerificationTokenUsed
	}
	if !now.Before(record.ExpiresAt) {
		return ErrVerificationTokenExpired
	}

	if err := s.tokens.ConsumeVerificationToken(ctx, digest, now); err != nil {
		if errors.Is(err, ErrInvalidVerificationToken) {
			return ErrVerificationTokenUsed
		}
		return err
	}

	return s.users.MarkEmailVerified(ctx, record.UserID, now)
}

import (
	"context"
	"errors"
	"net/http"

	"github.com/gin-gonic/gin"
	"github.com/kdxlabs/kdxlabsllm/internal/platform/httpx"
)

type EmailVerificationServiceAPI interface {
	Verify(context.Context, string) error
}

type VerificationHandler struct {
	service EmailVerificationServiceAPI
}

func NewVerificationHandler(service EmailVerificationServiceAPI) *VerificationHandler {
	return &VerificationHandler{service: service}
}

type verifyEmailRequest struct {
	Token string `json:"token" validate:"required,min=20,max=4096"`
}

func (h *VerificationHandler) VerifyEmail(c *gin.Context) {
	var request verifyEmailRequest
	if !httpx.BindJSON(c, &request) {
		return
	}

	err := h.service.Verify(c.Request.Context(), request.Token)
	if err != nil {
		switch {
		case errors.Is(err, ErrInvalidVerificationToken),
			errors.Is(err, ErrVerificationTokenNotFound):
			httpx.BadRequest(
				c,
				"INVALID_VERIFICATION_TOKEN",
				"The verification token is invalid.",
			)
		case errors.Is(err, ErrVerificationTokenExpired):
			httpx.BadRequest(
				c,
				"VERIFICATION_TOKEN_EXPIRED",
				"The verification token has expired.",
			)
		case errors.Is(err, ErrVerificationTokenUsed):
			httpx.Conflict(
				c,
				"VERIFICATION_TOKEN_USED",
				"The verification token has already been used.",
			)
		default:
			httpx.InternalServerError(c)
		}
		return
	}

	httpx.JSON(c, http.StatusOK, gin.H{
		"message": "Email verified successfully.",
	})
}

package identity

import (
	"context"
	"errors"
	"time"
)

var (
	ErrInvalidPasswordResetToken = errors.New("invalid password reset token")
	ErrPasswordResetExpired      = errors.New("password reset token expired")
	ErrPasswordResetUsed          = errors.New("password reset token already used")
)

type PasswordResetToken struct {
	ID          string
	UserID      string
	TokenDigest string
	CreatedAt   time.Time
	ExpiresAt   time.Time
	ConsumedAt  *time.Time
}

func (t PasswordResetToken) Validate() error {
	if t.ID == "" || t.UserID == "" {
		return errors.New("password reset token ID and user ID are required")
	}
	if len(t.TokenDigest) != 64 {
		return errors.New("password reset token digest must be 64 characters")
	}
	if t.CreatedAt.IsZero() || !t.ExpiresAt.After(t.CreatedAt) {
		return errors.New("password reset expiry must follow creation")
	}
	if t.ConsumedAt != nil && t.ConsumedAt.Before(t.CreatedAt) {
		return errors.New("password reset consumption precedes creation")
	}
	return nil
}

func (t PasswordResetToken) IsActive(now time.Time) bool {
	return t.ConsumedAt == nil && now.Before(t.ExpiresAt)
}

type PasswordResetTokenRepository interface {
	CreatePasswordResetToken(context.Context, PasswordResetToken) error
	GetPasswordResetTokenByDigest(context.Context, string) (PasswordResetToken, error)
	ConsumePasswordResetToken(context.Context, string, time.Time) error
	InvalidateUserPasswordResetTokens(context.Context, string, time.Time) error
}

import (
	"context"
	"errors"
	"fmt"
	"time"

	"github.com/jackc/pgx/v5"
	"github.com/jackc/pgx/v5/pgxpool"

var ErrPasswordResetTokenNotFound = errors.New("password reset token not found")

type PostgresPasswordResetTokenRepository struct {
	pool *pgxpool.Pool
}

var _ PasswordResetTokenRepository = (*PostgresPasswordResetTokenRepository)(nil)

func NewPostgresPasswordResetTokenRepository(
	pool *pgxpool.Pool,
) (*PostgresPasswordResetTokenRepository, error) {
	if pool == nil {
		return nil, errors.New("password reset repository: pool is required")
	}
	return &PostgresPasswordResetTokenRepository{pool: pool}, nil
}

func (r *PostgresPasswordResetTokenRepository) CreatePasswordResetToken(
	ctx context.Context,
	token PasswordResetToken,
) error {
	if ctx == nil {
		return errors.New("password reset repository: context is required")
	}
	if err := token.Validate(); err != nil {
		return fmt.Errorf("validate password reset token: %w", err)
	}

	_, err := r.pool.Exec(ctx, `
		INSERT INTO password_reset_tokens (
			id, user_id, token_digest, created_at, expires_at, consumed_at
		)
		VALUES ($1, $2, $3, $4, $5, $6)
	`,
		token.ID,
		token.UserID,
		token.TokenDigest,
		token.CreatedAt,
		token.ExpiresAt,
		token.ConsumedAt,
	)
	return err
}

func (r *PostgresPasswordResetTokenRepository) GetPasswordResetTokenByDigest(
	ctx context.Context,
	digest string,
) (PasswordResetToken, error) {
	if ctx == nil {
		return PasswordResetToken{}, errors.New("password reset repository: context is required")
	}

	var token PasswordResetToken
	err := r.pool.QueryRow(ctx, `
		SELECT id, user_id, token_digest, created_at, expires_at, consumed_at
		FROM password_reset_tokens
		WHERE token_digest = $1
	`, digest).Scan(
		&token.ID,
		&token.UserID,
		&token.TokenDigest,
		&token.CreatedAt,
		&token.ExpiresAt,
		&token.ConsumedAt,
	)

	if errors.Is(err, pgx.ErrNoRows) {
		return PasswordResetToken{}, ErrPasswordResetTokenNotFound
	}
	return token, err
}

func (r *PostgresPasswordResetTokenRepository) ConsumePasswordResetToken(
	ctx context.Context,
	digest string,
	consumedAt time.Time,
) error {
	if ctx == nil {
		return errors.New("password reset repository: context is required")
	}

	tag, err := r.pool.Exec(ctx, `
		UPDATE password_reset_tokens
		SET consumed_at = $2
		WHERE token_digest = $1
		  AND consumed_at IS NULL
		  AND expires_at > $2
	`, digest, consumedAt)
	if err != nil {
		return err
	}
	if tag.RowsAffected() == 0 {
		return ErrInvalidPasswordResetToken
	}
	return nil
}

func (r *PostgresPasswordResetTokenRepository) InvalidateUserPasswordResetTokens(
	ctx context.Context,
	userID string,
	invalidatedAt time.Time,
) error {
	if ctx == nil {
		return errors.New("password reset repository: context is required")
	}

	_, err := r.pool.Exec(ctx, `
		UPDATE password_reset_tokens
		SET consumed_at = $2
		WHERE user_id = $1
		  AND consumed_at IS NULL
	`, userID, invalidatedAt)
	return err
}

package identity

import (
	"context"
	"errors"
	"time"

	"github.com/kdxlabs/kdxlabsllm/internal/platform/security"
)

type PasswordResetNotifier interface {
	SendPasswordReset(
		ctx context.Context,
		email string,
		token string,
		expiresAt time.Time,
	) error
}

type PasswordResetService struct {
	users    UserRepository
	tokens   PasswordResetTokenRepository
	sessions SessionRepository
	clock    Clock
	notifier PasswordResetNotifier
	tokenTTL time.Duration
}

func NewPasswordResetService(
	users UserRepository,
	tokens PasswordResetTokenRepository,
	sessions SessionRepository,
	clock Clock,
	notifier PasswordResetNotifier,
	tokenTTL time.Duration,
) (*PasswordResetService, error) {
	if users == nil || tokens == nil || sessions == nil || clock == nil || notifier == nil {
		return nil, errors.New("password reset service: all dependencies are required")
	}
	if tokenTTL < time.Minute || tokenTTL > time.Hour {
		return nil, errors.New("password reset service: token TTL must be between 1 minute and 1 hour")
	}

	return &PasswordResetService{
		users: users,
		tokens: tokens,
		sessions: sessions,
		clock: clock,
		notifier: notifier,
		tokenTTL: tokenTTL,
	}, nil
}

// RequestReset deliberately returns success for unknown email addresses.
// This prevents the endpoint from revealing whether an account exists.
func (s *PasswordResetService) RequestReset(
	ctx context.Context,
	rawEmail string,
) error {
	if ctx == nil {
		return errors.New("password reset service: context is required")
	}

	email, err := NormalizeEmail(rawEmail)
	if err != nil {
		// Invalid addresses are not actionable, but do not disclose account state.
		return nil
	}

	user, err := s.users.GetByEmail(ctx, email)
	if err != nil {
		if errors.Is(err, ErrUserNotFound) {
			return nil
		}
		return err
	}
	if user.Status != UserStatusActive {
		return nil
	}

	now := s.clock.Now()
	rawToken, err := security.GenerateOpaqueToken(32)
	if err != nil {
		return err
	}
	digest, err := security.DigestToken(rawToken)
	if err != nil {
		return err
	}
	id, err := NewID("prt")
	if err != nil {
		return err
	}

	if err := s.tokens.InvalidateUserPasswordResetTokens(ctx, user.ID, now); err != nil {
		return err
	}

	record := PasswordResetToken{
		ID: id,
		UserID: user.ID,
		TokenDigest: digest,
		CreatedAt: now,
		ExpiresAt: now.Add(s.tokenTTL),
	}
	if err := s.tokens.CreatePasswordResetToken(ctx, record); err != nil {
		return err
	}

	if err := s.notifier.SendPasswordReset(ctx, user.Email, rawToken, record.ExpiresAt); err != nil {
		// Do not return the token to callers or log it. Invalidate it if delivery fails.
		_ = s.tokens.InvalidateUserPasswordResetTokens(ctx, user.ID, s.clock.Now())
		return err
	}

	return nil
}

func (s *PasswordResetService) ResetPassword(
	ctx context.Context,
	rawToken string,
	newPassword string,
) error {
	if ctx == nil {
		return errors.New("password reset service: context is required")
	}
	if rawToken == "" {
		return ErrInvalidPasswordResetToken
	}
	if err := ValidatePassword(newPassword); err != nil {
		return err
	}

	digest, err := security.DigestToken(rawToken)
	if err != nil {
		return ErrInvalidPasswordResetToken
	}

	record, err := s.tokens.GetPasswordResetTokenByDigest(ctx, digest)
	if err != nil {
		if errors.Is(err, ErrPasswordResetTokenNotFound) {
			return ErrInvalidPasswordResetToken
		}
		return err
	}

	now := s.clock.Now()
	if record.ConsumedAt != nil {
		return ErrPasswordResetUsed
	}
	if !now.Before(record.ExpiresAt) {
		return ErrPasswordResetExpired
	}

	passwordHash, err := security.HashPassword(newPassword)
	if err != nil {
		return err
	}

	// Consume before changing the password so concurrent reuse is rejected.
	if err := s.tokens.ConsumePasswordResetToken(ctx, digest, now); err != nil {
		if errors.Is(err, ErrInvalidPasswordResetToken) {
			return ErrPasswordResetUsed
		}
		return err
	}

	if err := s.users.UpdatePasswordHash(ctx, record.UserID, passwordHash, now); err != nil {
		return err
	}

	// A successful reset invalidates every existing login session.
	return s.sessions.RevokeAllUserSessions(ctx, record.UserID, now)
}

import (
	"context"
	"errors"
	"net/http"

	"github.com/gin-gonic/gin"
	"github.com/kdxlabs/kdxlabsllm/internal/platform/httpx"
)

type PasswordResetServiceAPI interface {
	RequestReset(context.Context, string) error
	ResetPassword(context.Context, string, string) error
}

type PasswordResetHandler struct {
	service PasswordResetServiceAPI
}

func NewPasswordResetHandler(service PasswordResetServiceAPI) *PasswordResetHandler {
	return &PasswordResetHandler{service: service}
}

type requestPasswordResetRequest struct {
	Email string `json:"email" validate:"required,email,max=254"`
}

type confirmPasswordResetRequest struct {
	Token       string `json:"token" validate:"required,min=20,max=4096"`
	NewPassword string `json:"new_password" validate:"required,min=12,max=128"`
}

func (h *PasswordResetHandler) RequestReset(c *gin.Context) {
	var request requestPasswordResetRequest
	if !httpx.BindJSON(c, &request) {
		return
	}

	if err := h.service.RequestReset(c.Request.Context(), request.Email); err != nil {
		httpx.InternalServerError(c)
		return
	}

	// Same response for known and unknown addresses.
	httpx.JSON(c, http.StatusAccepted, gin.H{
		"message": "If an eligible account exists, password reset instructions will be sent.",
	})
}

func (h *PasswordResetHandler) ConfirmReset(c *gin.Context) {
	var request confirmPasswordResetRequest
	if !httpx.BindJSON(c, &request) {
		return
	}

	err := h.service.ResetPassword(
		c.Request.Context(),
		request.Token,
		request.NewPassword,
	)
	if err != nil {
		switch {
		case errors.Is(err, ErrInvalidPasswordResetToken):
			httpx.BadRequest(c, "INVALID_RESET_TOKEN", "The password reset token is invalid.")
		case errors.Is(err, ErrPasswordResetExpired):
			httpx.BadRequest(c, "RESET_TOKEN_EXPIRED", "The password reset token has expired.")
		case errors.Is(err, ErrPasswordResetUsed):
			httpx.Conflict(c, "RESET_TOKEN_USED", "The password reset token has already been used.")
		case errors.Is(err, ErrWeakPassword):
			httpx.BadRequest(c, "WEAK_PASSWORD", "The new password does not meet security requirements.")
		default:
			httpx.InternalServerError(c)
		}
		return
	}

	httpx.JSON(c, http.StatusOK, gin.H{
		"message": "Password reset successfully. Please sign in again.",
	})
}
