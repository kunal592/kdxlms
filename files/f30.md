func NewPostgresLessonRepository(db *pgxpool.Pool) (*PostgresLessonRepository, error) {
	if db == nil {

const lessonColumns = `
	id, tenant_id, course_id, title, description, lesson_type,

func scanLesson(row pgx.Row) (Lesson, error) {
	var lesson Lesson
	var lessonType, status string

	err := row.Scan(
		&lesson.ID,
		&lesson.TenantID,
		&lesson.CourseID,
		&lesson.Title,
		&lesson.Description,
		&lessonType,
		&lesson.Content,
		&lesson.ContentURL,
		&lesson.Position,
		&status,
		&lesson.PublishedAt,
		&lesson.CreatedAt,
		&lesson.UpdatedAt,
	)
	if errors.Is(err, pgx.ErrNoRows) {
		return Lesson{}, ErrLessonNotFound
	}
	if err != nil {
		return Lesson{}, fmt.Errorf("scan lesson: %w", err)
	}

	lesson.Type = LessonType(lessonType)
	lesson.Status = LessonStatus(status)
	return lesson, nil
}

func (r *PostgresLessonRepository) CreateLesson(
	ctx context.Context,
	lesson Lesson,
) error {
	if ctx == nil {
		return errors.New("lesson repository: context is required")
	}
	if err := lesson.Validate(); err != nil {
		return err
	}

	_, err := r.db.Exec(ctx, `
		INSERT INTO course_lessons (
			id, tenant_id, course_id, title, description, lesson_type,
			content, content_url, position, status, published_at,
			created_at, updated_at
		) VALUES (
			$1,$2,$3,$4,$5,$6,$7,NULLIF($8,''),$9,$10,$11,$12,$13
		)
	`,
		lesson.ID, lesson.TenantID, lesson.CourseID, lesson.Title,
		lesson.Description, string(lesson.Type), lesson.Content,
		lesson.ContentURL, lesson.Position, string(lesson.Status),
		lesson.PublishedAt, lesson.CreatedAt, lesson.UpdatedAt,
	)
	if err != nil {
		return mapLessonWriteError(err)
	}
	return nil
}

func (r *PostgresLessonRepository) GetLesson(
	ctx context.Context,
	tenantID, courseID, lessonID string,
) (Lesson, error) {
	if ctx == nil {
		return Lesson{}, errors.New("lesson repository: context is required")
	}
	return scanLesson(r.db.QueryRow(ctx, `
		SELECT `+lessonColumns+`
		FROM course_lessons
		WHERE tenant_id = $1 AND course_id = $2 AND id = $3
	`, tenantID, courseID, lessonID))

func (r *PostgresLessonRepository) ListLessons(
	ctx context.Context,
	tenantID, courseID string,
	publishedOnly bool,
	limit, offset int,
) ([]Lesson, int64, error) {
	if ctx == nil {
		return nil, 0, errors.New("lesson repository: context is required")
	}
	if strings.TrimSpace(tenantID) == "" ||
		strings.TrimSpace(courseID) == "" ||
		limit < 1 || limit > 100 || offset < 0 {
		return nil, 0, errors.New("lesson repository: invalid list parameters")
	}

	statusFilter := ""
	if publishedOnly {
		statusFilter = " AND status = 'published' "
	}

	countQuery := `
		SELECT COUNT(*)
		FROM course_lessons
		WHERE tenant_id = $1 AND course_id = $2 ` + statusFilter

	var total int64
	if err := r.db.QueryRow(ctx, countQuery, tenantID, courseID).Scan(&total); err != nil {
		return nil, 0, fmt.Errorf("count lessons: %w", err)
	}

	query := `
		SELECT ` + lessonColumns + `
		FROM course_lessons
		WHERE tenant_id = $1 AND course_id = $2 ` + statusFilter + `
		ORDER BY position ASC, id ASC
		LIMIT $3 OFFSET $4
	`
	rows, err := r.db.Query(ctx, query, tenantID, courseID, limit, offset)
	if err != nil {
		return nil, 0, fmt.Errorf("list lessons: %w", err)
	}
	defer rows.Close()

	lessons := make([]Lesson, 0)
	for rows.Next() {
		lesson, scanErr := scanLesson(rows)
		if scanErr != nil {
			return nil, 0, scanErr
		}
		lessons = append(lessons, lesson)
	}
	if err := rows.Err(); err != nil {
		return nil, 0, fmt.Errorf("iterate lessons: %w", err)
	}

	return lessons, total, nil
}

	lesson Lesson,

		return mapLessonWriteError(err)
	}
	if tag.RowsAffected() == 0 {
		return ErrLessonNotFound
	}
	return nil

func (r *PostgresLessonRepository) SetLessonStatus(
	ctx context.Context,
	tenantID, courseID, lessonID string,
	status LessonStatus,
	publishedAt *time.Time,
	updatedAt time.Time,
) error {
	if ctx == nil {
		return errors.New("lesson repository: context is required")
	}
	if status != LessonStatusDraft && status != LessonStatusPublished {
		return ErrLessonInvalidState
	}
	if status == LessonStatusPublished && publishedAt == nil {
		return errors.New("published lessons require a publication timestamp")
	}
	if status == LessonStatusDraft {
		publishedAt = nil
	}

	tag, err := r.db.Exec(ctx, `
		UPDATE course_lessons
		SET status = $4, published_at = $5, updated_at = $6
		WHERE tenant_id = $1 AND course_id = $2 AND id = $3
	`,
		tenantID, courseID, lessonID, string(status), publishedAt, updatedAt.UTC(),
	)
	if err != nil {
		return mapLessonWriteError(err)
	}
	if tag.RowsAffected() == 0 {
		return ErrLessonNotFound
	}
	return nil
}

func mapLessonWriteError(err error) error {
	var pgErr *pgconn.PgError
	if errors.As(err, &pgErr) {
		switch {
		case pgErr.Code == "23505" &&
			pgErr.ConstraintName == "course_lessons_course_position_unique":
			return errors.New("lesson position already exists")
		case pgErr.Code == "23503" &&
			pgErr.ConstraintName == "course_lessons_course_fk":
			return ErrLessonInvalidID
		case pgErr.Code == "23514" &&
			strings.HasPrefix(pgErr.ConstraintName, "course_lessons_"):
			return ErrLessonInvalidState
		}
	}
	return fmt.Errorf("lesson database operation: %w", err)
}

	ContentURL  string
	Position    int

func (s *LessonService) Create(
	ctx context.Context,
	input CreateLessonInput,
) (Lesson, error) {
	if ctx == nil {
		return Lesson{}, errors.New("lesson service: context is required")
	}

	tenantID := strings.TrimSpace(input.TenantID)
	courseID := strings.TrimSpace(input.CourseID)
	if tenantID == "" || courseID == "" {
		return Lesson{}, ErrLessonInvalidID
	}

	parent, err := s.courses.GetByID(ctx, tenantID, courseID)
	if err != nil {
		return Lesson{}, err
	}
	if parent.Status == StatusArchived {
		return Lesson{}, ErrAccessDenied
	}

	title := strings.TrimSpace(input.Title)
	description := strings.TrimSpace(input.Description)
	contentURL, err := validateLessonURL(input.Type, input.ContentURL)
	if err != nil {
		return Lesson{}, err
	}

	now := s.clock.Now().UTC()
	lesson := Lesson{
		ID: s.newID("les"),
		TenantID: tenantID,
		CourseID: courseID,
		Title: title,
		Description: description,
		Type: input.Type,
		Content: input.Content,
		ContentURL: contentURL,
		Position: input.Position,
		Status: LessonStatusDraft,
		CreatedAt: now,
		UpdatedAt: now,
	}
	if err := lesson.Validate(); err != nil {
		return Lesson{}, err
	}
	if err := s.lessons.CreateLesson(ctx, lesson); err != nil {
		return Lesson{}, err
	}
	return lesson, nil
}

	ctx context.Context,
	tenantID, courseID, lessonID string,

	if err != nil {
		return nil, 0, err
	}

	)

func (s *LessonService) Update(
	ctx context.Context,
	tenantID, courseID, lessonID string,
	input UpdateLessonInput,
) error {
	if ctx == nil {
		return errors.New("lesson service: context is required")
	}
	lesson, err := s.lessons.GetLesson(ctx, tenantID, courseID, lessonID)
	if err != nil {
		return err
	}
	if lesson.Status == LessonStatusPublished {
		return ErrAccessDenied
	}

	contentURL, err := validateLessonURL(input.Type, input.ContentURL)
	if err != nil {
		return err
	}

	lesson.Title = strings.TrimSpace(input.Title)
	lesson.Description = strings.TrimSpace(input.Description)
	lesson.Type = input.Type
	lesson.Content = input.Content
	lesson.ContentURL = contentURL
	lesson.Position = input.Position
	lesson.UpdatedAt = s.clock.Now().UTC()

	if err := lesson.Validate(); err != nil {
		return err
	}
	return s.lessons.UpdateLesson(ctx, lesson)
}

func (s *LessonService) Publish(
	ctx context.Context,
	tenantID, courseID, lessonID string,
) error {
	if ctx == nil {
		return errors.New("lesson service: context is required")
	}
	parent, err := s.courses.GetByID(ctx, tenantID, courseID)
	if err != nil {
		return err
	}
	if parent.Status != StatusPublished {
		return ErrAccessDenied
	}

	now := s.clock.Now().UTC()
	return s.lessons.SetLessonStatus(
		ctx, tenantID, courseID, lessonID,
		LessonStatusPublished, &now, now,
	)
}

	ctx context.Context,
	tenantID, courseID, lessonID string,
) error {

		return errors.New("lesson service: context is required")
	}
	now := s.clock.Now().UTC()
	return s.lessons.SetLessonStatus(
		ctx, tenantID, courseID, lessonID,
		LessonStatusDraft, nil, now,
	)
}

func validateLessonURL(kind LessonType, raw string) (string, error) {
	raw = strings.TrimSpace(raw)
	if kind == LessonTypeText {
		if raw != "" {
			return "", errors.New("text lessons must not provide a content URL")
		}
		return "", nil
	}
	if kind != LessonTypeVideo && kind != LessonTypeLink {
		return "", ErrLessonInvalidType
	}

	parsed, err := url.ParseRequestURI(raw)
	if err != nil || parsed.Host == "" ||
		(parsed.Scheme != "https" && parsed.Scheme != "http") ||
		parsed.User != nil {
		return "", errors.New("lesson URL must be a valid HTTP or HTTPS URL")
	}
	return parsed.String(), nil
}

func (h *LessonHTTPHandler) Create(c *gin.Context) {
	var request createLessonRequest
	if err := c.ShouldBindJSON(&request); err != nil {
		c.JSON(http.StatusBadRequest, gin.H{"error": "INVALID_REQUEST"})

		tenant.RequireTenantPermission(authorizer, tenant.PermissionCourseUpdate),
		handler.Create,
	)
	lessons.GET(
		"/:lesson_id",
		tenant.RequireTenantPermission(authorizer, tenant.PermissionCourseRead),
		handler.Get,
	)
	lessons.PUT(
		"/:lesson_id",
		tenant.RequireTenantPermission(authorizer, tenant.PermissionCourseUpdate),
		handler.Update,
	)
	lessons.POST(
		"/:lesson_id/publish",
		tenant.RequireTenantPermission(authorizer, tenant.PermissionCoursePublish),
		handler.Publish,
	)
	lessons.POST(
		"/:lesson_id/unpublish",

	Status      Status     `json:"status"`

	ErrNotFound          = errors.New("enrollment not found")
	ErrAlreadyExists     = errors.New("user is already enrolled")

		strings.TrimSpace(e.TenantID) == "" ||
		strings.TrimSpace(e.CourseID) == "" ||
		strings.TrimSpace(e.UserID) == "" {
		return ErrInvalidID
	}
	if e.EnrolledAt.IsZero() || e.UpdatedAt.Before(e.EnrolledAt) {
		return errors.New("enrollment timestamps are invalid")

func (s *Service) Enroll(
	ctx context.Context,
	tenantID, courseID, userID string,

	if ctx == nil {
		return Enrollment{}, errors.New("enrollment service: context is required")
	}

	tenantID = strings.TrimSpace(tenantID)
	courseID = strings.TrimSpace(courseID)
	userID = strings.TrimSpace(userID)
	if tenantID == "" || courseID == "" || userID == "" {
		return Enrollment{}, ErrInvalidID
	}

	parent, err := s.courses.GetByID(ctx, tenantID, courseID)
	if err != nil {
		return Enrollment{}, err
	}
	if parent.Status != course.StatusPublished || parent.PublishedAt == nil {
		return Enrollment{}, ErrCourseUnavailable
	}

	if err == nil {
		if existing.Status == StatusActive || existing.Status == StatusCompleted {
			return Enrollment{}, ErrAlreadyExists

		// explicit re-enrollment workflow can be added later.
		return Enrollment{}, ErrAlreadyExists
	}
	if !errors.Is(err, ErrNotFound) {
		return Enrollment{}, err
	}

	record := Enrollment{
		ID: s.newID("enr"),
		TenantID: tenantID,
		CourseID: courseID,
		UserID: userID,
		Status: StatusActive,
		EnrolledAt: now,
		UpdatedAt: now,
	}
	if err := record.Validate(); err != nil {
		return Enrollment{}, err
	}
	if err := s.repository.Create(ctx, record); err != nil {
		if errors.Is(err, ErrAlreadyExists) {
			return Enrollment{}, ErrAlreadyExists
		}
		return Enrollment{}, err
	}
	return record, nil

		return
	}

	page, err := enrollmentQueryInt(c.Query("page"), 1)
	if err != nil {
		c.JSON(http.StatusBadRequest, gin.H{"error": "INVALID_PAGE"})
		return
	}
	pageSize, err := enrollmentQueryInt(c.Query("page_size"), 20)
	if err != nil || pageSize > 100 {
		c.JSON(http.StatusBadRequest, gin.H{"error": "INVALID_PAGE_SIZE"})
		return
	}

	items, total, err := h.service.ListForUser(
		c.Request.Context(),
		c.Param("tenant_id"),
		userID,
		page,
		pageSize,
	)
	if err != nil {
		writeEnrollmentError(c, err)
		return
	}
	c.JSON(http.StatusOK, gin.H{
		"data": items,
		"pagination": gin.H{
			"page": page, "page_size": pageSize, "total": total,
		},
	})
}

func enrollmentCurrentUserID(c *gin.Context) (string, bool) {
	value, exists := c.Get("authenticated_user")
	if !exists {
		return "", false
	}
	user, ok := value.(identity.PublicUser)
	if !ok || user.ID == "" {
		return "", false
	}
	return user.ID, true
}

func enrollmentQueryInt(raw string, fallback int) (int, error) {
	if raw == "" {
		return fallback, nil
	}
	n, err := strconv.Atoi(raw)
	if err != nil || n < 1 {
		return 0, ErrInvalidPagination
	}
	return n, nil
}

func writeEnrollmentError(c *gin.Context, err error) {
	switch {
	case errors.Is(err, ErrInvalidID), errors.Is(err, ErrInvalidPagination):
		c.JSON(http.StatusBadRequest, gin.H{"error": "INVALID_REQUEST"})
	case errors.Is(err, ErrAlreadyExists):
		c.JSON(http.StatusConflict, gin.H{"error": "ALREADY_ENROLLED"})
	case errors.Is(err, ErrCourseUnavailable), errors.Is(err, ErrCourseNotFound):
		// Do not disclose whether a hidden or unpublished course exists.
		c.JSON(http.StatusNotFound, gin.H{"error": "COURSE_NOT_AVAILABLE"})
	case errors.Is(err, ErrNotFound):
		c.JSON(http.StatusNotFound, gin.H{"error": "ENROLLMENT_NOT_FOUND"})
	case errors.Is(err, ErrInvalidTransition):
		c.JSON(http.StatusConflict, gin.H{"error": "INVALID_ENROLLMENT_TRANSITION"})
	case errors.Is(err, ErrAccessDenied):
		c.JSON(http.StatusForbidden, gin.H{"error": "FORBIDDEN"})
	default:
		c.JSON(http.StatusInternalServerError, gin.H{"error": "INTERNAL_ERROR"})
	}
}

	EnrollmentID string     `json:"enrollment_id"`
	LessonID     string     `json:"lesson_id"`
	Completed    bool       `json:"completed"`

	UpdatedAt    time.Time  `json:"updated_at"`
}

var (
	ErrProgressNotFound = errors.New("lesson progress not found")
	ErrInvalidProgress  = errors.New("lesson progress is invalid")
)

func (p LessonProgress) Validate() error {
	if strings.TrimSpace(p.ID) == "" ||
		strings.TrimSpace(p.TenantID) == "" ||
		strings.TrimSpace(p.CourseID) == "" ||
		strings.TrimSpace(p.EnrollmentID) == "" ||
		strings.TrimSpace(p.LessonID) == "" {
		return ErrInvalidProgress
	}

	if p.CreatedAt.IsZero() || p.UpdatedAt.Before(p.CreatedAt) {
		return ErrInvalidProgress
	}

	if p.Completed {
		if p.CompletedAt == nil ||
			p.CompletedAt.Before(p.CreatedAt) ||
			p.CompletedAt.After(p.UpdatedAt) {
			return ErrInvalidProgress
		}
	} else if p.CompletedAt != nil {
		return ErrInvalidProgress
	}

	return nil
}

		ORDER BY created_at ASC, id ASC
		LIMIT $4 OFFSET $5
	`, tenantID, courseID, enrollmentID, limit, offset)
	if err != nil {
		return nil, 0, fmt.Errorf("list lesson progress: %w", err)
	}
	defer rows.Close()

	items := make([]LessonProgress, 0)
	for rows.Next() {
		item, err := scanProgressRow(rows)
		if err != nil {
			return nil, 0, fmt.Errorf("scan lesson progress: %w", err)
		}
		items = append(items, item)
	}
	if err := rows.Err(); err != nil {
		return nil, 0, fmt.Errorf("iterate lesson progress: %w", err)
	}
	return items, total, nil
}

func (r *PostgresProgressRepository) MarkLessonCompleted(
	ctx context.Context,
	tenantID, courseID, enrollmentID, lessonID string,
	completedAt time.Time,
) (LessonProgress, error) {
	if ctx == nil {
		return LessonProgress{}, errors.New("progress repository: context is required")
	}
	if tenantID == "" || courseID == "" || enrollmentID == "" ||
		lessonID == "" || completedAt.IsZero() {
		return LessonProgress{}, ErrInvalidProgress
	}
	completedAt = completedAt.UTC()

	// Completion is monotonic: once completed, a lesson cannot be reset
	// through this repository method.
	row := r.db.QueryRow(ctx, `
		UPDATE lesson_progress
		SET completed = TRUE,
		    completed_at = COALESCE(completed_at, $5),
		    updated_at = GREATEST(updated_at, $5)
		WHERE tenant_id = $1
		  AND course_id = $2
		  AND enrollment_id = $3
		  AND lesson_id = $4
		RETURNING id, tenant_id, course_id, enrollment_id, lesson_id,
		          completed, completed_at, created_at, updated_at
	`, tenantID, courseID, enrollmentID, lessonID, completedAt)

	p, err := scanProgress(row)
	if err == nil {
		return p, nil
	}
	if !errors.Is(err, ErrProgressNotFound) {
		return LessonProgress{}, err
	}
	return LessonProgress{}, ErrProgressNotFound
}

func scanProgress(row pgx.Row) (LessonProgress, error) {
	var p LessonProgress
	if err := row.Scan(
		&p.ID, &p.TenantID, &p.CourseID, &p.EnrollmentID, &p.LessonID,
		&p.Completed, &p.CompletedAt, &p.CreatedAt, &p.UpdatedAt,
	); err != nil {
		if errors.Is(err, pgx.ErrNoRows) {
			return LessonProgress{}, ErrProgressNotFound
		}
		return LessonProgress{}, fmt.Errorf("scan lesson progress row: %w", err)
	}
	if err := p.Validate(); err != nil {
		return LessonProgress{}, fmt.Errorf("invalid lesson progress record: %w", err)
	}
	return p, nil
}

type progressRowScanner interface {
	Scan(dest ...any) error
}

func scanProgressRow(row progressRowScanner) (LessonProgress, error) {
	var p LessonProgress
	if err := row.Scan(
		&p.ID, &p.TenantID, &p.CourseID, &p.EnrollmentID, &p.LessonID,
		&p.Completed, &p.CompletedAt, &p.CreatedAt, &p.UpdatedAt,
	); err != nil {
		return LessonProgress{}, err
	}
	p.CompletedAt = normalizeNullableTime(p.CompletedAt)
	if err := p.Validate(); err != nil {
		return LessonProgress{}, err
	}
	return p, nil
}

func normalizeNullableTime(value *time.Time) *time.Time {
	if value == nil {
		return nil
	}
	normalized := value.UTC()
	return &normalized
}

func mapProgressWriteError(err error) error {
	var pgErr *pgconn.PgError
	if !errors.As(err, &pgErr) {
		return fmt.Errorf("write lesson progress: %w", err)
	}
	switch {
	case pgErr.Code == "23505" &&
		pgErr.ConstraintName == "lesson_progress_enrollment_lesson_unique":
		return ErrProgressAlreadyExists
	case pgErr.Code == "23503" &&
		pgErr.ConstraintName == "lesson_progress_enrollment_fk":
		return ErrEnrollmentNotFound
	case pgErr.Code == "23503" &&
		pgErr.ConstraintName == "lesson_progress_lesson_fk":
		return ErrLessonNotFound
	default:
		return fmt.Errorf("write lesson progress: %w", err)
	}
}

		return LessonProgress{}, ErrInvalidTransition

		return LessonProgress{}, err

		return LessonProgress{}, ErrLessonNotFound

	_, err = s.progress.GetProgress(ctx, tenantID, courseID, enrollmentID, lessonID)
	if errors.Is(err, ErrProgressNotFound) {
		record := LessonProgress{
			ID: s.newID("prg"),
			TenantID: tenantID,
			CourseID: courseID,
			EnrollmentID: enrollmentID,
			LessonID: lessonID,
			Completed: false,
			CreatedAt: now,
			UpdatedAt: now,

		}
