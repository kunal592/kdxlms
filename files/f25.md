package identity

import (
	"context"
	"errors"
	"fmt"
	"strings"
	"time"

	"github.com/kdxlabs/kdxlabsllm/internal/platform/security"
)

var (
	ErrInvalidRegistration = errors.New("invalid registration")
	ErrAuthenticationUnavailable = errors.New("authentication service unavailable")
)

type AuthService struct {
	users       UserRepository
	sessions    SessionRepository
	clock       Clock
	sessionTTL  time.Duration
}

type RegisterInput struct {
	Email       string
	DisplayName string
	Password    string
}

type LoginInput struct {
	Email    string
	Password string
}

type LoginResult struct {
	User             PublicUser
	AccessToken      string
	AccessExpiresAt  time.Time
}

func NewAuthService(
	users UserRepository,
	sessions SessionRepository,
	clock Clock,
	sessionTTL time.Duration,
) (*AuthService, error) {
	if users == nil || sessions == nil {
		return nil, errors.New("user and session repositories are required")
	}
	if clock == nil {
		clock = SystemClock{}
	}
	if sessionTTL < time.Minute || sessionTTL > 24*time.Hour {
		return nil, errors.New("session TTL must be between one minute and 24 hours")
	}

	return &AuthService{
		users:      users,
		sessions:   sessions,
		clock:      clock,
		sessionTTL: sessionTTL,
	}, nil
}

func (s *AuthService) Register(
	ctx context.Context,
	input RegisterInput,
) (PublicUser, error) {
	if err := ctx.Err(); err != nil {
		return PublicUser{}, err
	}

	email, err := NormalizeEmail(input.Email)
	if err != nil {
		return PublicUser{}, ErrInvalidRegistration
	}

	name := strings.TrimSpace(input.DisplayName)
	if name == "" || len([]rune(name)) > 120 {
		return PublicUser{}, ErrInvalidRegistration
	}

	if err := ValidatePasswordPolicy(input.Password); err != nil {
		return PublicUser{}, ErrWeakPassword
	}

	passwordHash, err := security.HashPassword(input.Password)
	if err != nil {
		return PublicUser{}, fmt.Errorf("hash password: %w", err)
	}

	id, err := NewID("usr")
	if err != nil {
		return PublicUser{}, fmt.Errorf("generate user ID: %w", err)
	}

	now := s.clock.Now().UTC()
	user := User{
		ID:           id,
		Email:        email,
		DisplayName:  name,
		PasswordHash: passwordHash,
		Status:       UserStatusPendingVerification,
		CreatedAt:    now,
		UpdatedAt:    now,
	}

	if err := user.Validate(); err != nil {
		return PublicUser{}, fmt.Errorf("validate new user: %w", err)
	}

	if err := s.users.Create(ctx, user); err != nil {
		if errors.Is(err, ErrEmailAlreadyExists) {
			return PublicUser{}, ErrEmailAlreadyExists
		}
		return PublicUser{}, fmt.Errorf("create user: %w", err)
	}

	return user.Public(), nil
}

func (s *AuthService) Login(
	ctx context.Context,
	input LoginInput,
) (LoginResult, error) {
	if err := ctx.Err(); err != nil {
		return LoginResult{}, err
	}

	email, err := NormalizeEmail(input.Email)
	if err != nil || input.Password == "" {
		return LoginResult{}, ErrInvalidCredentials
	}

	user, err := s.users.GetByEmail(ctx, email)
	if err != nil {
		// Do not reveal whether an account exists.
		if errors.Is(err, ErrUserNotFound) {
			return LoginResult{}, ErrInvalidCredentials
		}
		return LoginResult{}, fmt.Errorf("load user: %w", err)
	}

	valid, err := security.VerifyPassword(input.Password, user.PasswordHash)
	if err != nil {
		// A malformed stored hash is an internal data problem, not a reason
		// to disclose details or accept the login.
		return LoginResult{}, fmt.Errorf("verify stored password hash: %w", err)
	}
	if !valid {
		return LoginResult{}, ErrInvalidCredentials
	}

	switch user.Status {
	case UserStatusPendingVerification:
		return LoginResult{}, ErrEmailNotVerified
	case UserStatusSuspended, UserStatusDisabled:
		return LoginResult{}, ErrAccountDisabled
	case UserStatusActive:
		// Continue.
	default:
		return LoginResult{}, ErrInvalidUserState
	}

	token, err := security.GenerateOpaqueToken(32)
	if err != nil {
		return LoginResult{}, fmt.Errorf("generate session token: %w", err)
	}

	digest, err := security.DigestToken(token)
	if err != nil {
		return LoginResult{}, fmt.Errorf("digest session token: %w", err)
	}

	sessionID, err := NewID("ses")
	if err != nil {
		return LoginResult{}, fmt.Errorf("generate session ID: %w", err)
	}

	now := s.clock.Now().UTC()
	session := Session{
		ID:          sessionID,
		UserID:      user.ID,
		TokenDigest: digest,
		CreatedAt:   now,
		ExpiresAt:   now.Add(s.sessionTTL),
	}

	if err := s.sessions.CreateSession(ctx, session); err != nil {
		return LoginResult{}, fmt.Errorf("create session: %w", err)
	}

	return LoginResult{
		User:            user.Public(),
		AccessToken:     token,
		AccessExpiresAt: session.ExpiresAt,
	}, nil
}

package identity

import (
	"errors"
	"net/http"
	"time"

	"github.com/gin-gonic/gin"
	"github.com/kdxlabs/kdxlabsllm/internal/platform/httpx"
)

type AuthServiceAPI interface {
	Register(context.Context, RegisterInput) (PublicUser, error)
	Login(context.Context, LoginInput) (LoginResult, error)
}

type AuthHandler struct {
	service AuthServiceAPI
}

func NewAuthHandler(service AuthServiceAPI) *AuthHandler {
	return &AuthHandler{service: service}
}

type registerRequest struct {
	Email       string `json:"email" validate:"required,email,max=254"`
	DisplayName string `json:"display_name" validate:"required,min=1,max=120"`
	Password    string `json:"password" validate:"required,min=12,max=128"`
}

type loginRequest struct {
	Email    string `json:"email" validate:"required,email,max=254"`
	Password string `json:"password" validate:"required,min=1,max=128"`
}

type loginResponse struct {
	User      PublicUser `json:"user"`
	TokenType string     `json:"token_type"`
	Token     string     `json:"access_token"`
	ExpiresAt time.Time  `json:"expires_at"`
}

func (h *AuthHandler) Register(c *gin.Context) {
	var request registerRequest
	if !httpx.BindJSON(c, &request) {
		return
	}

	user, err := h.service.Register(c.Request.Context(), RegisterInput{
		Email: request.Email, DisplayName: request.DisplayName, Password: request.Password,
	})
	if err != nil {
		switch {
		case errors.Is(err, ErrEmailAlreadyExists):
			httpx.Conflict(c, "EMAIL_ALREADY_EXISTS", "An account with this email already exists.")
		case errors.Is(err, ErrWeakPassword):
			httpx.BadRequest(c, "WEAK_PASSWORD", "The password does not meet security requirements.")
		case errors.Is(err, ErrInvalidRegistration), errors.Is(err, ErrInvalidEmail):
			httpx.BadRequest(c, "INVALID_REGISTRATION", "The registration details are invalid.")
		default:
			httpx.InternalServerError(c)
		}
		return
	}

	httpx.Created(c, user)
}

func (h *AuthHandler) Login(c *gin.Context) {
	var request loginRequest
	if !httpx.BindJSON(c, &request) {
		return
	}

	result, err := h.service.Login(c.Request.Context(), LoginInput{
		Email: request.Email, Password: request.Password,
	})
	if err != nil {
		switch {
		case errors.Is(err, ErrInvalidCredentials):
			httpx.Unauthorized(c, "INVALID_CREDENTIALS", "Email or password is incorrect.")
		case errors.Is(err, ErrEmailNotVerified):
			httpx.Forbidden(c, "EMAIL_NOT_VERIFIED", "Verify your email before signing in.")
		case errors.Is(err, ErrAccountDisabled):
			httpx.Forbidden(c, "ACCOUNT_UNAVAILABLE", "This account is unavailable.")
		default:
			httpx.InternalServerError(c)
		}
		return
	}

	httpx.JSON(c, http.StatusOK, loginResponse{
		User: result.User, TokenType: "Bearer",
		Token: result.AccessToken, ExpiresAt: result.AccessExpiresAt,
	})
}

package identity

import (
	"errors"

	"github.com/jackc/pgx/v5/pgconn"
)

const usersEmailUniqueConstraint = "users_email_unique_idx"

func mapPostgresError(err error) error {
	if err == nil {
		return nil
	}

	var pgErr *pgconn.PgError
	if errors.As(err, &pgErr) {
		if pgErr.Code == "23505" &&
			(pgErr.ConstraintName == usersEmailUniqueConstraint ||
				pgErr.ConstraintName == "users_email_key") {
			return ErrEmailAlreadyExists
		}
	}

	return err
}

package identity

import (
	"context"
	"errors"
	"fmt"
	"time"

	"github.com/jackc/pgx/v5"
	"github.com/jackc/pgx/v5/pgxpool"

type PostgresUserRepository struct {
	pool *pgxpool.Pool
}

var _ UserRepository = (*PostgresUserRepository)(nil)

func NewPostgresUserRepository(pool *pgxpool.Pool) (*PostgresUserRepository, error) {
	if pool == nil {
		return nil, errors.New("postgres user repository: pool is required")
	}

	return &PostgresUserRepository{pool: pool}, nil
}

func (r *PostgresUserRepository) Create(ctx context.Context, user User) error {
	if ctx == nil {
		return errors.New("postgres user repository: context is required")
	}
	if err := user.Validate(); err != nil {
		return fmt.Errorf("validate user: %w", err)
	}

	_, err := r.pool.Exec(ctx, `
		INSERT INTO users (
			id, email, display_name, password_hash, status,
			email_verified_at, created_at, updated_at
		)
		VALUES ($1, $2, $3, $4, $5, $6, $7, $8)
	`,
		user.ID,
		user.Email,
		user.DisplayName,
		user.PasswordHash,
		string(user.Status),
		user.EmailVerifiedAt,
		user.CreatedAt,
		user.UpdatedAt,
	)
	return mapPostgresError(err)
}

func (r *PostgresSessionRepository) RotateSession(
	ctx context.Context,
	oldSessionID string,
	newSession Session,
	rotatedAt time.Time,
) error {
	if ctx == nil {
		return errors.New("postgres session repository: context is required")
	}
	if err := newSession.Validate(); err != nil {
		return fmt.Errorf("validate replacement session: %w", err)
	}

	tx, err := r.pool.Begin(ctx)
	if err != nil {
		return err
	}
	defer func() {
		_ = tx.Rollback(ctx)
	}()

	var oldUserID string
	var oldRevokedAt *time.Time
	err = tx.QueryRow(ctx, `
		SELECT user_id, revoked_at
		FROM auth_sessions
		WHERE id = $1
		FOR UPDATE
	`, oldSessionID).Scan(&oldUserID, &oldRevokedAt)
	if errors.Is(err, pgx.ErrNoRows) {
		return ErrSessionNotFound
	}
	if err != nil {
		return err
	}
	if oldRevokedAt != nil {
		return ErrInvalidUserState
	}
	if oldUserID != newSession.UserID {
		return ErrInvalidUserID
	}

	_, err = tx.Exec(ctx, `
		INSERT INTO auth_sessions (
			id, user_id, token_digest, created_at, expires_at,
			revoked_at, replaced_by_id, last_used_at
		)
		VALUES ($1, $2, $3, $4, $5, $6, $7, $8)
	`,
		newSession.ID,
		newSession.UserID,
		newSession.TokenDigest,
		newSession.CreatedAt,
		newSession.ExpiresAt,
		newSession.RevokedAt,
		newSession.ReplacedByID,
		newSession.LastUsedAt,
	)
	if err != nil {
		return err
	}

	tag, err := tx.Exec(ctx, `
		UPDATE auth_sessions
		SET revoked_at = $2, replaced_by_id = $3, last_used_at = $2
		WHERE id = $1 AND revoked_at IS NULL
	`, oldSessionID, rotatedAt, newSession.ID)
	if err != nil {
		return err
	}
	if tag.RowsAffected() != 1 {
		return ErrSessionNotFound
	}

	return tx.Commit(ctx)
}

const authenticatedUserContextKey = "authenticated_user"

type AuthMiddleware struct {
	users    UserRepository
	sessions SessionRepository
	clock    Clock
}

func NewAuthMiddleware(
	users UserRepository,
	sessions SessionRepository,
	clock Clock,
) (*AuthMiddleware, error) {
	if users == nil || sessions == nil || clock == nil {
		return nil, errors.New("auth middleware: users, sessions and clock are required")
	}
	return &AuthMiddleware{
		users: users, sessions: sessions, clock: clock,
	}, nil
}

func (m *AuthMiddleware) RequireAuthentication() gin.HandlerFunc {
	return func(c *gin.Context) {
		token, ok := bearerToken(c.GetHeader("Authorization"))
		if !ok {
			httpx.Unauthorized(
				c,
				"AUTHENTICATION_REQUIRED",
				"A valid bearer token is required.",
			)
			return
		}

		digest, err := security.DigestToken(token)
		if err != nil {
			httpx.Unauthorized(c, "INVALID_TOKEN", "The access token is invalid.")
			return
		}

		session, err := m.sessions.GetSessionByTokenDigest(c.Request.Context(), digest)
		if err != nil {
			if errors.Is(err, ErrSessionNotFound) {
				httpx.Unauthorized(c, "INVALID_TOKEN", "The access token is invalid.")
			} else {
				httpx.InternalServerError(c)
			}
			return
		}

		if !session.IsActive(m.clock.Now()) {
			httpx.Unauthorized(c, "SESSION_EXPIRED", "The session is no longer active.")
			return
		}

		user, err := m.users.GetByID(c.Request.Context(), session.UserID)
		if err != nil {
			if errors.Is(err, ErrUserNotFound) {
				httpx.Unauthorized(c, "INVALID_TOKEN", "The access token is invalid.")
			} else {
				httpx.InternalServerError(c)
			}
			return
		}

		if user.Status != UserStatusActive {
			httpx.Forbidden(c, "ACCOUNT_UNAVAILABLE", "This account is unavailable.")
			return
		}

		c.Set(authenticatedUserContextKey, user.Public())
		c.Next()
	}
}

func bearerToken(header string) (string, bool) {
	parts := strings.Fields(header)
	if len(parts) != 2 || !strings.EqualFold(parts[0], "Bearer") {
		return "", false
	}

	token := parts[1]
	if len(token) < 20 || len(token) > 4096 {
		return "", false
	}
	if strings.TrimSpace(token) != token {
		return "", false
	}
	return token, true
}

func AuthenticatedUser(c *gin.Context) (PublicUser, bool) {
	if c == nil {
		return PublicUser{}, false
	}

	value, exists := c.Get(authenticatedUserContextKey)
	if !exists {
		return PublicUser{}, false
	}

	user, ok := value.(PublicUser)
	return user, ok
}

func RequireVerifiedEmail() gin.HandlerFunc {
	return func(c *gin.Context) {
		user, ok := AuthenticatedUser(c)
		if !ok {
			httpx.Unauthorized(
				c,
				"AUTHENTICATION_REQUIRED",
				"Authentication is required.",
			)
			return
		}

		if user.EmailVerifiedAt == nil {
			httpx.Fail(
				c,
				http.StatusForbidden,
				"EMAIL_NOT_VERIFIED",
				"Verify your email before accessing this resource.",
			)
			return
		}

		c.Next()
	}
}

func (s *AuthService) Logout(ctx context.Context, rawToken string) error {
	if ctx == nil {
		return errors.New("logout: context is required")
	}
	if rawToken == "" {
		return ErrInvalidCredentials
	}

	digest, err := security.DigestToken(rawToken)

			return nil
		}
		return err
	}

}

		return err
	}

}
