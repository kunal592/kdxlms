package httpx

import (

func TestJSONResponse(t *testing.T) {
	router := gin.New()
	router.Use(RequestID())

	router.GET("/test", func(c *gin.Context) {
		JSON(c, http.StatusOK, gin.H{"message": "success"})
	})

	recorder := httptest.NewRecorder()
	request := httptest.NewRequest(http.MethodGet, "/test", nil)

	router.ServeHTTP(recorder, request)

	if recorder.Code != http.StatusOK {
		t.Fatalf("status = %d, want %d", recorder.Code, http.StatusOK)
	}

	var response Envelope
	if err := json.Unmarshal(recorder.Body.Bytes(), &response); err != nil {
		t.Fatalf("decode response: %v", err)
	}

	if response.Error != nil {
		t.Fatal("expected no error in successful response")
	}

	if response.RequestID == "" {
		t.Fatal("expected request ID in response body")
	}

	if recorder.Header().Get("Content-Type") != "application/json; charset=utf-8" {
		t.Fatalf("unexpected Content-Type: %q", recorder.Header().Get("Content-Type"))
	}
}

func TestPaginatedResponse(t *testing.T) {
	router := gin.New()
	router.GET("/test", func(c *gin.Context) {
		Paginated(c, http.StatusOK, []string{"a", "b"}, 2, 10, 25)
	})

	recorder := httptest.NewRecorder()
	request := httptest.NewRequest(http.MethodGet, "/test", nil)

	router.ServeHTTP(recorder, request)

	if recorder.Code != http.StatusOK {
		t.Fatalf("status = %d, want %d", recorder.Code, http.StatusOK)
	}

	var response Envelope
	if err := json.Unmarshal(recorder.Body.Bytes(), &response); err != nil {
		t.Fatalf("decode response: %v", err)
	}

	if response.Pagination == nil {
		t.Fatal("expected pagination metadata")
	}

	if response.Pagination.Page != 2 {
		t.Errorf("page = %d, want 2", response.Pagination.Page)
	}

	if response.Pagination.PageSize != 10 {
		t.Errorf("page size = %d, want 10", response.Pagination.PageSize)
	}

	if response.Pagination.TotalItems != 25 {
		t.Errorf("total items = %d, want 25", response.Pagination.TotalItems)
	}

	if response.Pagination.TotalPages != 3 {
		t.Errorf("total pages = %d, want 3", response.Pagination.TotalPages)
	}
}

func TestInternalServerErrorDoesNotExposeDetails(t *testing.T) {
	router := gin.New()
	router.GET("/test", func(c *gin.Context) {
		InternalServerError(c)
	})

	recorder := httptest.NewRecorder()
	request := httptest.NewRequest(http.MethodGet, "/test", nil)

	router.ServeHTTP(recorder, request)

	if recorder.Code != http.StatusInternalServerError {
		t.Fatalf("status = %d, want %d", recorder.Code, http.StatusInternalServerError)
	}

	body := recorder.Body.String()

	for _, forbidden := range []string{
		"password",
		"database",
		"stack trace",
		"panic",
	} {
		if containsFold(body, forbidden) {
			t.Errorf("response unexpectedly contains %q", forbidden)
		}
	}
}

func TestBuildPagination(t *testing.T) {
	tests := []struct {
		name          string
		page          int
		pageSize      int
		totalItems    int64
		wantTotalPage int64
		wantPage      int
	}{
		{
			name:          "multiple pages",
			page:          1,
			pageSize:      10,
			totalItems:    25,
			wantTotalPage: 3,
			wantPage:      1,
		},
		{
			name:          "exact division",
			page:          1,
			pageSize:      10,
			totalItems:    20,
			wantTotalPage: 2,
			wantPage:      1,
		},
		{
			name:          "no results",
			page:          1,
			pageSize:      10,
			totalItems:    0,
			wantTotalPage: 0,
			wantPage:      1,
		},
		{
			name:          "single result",
			page:          1,
			pageSize:      10,
			totalItems:    1,
			wantTotalPage: 1,
			wantPage:      1,
		},
		{
			name:          "invalid page",
			page:          0,
			pageSize:      10,
			totalItems:    20,
			wantTotalPage: 0,
			wantPage:      0,
		},
		{
			name:          "invalid page size",
			page:          1,
			pageSize:      0,
			totalItems:    20,
			wantTotalPage: 0,
			wantPage:      0,
		},
		{
			name:          "negative total",
			page:          1,
			pageSize:      10,
			totalItems:    -1,
			wantTotalPage: 0,
			wantPage:      0,
		},
	}

	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			got := BuildPagination(tt.page, tt.pageSize, tt.totalItems)

			if got.TotalPages != tt.wantTotalPage {
				t.Errorf("total pages = %d, want %d", got.TotalPages, tt.wantTotalPage)
			}

			if got.Page != tt.wantPage {
				t.Errorf("page = %d, want %d", got.Page, tt.wantPage)
			}
		})
	}
}

func containsFold(value, substring string) bool {
	return len(substring) == 0 ||
		len(value) >= len(substring) &&
			findFold(value, substring)
}

func findFold(value, substring string) bool {
	for i := 0; i+len(substring) <= len(value); i++ {
		if equalFoldASCII(value[i:i+len(substring)], substring) {
			return true
		}
	}
	return false
}

func equalFoldASCII(a, b string) bool {
	if len(a) != len(b) {
		return false
	}

	for i := range a {
		ac, bc := a[i], b[i]

		if ac >= 'A' && ac <= 'Z' {
			ac += 'a' - 'A'
		}

		if bc >= 'A' && bc <= 'Z' {
			bc += 'a' - 'A'
		}

		if ac != bc {
			return false
		}
	}

	return true
}
