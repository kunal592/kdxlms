import (
	"encoding/json"
	"errors"
	"io"
	"net/http"

	"github.com/gin-gonic/gin"
	"github.com/go-playground/validator/v10"

// BindJSON decodes exactly one JSON object into dst.
//
// It rejects unknown fields and trailing JSON values. Request body size limits
// should be configured with BodyLimit middleware before this helper is used.
//
// Returns true only when decoding and structural validation succeed.
// Callers should return immediately when BindJSON returns false.
func BindJSON(c *gin.Context, dst any) bool {
	if c == nil || c.Request == nil || c.Request.Body == nil || dst == nil {
		if c != nil {
			BadRequest(c, "INVALID_REQUEST", "The request could not be processed.")
		}
		return false
	}

	decoder := json.NewDecoder(c.Request.Body)
	decoder.DisallowUnknownFields()

	if err := decoder.Decode(dst); err != nil {
		var maxBytesErr *http.MaxBytesError
		if errors.As(err, &maxBytesErr) {
			c.AbortWithStatusJSON(http.StatusRequestEntityTooLarge, Envelope{
				Error: &ErrorDetail{
					Code:    "REQUEST_BODY_TOO_LARGE",
					Message: "The request body exceeds the permitted size.",
				},
				RequestID: GetRequestID(c),
			})
			return false
		}

		BadRequest(c, "INVALID_JSON", "The request body must contain valid JSON.")
		return false
	}

	// Reject trailing values such as: {"name":"a"} {"name":"b"}
	var extra any
	err := decoder.Decode(&extra)

	switch {
	case errors.Is(err, io.EOF):
		// Exactly one JSON value was supplied.
	case err == nil:
		BadRequest(c, "INVALID_JSON", "The request must contain exactly one JSON value.")
		return false
	default:
		var maxBytesErr *http.MaxBytesError
		if errors.As(err, &maxBytesErr) {
			c.AbortWithStatusJSON(http.StatusRequestEntityTooLarge, Envelope{
				Error: &ErrorDetail{
					Code:    "REQUEST_BODY_TOO_LARGE",
					Message: "The request body exceeds the permitted size.",
				},
				RequestID: GetRequestID(c),
			})
			return false
		}

		BadRequest(c, "INVALID_JSON", "The request body must contain valid JSON.")
		return false
	}

	// Apply the project's shared validator to struct tags such as `validate:"required"`.
	if v, ok := validator.New().Struct(dst); ok == nil {
		_ = v
	} else {
		var validationErrors validator.ValidationErrors
		if errors.As(ok, &validationErrors) {
			BadRequest(c, "VALIDATION_FAILED", "One or more request fields are invalid.")
			return false
		}

		// A validator configuration or custom validation error should not
		// expose internal implementation details to the client.
		InternalServerError(c)
		return false
	}

	return true
}

package httpx

import (
	"mime"
	"net/http"
	"strings"

	"github.com/gin-gonic/gin"
)

// RequireJSONContentType requires application/json for requests that may
// carry a body. It allows bodyless requests and does not inspect the body.
//
// Register it only on routes that accept JSON, not on file-upload routes.
func RequireJSONContentType() gin.HandlerFunc {
	return func(c *gin.Context) {
		method := strings.ToUpper(c.Request.Method)

		switch method {
		case http.MethodPost, http.MethodPut, http.MethodPatch:
			// These methods commonly carry request payloads.
		default:
			c.Next()
			return
		}

		// An empty body does not need a Content-Type header.
		if c.Request.Body == nil || c.Request.ContentLength == 0 {
			c.Next()
			return
		}

		raw := strings.TrimSpace(c.GetHeader("Content-Type"))
		if raw == "" {
			unsupportedMediaType(c)
			return
		}

		mediaType, _, err := mime.ParseMediaType(raw)
		if err != nil || !strings.EqualFold(mediaType, "application/json") {
			unsupportedMediaType(c)
			return
		}

		c.Next()
	}
}

func unsupportedMediaType(c *gin.Context) {
	c.AbortWithStatusJSON(http.StatusUnsupportedMediaType, Envelope{
		Error: &ErrorDetail{
			Code:    "UNSUPPORTED_MEDIA_TYPE",
			Message: "This endpoint accepts application/json requests.",
		},
		RequestID: GetRequestID(c),
	})
}

package httpx

import (
	"net/http"
	"strconv"
	"strings"

	"github.com/gin-gonic/gin"
)

const (
	DefaultPageSize = 20
	MaxPageSize     = 100
	MaxPageNumber   = 1_000_000
)

// ParsePagination reads the optional page and page_size query parameters.
//
// Defaults are page=1 and page_size=20. Invalid values return HTTP 400.
// Page size is capped at MaxPageSize; page numbers above MaxPageNumber are
// rejected to avoid unreasonable offsets in downstream database queries.
func ParsePagination(c *gin.Context) (page, pageSize int, ok bool) {
	page = 1
	pageSize = DefaultPageSize

	rawPage := strings.TrimSpace(c.Query("page"))
	if rawPage != "" {
		value, err := strconv.Atoi(rawPage)
		if err != nil || value < 1 || value > MaxPageNumber {
			BadRequest(
				c,
				"INVALID_PAGE",
				"Page must be a positive integer within the supported range.",
			)
			return 0, 0, false
		}
		page = value
	}

	rawPageSize := strings.TrimSpace(c.Query("page_size"))
	if rawPageSize != "" {
		value, err := strconv.Atoi(rawPageSize)
		if err != nil || value < 1 {
			BadRequest(
				c,
				"INVALID_PAGE_SIZE",
				"Page size must be a positive integer.",
			)
			return 0, 0, false
		}

		if value > MaxPageSize {
			value = MaxPageSize
		}
		pageSize = value
	}

	return page, pageSize, true
}

// PaginationOffset computes a database offset without overflowing int.
// It returns false if the requested offset cannot be represented safely.
func PaginationOffset(page, pageSize int) (int, bool) {
	if page < 1 || pageSize < 1 || page > MaxPageNumber {
		return 0, false
	}

	maxInt := int(^uint(0) >> 1)
	if page-1 > maxInt/pageSize {
		return 0, false
	}

	return (page - 1) * pageSize, true
}

// RejectUnsupportedQueryParameters is intended for endpoints with a fixed
// query contract. Pass the names of all supported query parameters.
func RejectUnsupportedQueryParameters(
	c *gin.Context,
	allowed ...string,
) bool {
	allow := make(map[string]struct{}, len(allowed))
	for _, key := range allowed {
		allow[key] = struct{}{}

			BadRequest(
				c,
				"UNSUPPORTED_QUERY_PARAMETER",
				"The request contains an unsupported query parameter.",
			)
			return false
		}
	}

	return true
}
