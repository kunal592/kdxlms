	sensitiveMarkers := []string{
		"password",
		"passwd",
		"secret",
		"token",
		"authorization",
		"bearer ",
		"api_key",
		"apikey",
		"connection string",
		"postgres://",
		"postgresql://",
	}

	for _, marker := range sensitiveMarkers {
		if strings.Contains(lower, marker) {
			return "internal processing error"
		}
	}

	const maxLength = 256
	if len(message) > maxLength {
		return message[:maxLength]
	}
	return message

import (
	"context"
	"sync"
	"time"
)

type AssessmentAuditOutboxMetricsSnapshot struct {
	Batches          uint64
	Processed        uint64
	FailedBatches    uint64
	TotalDuration    time.Duration
	LastBatchAt      time.Time
	LastFailureAt    time.Time
	LastFailureClass string
}

type InMemoryAssessmentAuditOutboxMetrics struct {
	mu    sync.RWMutex
	state AssessmentAuditOutboxMetricsSnapshot
	clock func() time.Time
}

func NewInMemoryAssessmentAuditOutboxMetrics() *InMemoryAssessmentAuditOutboxMetrics {
	return &InMemoryAssessmentAuditOutboxMetrics{clock: time.Now}
}

func (m *InMemoryAssessmentAuditOutboxMetrics) ObserveAssessmentAuditOutboxBatch(
	_ context.Context,
	processed int,
	duration time.Duration,
	err error,
) {
	if m == nil {
		return
	}
	m.mu.Lock()
	defer m.mu.Unlock()

	now := m.clock().UTC()
	m.state.Batches++
	m.state.LastBatchAt = now

	if processed > 0 {
		m.state.Processed += uint64(processed)
	}
	if duration > 0 {
		m.state.TotalDuration += duration
	}

	if err != nil {
		m.state.FailedBatches++
		m.state.LastFailureAt = now
		// Do not retain raw error messages, which may contain secrets.
		m.state.LastFailureClass = "processing_error"
	}
}

func (m *InMemoryAssessmentAuditOutboxMetrics) Snapshot() AssessmentAuditOutboxMetricsSnapshot {
	if m == nil {
		return AssessmentAuditOutboxMetricsSnapshot{}
	}
	m.mu.RLock()
	defer m.mu.RUnlock()
	return m.state
}

package assessment

import (
	"encoding/base64"
	"errors"
	"strings"
)

const deadLetterCursorMaxLength = 512

type AssessmentAuditDeadLetterCursor struct {
	CreatedAt string
	EventID   string
}

func EncodeAssessmentAuditDeadLetterCursor(cursor AssessmentAuditDeadLetterCursor) (string, error) {
	if cursor.CreatedAt == "" || cursor.EventID == "" {
		return "", ErrDeadLetterInvalidCursor
	}
	if strings.ContainsAny(cursor.CreatedAt+cursor.EventID, "\r\n\x00") {
		return "", ErrDeadLetterInvalidCursor
	}

	raw := cursor.CreatedAt + "\n" + cursor.EventID
	encoded := base64.RawURLEncoding.EncodeToString([]byte(raw))
	if len(encoded) > deadLetterCursorMaxLength {
		return "", ErrDeadLetterInvalidCursor
	}
	return encoded, nil
}

func DecodeAssessmentAuditDeadLetterCursor(value string) (AssessmentAuditDeadLetterCursor, error) {
	if value == "" || len(value) > deadLetterCursorMaxLength {
		return AssessmentAuditDeadLetterCursor{}, ErrDeadLetterInvalidCursor
	}

	raw, err := base64.RawURLEncoding.DecodeString(value)
	if err != nil {
		return AssessmentAuditDeadLetterCursor{}, ErrDeadLetterInvalidCursor
	}

	parts := strings.Split(string(raw), "\n")
	if len(parts) != 2 || parts[0] == "" || parts[1] == "" {
		return AssessmentAuditDeadLetterCursor{}, errors.Join(

	tenantID string,
	actorUserID string,
	eventID string,
	action string,

	actorUserID = strings.TrimSpace(actorUserID)
	eventID = strings.TrimSpace(eventID)

	input := ""

		input += string(rune(len(part))) + ":" + part + "|"
	}

	sum := sha256.Sum256([]byte(input))

}

		Message: message,
	})

	"crypto/sha256"
	"encoding/binary"
	"encoding/hex"
	"errors"
	"strings"
)

func AssessmentAuditDeadLetterAdminIdempotencyKey(
	tenantID string,
	actorUserID string,
	eventID string,
	action string,
) (string, error) {
	parts := []string{
		strings.TrimSpace(tenantID),
		strings.TrimSpace(actorUserID),
		strings.TrimSpace(eventID),
		strings.TrimSpace(action),
	}

	for _, part := range parts {
		if part == "" {
			return "", errors.New("tenant, actor, event and action are required")
		}
		if len(part) > 1024 {
			return "", errors.New("administrative action field is too long")
		}
	}

	hash := sha256.New()
	var length [4]byte

	for _, part := range parts {
		binary.BigEndian.PutUint32(length[:], uint32(len(part)))
		_, _ = hash.Write(length[:])
		_, _ = hash.Write([]byte(part))
	}

	return hex.EncodeToString(hash.Sum(nil)), nil
}

type AssessmentAuditDeadLetterMemoryIdempotencyStore struct {

	records map[string]AssessmentAuditDeadLetterIdempotencyRecord
	maxSize int
	clock   func() time.Time
}

func NewAssessmentAuditDeadLetterMemoryIdempotencyStore(
	maxSize int,
) (*AssessmentAuditDeadLetterMemoryIdempotencyStore, error) {
	if maxSize < 1 {
		return nil, errors.New("idempotency store capacity must be positive")
	}
	return &AssessmentAuditDeadLetterMemoryIdempotencyStore{
		records: make(map[string]AssessmentAuditDeadLetterIdempotencyRecord),
		maxSize: maxSize,
		clock:   time.Now,
	}, nil
}

func (s *AssessmentAuditDeadLetterMemoryIdempotencyStore) Begin(
	ctx context.Context,
	key string,
	fingerprint string,
	ttl time.Duration,
) (AssessmentAuditDeadLetterIdempotencyRecord, bool, error) {
	if s == nil {
		return AssessmentAuditDeadLetterIdempotencyRecord{}, false, errors.New("idempotency store is not initialized")
	}
	if err := ctx.Err(); err != nil {
		return AssessmentAuditDeadLetterIdempotencyRecord{}, false, err
	}
	if strings.TrimSpace(key) == "" || strings.TrimSpace(fingerprint) == "" {
		return AssessmentAuditDeadLetterIdempotencyRecord{}, false, errors.New("idempotency key and fingerprint are required")
	}
	if len(key) > 128 || len(fingerprint) > 128 {
		return AssessmentAuditDeadLetterIdempotencyRecord{}, false, errors.New("idempotency key or fingerprint is too long")
	}
	if ttl <= 0 {
		return AssessmentAuditDeadLetterIdempotencyRecord{}, false, errors.New("idempotency TTL must be positive")
	}

	s.mu.Lock()
	defer s.mu.Unlock()

	now := s.clock().UTC()
	s.removeExpiredLocked(now)

	if existing, ok := s.records[key]; ok {
		if existing.Fingerprint != fingerprint {
			return AssessmentAuditDeadLetterIdempotencyRecord{}, false, ErrDeadLetterIdempotencyConflict
		}
		if existing.Completed {
			return existing, false, nil
		}
		return existing, false, ErrDeadLetterIdempotencyBusy
	}

	if len(s.records) >= s.maxSize {
		return AssessmentAuditDeadLetterIdempotencyRecord{}, false, errors.New("idempotency store capacity reached")
	}

	record := AssessmentAuditDeadLetterIdempotencyRecord{
		Key:         key,
		Fingerprint: fingerprint,
		CreatedAt:   now,
		ExpiresAt:   now.Add(ttl),
	}
	s.records[key] = record
	return record, true, nil
}

func (s *AssessmentAuditDeadLetterMemoryIdempotencyStore) Complete(
	ctx context.Context,
	key string,
	fingerprint string,
) error {
	if s == nil {
		return errors.New("idempotency store is not initialized")
	}
	if err := ctx.Err(); err != nil {
		return err
	}

	s.mu.Lock()
	defer s.mu.Unlock()

	now := s.clock().UTC()
	s.removeExpiredLocked(now)

	record, ok := s.records[key]
	if !ok {
		return ErrDeadLetterIdempotencyExpired
	}
	if record.Fingerprint != fingerprint {
		return ErrDeadLetterIdempotencyConflict
	}
	record.Completed = true
	s.records[key] = record
	return nil
}

func (s *AssessmentAuditDeadLetterMemoryIdempotencyStore) removeExpiredLocked(now time.Time) {
	for key, record := range s.records {
		if !record.ExpiresAt.After(now) {
			delete(s.records, key)
		}
	}
}

		if len(part) > 1024 {
			return "", errors.New("fingerprint field exceeds maximum length")
		}
	}

	hash := sha256.New()
	var size [4]byte
	for _, part := range parts {
		binary.BigEndian.PutUint32(size[:], uint32(len(part)))
		_, _ = hash.Write(size[:])
		_, _ = hash.Write([]byte(part))
	}
	return hex.EncodeToString(hash.Sum(nil)), nil
}

	"context"
	"errors"
	"time"

type AssessmentAuditDeadLetterIdempotencyCleaner interface {
	CleanupExpired(ctx context.Context, before time.Time) (int, error)
}

type AssessmentAuditDeadLetterIdempotencyCleanupRunner struct {
	cleaner  AssessmentAuditDeadLetterIdempotencyCleaner
	interval time.Duration
}

func NewAssessmentAuditDeadLetterIdempotencyCleanupRunner(
	cleaner AssessmentAuditDeadLetterIdempotencyCleaner,
	interval time.Duration,
) (*AssessmentAuditDeadLetterIdempotencyCleanupRunner, error) {
	if cleaner == nil {
		return nil, errors.New("idempotency cleaner is required")
	}
	if interval <= 0 {
		return nil, errors.New("cleanup interval must be positive")
	}
	return &AssessmentAuditDeadLetterIdempotencyCleanupRunner{
		cleaner:  cleaner,
		interval: interval,
	}, nil
}

func (r *AssessmentAuditDeadLetterIdempotencyCleanupRunner) Run(ctx context.Context) error {
	if r == nil || r.cleaner == nil {
		return errors.New("idempotency cleanup runner is not initialized")
	}

	ticker := time.NewTicker(r.interval)
	defer ticker.Stop()

	for {
		if _, err := r.cleaner.CleanupExpired(ctx, time.Now().UTC()); err != nil {
			if ctx.Err() != nil {
				return ctx.Err()
			}
			// Retry on the next interval; transient cleanup failures do not stop the runner.
		}

		select {
		case <-ctx.Done():
			return ctx.Err()
		case <-ticker.C:
		}
	}
}
