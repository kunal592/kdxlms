	"time"

var ErrAssessmentAuditCursorPolicyAdminArchiveScheduler = errors.New("policy archive scheduler error")

type AssessmentAuditCursorPolicyAdminArchiveScheduler struct {
	Runner   AssessmentAuditCursorPolicyAdminEventArchiveRunner
	Interval time.Duration
}

func (s AssessmentAuditCursorPolicyAdminArchiveScheduler) Run(ctx context.Context) error {
	if ctx == nil || s.Interval < time.Minute || s.Interval > 24*time.Hour {
		return ErrAssessmentAuditCursorPolicyAdminArchiveScheduler
	}

	ticker := time.NewTicker(s.Interval)
	defer ticker.Stop()

	for {
		select {
		case <-ctx.Done():
			return ctx.Err()
		case <-ticker.C:
			_, err := s.Runner.Run(ctx)
			if err != nil && !errors.Is(err, context.Canceled) && !errors.Is(err, context.DeadlineExceeded) {
				// Continue scheduling; operational metrics should track failures.
				continue
			}
			if err != nil {
				return err
			}
		}
	}
}

type AssessmentAuditCursorPolicyAdminArchiveSchedulerGroup struct {
	mu      sync.Mutex
	running bool
}

func (g *AssessmentAuditCursorPolicyAdminArchiveSchedulerGroup) Start(
	ctx context.Context,
	scheduler AssessmentAuditCursorPolicyAdminArchiveScheduler,
) error {
	if ctx == nil {
		return ErrAssessmentAuditCursorPolicyAdminArchiveScheduler
	}

	g.mu.Lock()
	if g.running {
		g.mu.Unlock()
		return ErrAssessmentAuditCursorPolicyAdminArchiveScheduler
	}
	g.running = true
	g.mu.Unlock()

	go func() {
		defer func() {
			g.mu.Lock()
			g.running = false
			g.mu.Unlock()
		}()
		_ = scheduler.Run(ctx)
	}()
	return nil
}

func (l *AssessmentAuditCursorPolicyAdminSchedulerLifecycle) Start(
	parent context.Context,
	scheduler AssessmentAuditCursorPolicyAdminArchiveScheduler,
) error {
	if parent == nil {
		return ErrAssessmentAuditCursorPolicyAdminSchedulerLifecycle
	}

	l.mu.Lock()
	defer l.mu.Unlock()

	if l.cancel != nil {
		return ErrAssessmentAuditCursorPolicyAdminSchedulerLifecycle
	}

	ctx, cancel := context.WithCancel(parent)
	done := make(chan struct{})
	l.cancel = cancel
	l.done = done

	go func() {
		defer close(done)
		_ = scheduler.Run(ctx)
	}()
	return nil
}

	if ctx == nil {
		return ErrAssessmentAuditCursorPolicyAdminSchedulerLifecycle

	l.mu.Lock()
	cancel := l.cancel
	done := l.done
	l.mu.Unlock()

	if cancel == nil || done == nil {
		return nil
	}
	cancel()

	select {
	case <-done:
		l.mu.Lock()
		l.cancel = nil
		l.done = nil
		l.mu.Unlock()
		return nil
	case <-ctx.Done():
		return ctx.Err()

		a.Delay < 0 || a.Delay > time.Minute {
		return 0, ErrAssessmentAuditCursorPolicyAdminArchiveRetry
	}

	var lastErr error
	for attempt := 0; attempt < a.MaxAttempts; attempt++ {
		if err := ctx.Err(); err != nil {
			return 0, err
		}
		count, err := a.Archive.ArchivePolicyAdminEvents(ctx, cutoff, limit)
		if err == nil {
			return count, nil

			continue
		}

		select {
		case <-ctx.Done():
			if !timer.Stop() {
				<-timer.C
			}
			return 0, ctx.Err()
		case <-timer.C:
		}
	}

type AssessmentAuditCursorPolicyAdminSQLAdvisoryLock struct {
	DB  *sql.DB
	Key int64
}

func (l AssessmentAuditCursorPolicyAdminSQLAdvisoryLock) Acquire(ctx context.Context) (func(), error) {
	if l.DB == nil || ctx == nil {
		return nil, ErrAssessmentAuditCursorPolicyAdminArchiveSQLLock
	}

	conn, err := l.DB.Conn(ctx)
	if err != nil {
		return nil, errors.Join(ErrAssessmentAuditCursorPolicyAdminArchiveSQLLock, err)
	}

	var acquired bool
	if err := conn.QueryRowContext(ctx, "SELECT pg_try_advisory_lock($1)", l.Key).Scan(&acquired); err != nil {
		_ = conn.Close()
		return nil, errors.Join(ErrAssessmentAuditCursorPolicyAdminArchiveSQLLock, err)
	}
	if !acquired {
		_ = conn.Close()
		return nil, ErrAssessmentAuditCursorPolicyAdminArchiveSQLLock
	}

	release := func() {
		defer conn.Close()
		var unlocked bool
		_ = conn.QueryRowContext(context.Background(), "SELECT pg_advisory_unlock($1)", l.Key).Scan(&unlocked)
	}
	return release, nil
}
