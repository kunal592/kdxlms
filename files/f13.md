This file establishes a consistent API response format for KDXLabsLLM. Successful responses, paginated results, and errors will follow a predictable JSON structure.

package httpx

import (
	"net/http"

	"github.com/gin-gonic/gin"
)

// ErrorDetail describes a machine-readable API error.
type ErrorDetail struct {
	Code    string `json:"code"`
	Message string `json:"message"`
}

// Pagination describes a page of results.
type Pagination struct {
	Page       int   `json:"page"`
	PageSize   int   `json:"page_size"`
	TotalItems int64 `json:"total_items"`
	TotalPages int64 `json:"total_pages"`
}

// Envelope is the standard API response structure.
type Envelope struct {
	Data       any          `json:"data,omitempty"`
	Error      *ErrorDetail `json:"error,omitempty"`
	Pagination *Pagination  `json:"pagination,omitempty"`
	RequestID  string       `json:"request_id,omitempty"`
}

// JSON writes a successful JSON response.
func JSON(c *gin.Context, status int, data any) {
	c.JSON(status, Envelope{
		Data:      data,
		RequestID: GetRequestID(c),
	})
}

// Created writes an HTTP 201 response.
func Created(c *gin.Context, data any) {
	JSON(c, http.StatusCreated, data)
}

// NoContent writes an HTTP 204 response without a body.
func NoContent(c *gin.Context) {
	c.Status(http.StatusNoContent)
}

// Paginated writes a successful response containing pagination metadata.
func Paginated(
	c *gin.Context,
	status int,
	data any,
	page int,
	pageSize int,
	totalItems int64,
) {
	pagination := BuildPagination(page, pageSize, totalItems)

	c.JSON(status, Envelope{
		Data:       data,
		Pagination: &pagination,
		RequestID:  GetRequestID(c),
	})
}

// Fail writes a consistent API error response.
//
// Messages should be safe to expose to clients. Internal errors, database
// details, stack traces, tokens, and secrets must not be included.
func Fail(
	c *gin.Context,
	status int,
	code string,
	message string,
) {
	c.AbortWithStatusJSON(status, Envelope{
		Error: &ErrorDetail{
			Code:    code,
			Message: message,
		},
		RequestID: GetRequestID(c),
	})
}

// BadRequest writes an HTTP 400 response.
func BadRequest(c *gin.Context, code string, message string) {
	Fail(c, http.StatusBadRequest, code, message)
}

// Unauthorized writes an HTTP 401 response.
func Unauthorized(c *gin.Context, code string, message string) {
	Fail(c, http.StatusUnauthorized, code, message)
}

// Forbidden writes an HTTP 403 response.
func Forbidden(c *gin.Context, code string, message string) {
	Fail(c, http.StatusForbidden, code, message)
}

// NotFound writes an HTTP 404 response.
func NotFound(c *gin.Context, code string, message string) {
	Fail(c, http.StatusNotFound, code, message)
}

// Conflict writes an HTTP 409 response.
func Conflict(c *gin.Context, code string, message string) {
	Fail(c, http.StatusConflict, code, message)
}

// TooManyRequests writes an HTTP 429 response.
func TooManyRequests(c *gin.Context, code string, message string) {
	Fail(c, http.StatusTooManyRequests, code, message)
}

// InternalServerError writes a generic HTTP 500 response.
func InternalServerError(c *gin.Context) {
	Fail(
		c,
		http.StatusInternalServerError,
		"INTERNAL_ERROR",
		"An internal server error occurred.",
	)
}

// BuildPagination calculates pagination metadata.
//
// Invalid arguments return zero-valued metadata. Request handlers should
// validate page and pageSize before querying the database.
func BuildPagination(page int, pageSize int, totalItems int64) Pagination {
	if page < 1 || pageSize < 1 || totalItems < 0 {
		return Pagination{}
	}

	totalPages := totalItems / int64(pageSize)
	if totalItems%int64(pageSize) != 0 {
		totalPages++
	}

	return Pagination{
		Page:       page,
		PageSize:   pageSize,
		TotalItems: totalItems,
		TotalPages: totalPages,
	}
}
