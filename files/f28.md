package identity

import (
	"context"
	"errors"
	"strings"
	"time"
)

type SecurityEventType string

const (
	SecurityEventLoginSucceeded    SecurityEventType = "login_succeeded"
	SecurityEventLoginFailed       SecurityEventType = "login_failed"
	SecurityEventLogout            SecurityEventType = "logout"
	SecurityEventLogoutAll          SecurityEventType = "logout_all"
	SecurityEventPasswordChanged    SecurityEventType = "password_changed"
	SecurityEventPasswordReset      SecurityEventType = "password_reset"
	SecurityEventEmailVerified      SecurityEventType = "email_verified"
	SecurityEventSessionRevoked     SecurityEventType = "session_revoked"
)

type SecurityAuditEvent struct {
	ID        string
	UserID    string
	EventType SecurityEventType
	IPAddress string
	UserAgent string
	CreatedAt time.Time
}

func (e SecurityAuditEvent) Validate() error {
	if strings.TrimSpace(e.ID) == "" {
		return errors.New("audit event ID is required")
	}
	if strings.TrimSpace(e.UserID) == "" {
		return errors.New("audit event user ID is required")
	}
	if !validSecurityEventType(e.EventType) {
		return errors.New("audit event type is invalid")
	}
	if e.CreatedAt.IsZero() {
		return errors.New("audit event timestamp is required")
	}
	if len(e.IPAddress) > 64 || len(e.UserAgent) > 512 {
		return errors.New("audit event metadata exceeds limits")
	}
	return nil
}

func validSecurityEventType(eventType SecurityEventType) bool {
	switch eventType {
	case SecurityEventLoginSucceeded,
		SecurityEventLoginFailed,
		SecurityEventLogout,
		SecurityEventLogoutAll,
		SecurityEventPasswordChanged,
		SecurityEventPasswordReset,
		SecurityEventEmailVerified,
		SecurityEventSessionRevoked:
		return true
	default:
		return false
	}
}

type SecurityAuditRepository interface {
	RecordSecurityEvent(context.Context, SecurityAuditEvent) error
	ListSecurityEvents(context.Context, string, int, int) ([]SecurityAuditEvent, int64, error)
}

package identity

import (
	"context"
	"errors"
	"fmt"
	"strings"

	"github.com/jackc/pgx/v5"
	"github.com/jackc/pgx/v5/pgxpool"
)

type PostgresSecurityAuditRepository struct {
	pool *pgxpool.Pool
}

var _ SecurityAuditRepository = (*PostgresSecurityAuditRepository)(nil)

func NewPostgresSecurityAuditRepository(
	pool *pgxpool.Pool,
) (*PostgresSecurityAuditRepository, error) {
	if pool == nil {
		return nil, errors.New("security audit repository: pool is required")
	}
	return &PostgresSecurityAuditRepository{pool: pool}, nil
}

func (r *PostgresSecurityAuditRepository) RecordSecurityEvent(
	ctx context.Context,
	event SecurityAuditEvent,
) error {
	if ctx == nil {
		return errors.New("security audit repository: context is required")
	}
	if err := event.Validate(); err != nil {
		return fmt.Errorf("validate security audit event: %w", err)
	}

	_, err := r.pool.Exec(ctx, `
		INSERT INTO security_audit_events (
			id, user_id, event_type, ip_address, user_agent, created_at
		)
		VALUES ($1, $2, $3, $4, $5, $6)
	`,
		event.ID,
		event.UserID,
		string(event.EventType),
		event.IPAddress,
		event.UserAgent,
		event.CreatedAt,
	)
	return err
}

func (r *PostgresSecurityAuditRepository) ListSecurityEvents(
	ctx context.Context,
	userID string,
	limit int,
	offset int,
) ([]SecurityAuditEvent, int64, error) {
	if ctx == nil {
		return nil, 0, errors.New("security audit repository: context is required")
	}
	if strings.TrimSpace(userID) == "" {
		return nil, 0, ErrInvalidUserID
	}
	if limit < 1 || limit > 100 || offset < 0 {
		return nil, 0, errors.New("security audit repository: invalid pagination")
	}

	var total int64
	if err := r.pool.QueryRow(ctx, `
		SELECT COUNT(*)
		FROM security_audit_events
		WHERE user_id = $1
	`, userID).Scan(&total); err != nil {
		return nil, 0, err
	}

	rows, err := r.pool.Query(ctx, `
		SELECT id, user_id, event_type, ip_address, user_agent, created_at
		FROM security_audit_events
		WHERE user_id = $1
		ORDER BY created_at DESC, id DESC
		LIMIT $2 OFFSET $3
	`, userID, limit, offset)
	if err != nil {
		return nil, 0, err
	}
	defer rows.Close()

	events := make([]SecurityAuditEvent, 0, limit)
	for rows.Next() {
		var event SecurityAuditEvent
		var eventType string

		if err := rows.Scan(
			&event.ID,
			&event.UserID,
			&eventType,
			&event.IPAddress,
			&event.UserAgent,
			&event.CreatedAt,
		); err != nil {
			return nil, 0, err
		}

		event.EventType = SecurityEventType(eventType)
		events = append(events, event)
	}

	if err := rows.Err(); err != nil {
		return nil, 0, err
	}
	if errors.Is(rows.Err(), pgx.ErrNoRows) {
		return []SecurityAuditEvent{}, total, nil
	}

	return events, total, nil
}

		return err
	}

type SecurityAuditServiceAPI interface {
	List(
		context.Context,
		string,
		int,
		int,
	) ([]SecurityAuditEvent, int64, error)
}

type SecurityAuditHandler struct {
	service SecurityAuditServiceAPI
}

func NewSecurityAuditHandler(service SecurityAuditServiceAPI) (*SecurityAuditHandler, error) {
	if service == nil {
		return nil, errors.New("security audit handler: service is required")
	}
	return &SecurityAuditHandler{service: service}, nil
}

type securityAuditResponse struct {
	Events []SecurityAuditEvent `json:"events"`
	Total  int64                `json:"total"`
	Page   int                  `json:"page"`
	Limit  int                  `json:"limit"`
}

func (h *SecurityAuditHandler) List(c *gin.Context) {
	user, ok := AuthenticatedUser(c)
	if !ok || user.ID == "" {
		httpx.Unauthorized(c, "AUTHENTICATION_REQUIRED", "Authentication is required.")
		return
	}

	page, limit, valid := parseAuditPagination(c)
	if !valid {
		return
	}

	events, total, err := h.service.List(
		c.Request.Context(),
		user.ID,
		page,
		limit,
	)
	if err != nil {
		httpx.InternalServerError(c)
		return
	}

	httpx.JSON(c, http.StatusOK, securityAuditResponse{
		Events: events,
		Total:  total,
		Page:   page,
		Limit:  limit,
	})
}

func parseAuditPagination(c *gin.Context) (int, int, bool) {
	page := 1
	limit := 20

	if raw := c.Query("page"); raw != "" {
		parsed, err := strconv.Atoi(raw)
		if err != nil || parsed < 1 || parsed > 1_000_000 {
			httpx.BadRequest(c, "INVALID_PAGE", "The page parameter is invalid.")
			return 0, 0, false
		}
		page = parsed
	}

	if raw := c.Query("limit"); raw != "" {
		parsed, err := strconv.Atoi(raw)
		if err != nil || parsed < 1 || parsed > 100 {
			httpx.BadRequest(c, "INVALID_LIMIT", "The limit must be between 1 and 100.")
			return 0, 0, false
		}
		limit = parsed
	}

	return page, limit, true
}

package tenant

import (
	"errors"
	"regexp"
	"strings"
	"time"
)

type Status string

const (
	StatusActive    Status = "active"
	StatusSuspended Status = "suspended"
	StatusDisabled  Status = "disabled"
)

var slugPattern = regexp.MustCompile(`^[a-z0-9]+(?:-[a-z0-9]+)*$`)

type Tenant struct {
	ID          string    `json:"id"`
	Name        string    `json:"name"`
	Slug        string    `json:"slug"`
	OwnerUserID string    `json:"owner_user_id"`
	Status      Status    `json:"status"`
	CreatedAt   time.Time `json:"created_at"`
	UpdatedAt   time.Time `json:"updated_at"`
}

func NormalizeSlug(value string) (string, error) {
	slug := strings.ToLower(strings.TrimSpace(value))
	if len(slug) < 3 || len(slug) > 63 || !slugPattern.MatchString(slug) {
		return "", ErrInvalidSlug
	}
	return slug, nil
}

func (t Tenant) Validate() error {
	if strings.TrimSpace(t.ID) == "" {
		return errors.New("tenant ID is required")
	}
	if name := strings.TrimSpace(t.Name); name == "" || len(name) > 120 {
		return ErrInvalidName
	}
	if _, err := NormalizeSlug(t.Slug); err != nil {
		return err
	}
	if strings.TrimSpace(t.OwnerUserID) == "" {
		return ErrInvalidOwner
	}
	if !validStatus(t.Status) {
		return ErrInvalidStatus
	}
	if t.CreatedAt.IsZero() || t.UpdatedAt.Before(t.CreatedAt) {
		return errors.New("tenant timestamps are invalid")
	}
	return nil
}

func validStatus(status Status) bool {
	switch status {
	case StatusActive, StatusSuspended, StatusDisabled:
		return true
	default:
		return false
	}
}

		return Membership{}, ErrCannotChangeOwner
	}

		return Membership{}, ErrInvalidRole
	}

	record, err := s.tenants.GetByID(ctx, tenantID)
	if err != nil {
		return Membership{}, err
	}

		return Membership{}, ErrAccessDenied
	}

	if err != nil {
		// The tenant's initial owner may not yet have a membership record.
		if requesterID != record.OwnerUserID {
			return Membership{}, ErrAccessDenied
		}
	} else if requester.Status != MembershipActive ||
		(requester.Role != RoleOwner && requester.Role != RoleAdmin) {
		return Membership{}, ErrAccessDenied
	}

	now := s.clock.Now().UTC()
	status := MembershipInvited
	if requesterID == userID {
		status = MembershipActive
	}

	membership := Membership{
		ID:        s.newID("mem"),
		TenantID:  tenantID,
		UserID:    userID,
		Role:      role,
		Status:    status,
		CreatedAt: now,
		UpdatedAt: now,
	}
	if err := membership.Validate(); err != nil {
		return Membership{}, err
	}
	if err := s.memberships.CreateMembership(ctx, membership); err != nil {
		return Membership{}, err
	}
	return membership, nil
}

func NewTenantHTTPHandler(service TenantServiceAPI) (*TenantHTTPHandler, error) {
	if service == nil {
		return nil, errors.New("tenant handler: service is required")
	}
	return &TenantHTTPHandler{service: service}, nil
}

type createTenantRequest struct {
	Name string `json:"name" validate:"required,min=1,max=120"`
	Slug string `json:"slug" validate:"required,min=3,max=63"`
}

func (h *TenantHTTPHandler) Create(c *gin.Context) {
	user, ok := identity.AuthenticatedUser(c)
	if !ok || user.ID == "" {
		httpx.Unauthorized(c, "AUTHENTICATION_REQUIRED", "Authentication is required.")
		return
	}

	var request createTenantRequest
	if !httpx.BindJSON(c, &request) {
		return
	}

	record, err := h.service.Create(c.Request.Context(), CreateInput{
		Name: request.Name,
		Slug: request.Slug,
		OwnerUserID: user.ID,
	})
	if err != nil {
		switch {
		case errors.Is(err, ErrSlugTaken):
			httpx.Conflict(c, "TENANT_SLUG_TAKEN", "That tenant address is already in use.")
		case errors.Is(err, ErrInvalidSlug), errors.Is(err, ErrInvalidName):
			httpx.BadRequest(c, "INVALID_TENANT", "The tenant details are invalid.")
		default:
			httpx.InternalServerError(c)
		}
		return
	}

	httpx.Created(c, record)
}

func (h *TenantHTTPHandler) Get(c *gin.Context) {
	user, ok := identity.AuthenticatedUser(c)
	if !ok || user.ID == "" {
		httpx.Unauthorized(c, "AUTHENTICATION_REQUIRED", "Authentication is required.")
		return
	}

	record, err := h.service.Get(c.Request.Context(), c.Param("tenant_id"))
	if err != nil {
		if errors.Is(err, ErrNotFound) {
			httpx.NotFound(c, "TENANT_NOT_FOUND", "The requested tenant was not found.")
			return
		}
		if errors.Is(err, ErrInvalidTenantID) {
			httpx.BadRequest(c, "INVALID_TENANT_ID", "The tenant identifier is invalid.")
			return
		}
		httpx.InternalServerError(c)
		return
	}

	// Do not expose tenant details to unrelated authenticated users.
	if record.OwnerUserID != user.ID {
		httpx.Forbidden(c, "TENANT_ACCESS_DENIED", "You do not have access to this tenant.")
		return
	}

	httpx.JSON(c, http.StatusOK, record)
}

func (h *TenantHTTPHandler) ListOwned(c *gin.Context) {
	user, ok := identity.AuthenticatedUser(c)
	if !ok || user.ID == "" {
		httpx.Unauthorized(c, "AUTHENTICATION_REQUIRED", "Authentication is required.")
		return
	}

	page, pageSize := 1, 20
	if raw := c.Query("page"); raw != "" {
		value, err := strconv.Atoi(raw)
		if err != nil || value < 1 || value > 1_000_000 {
			httpx.BadRequest(c, "INVALID_PAGE", "The page parameter is invalid.")
			return
		}
		page = value
	}
	if raw := c.Query("limit"); raw != "" {
		value, err := strconv.Atoi(raw)
		if err != nil || value < 1 || value > 100 {
			httpx.BadRequest(c, "INVALID_LIMIT", "The limit must be between 1 and 100.")
			return
		}
		pageSize = value
	}

	records, total, err := h.service.ListOwned(
		c.Request.Context(),
		user.ID,
		page,
		pageSize,
	)
	if err != nil {
		httpx.InternalServerError(c)
		return
	}

	httpx.JSON(c, http.StatusOK, gin.H{
		"tenants": records,
		"total": total,
		"page": page,
		"limit": pageSize,
	})
}

	PermissionMemberInvite     Permission = "member:invite"
	PermissionMemberRemove     Permission = "member:remove"
	PermissionMemberRoleUpdate Permission = "member:role:update"

func RoleHasPermission(role Role, permission Permission) bool {
	switch role {
	case RoleOwner:
		return validPermission(permission)

	case RoleAdmin:
		switch permission {
		case PermissionTenantRead,
			PermissionTenantUpdate,
			PermissionMemberRead,
			PermissionMemberInvite,
			PermissionMemberRemove,
			PermissionMemberRoleUpdate,
			PermissionCourseRead,
			PermissionCourseCreate,
			PermissionCourseUpdate,
			PermissionCourseDelete,
			PermissionEnrollmentRead,
			PermissionEnrollmentManage:
			return true
		}

	case RoleTeacher:
		switch permission {
		case PermissionTenantRead,
			PermissionCourseRead,
			PermissionCourseCreate,
			PermissionCourseUpdate,
			PermissionEnrollmentRead:
			return true
		}

	case RoleStudent:
		switch permission {
		case PermissionTenantRead,
			PermissionCourseRead:
			return true
		}
	}

	return false
}

		return nil, errors.New("tenant authorizer: membership lookup is required")
	}

}

func (a *Authorizer) Authorize(
	ctx context.Context,
	tenantID string,
	userID string,
	permission Permission,
) error {
	if ctx == nil {
		return errors.New("tenant authorizer: context is required")
	}
	if strings.TrimSpace(tenantID) == "" ||
		strings.TrimSpace(userID) == "" ||
		!validPermission(permission) {
		return ErrAccessDenied
	}

	membership, err := a.memberships.GetMembership(ctx, tenantID, userID)
	if err != nil {
		return ErrAccessDenied
	}
	if membership.Status != MembershipActive {
		return ErrAccessDenied
	}
	if !RoleHasPermission(membership.Role, permission) {
		return ErrAccessDenied
	}
	return nil
