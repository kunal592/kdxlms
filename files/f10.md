## File 10 — `internal/platform/cache/redis_test.go`

Purpose: Test Redis configuration validation, nil handling, client cleanup, and connectivity failures without requiring a running Redis server for every test.

```

package cache

import (
	"context"
	"testing"
	"time"

	"github.com/redis/go-redis/v9"

	"github.com/kdxlabs/kdxlabsllm/internal/config"
)

func validRedisConfig() config.RedisConfig {
	return config.RedisConfig{
		Address:      "127.0.0.1:6379",
		Username:     "",
		Password:     "",
		DB:           0,
		TLS:          false,
		DialTimeout:  200 * time.Millisecond,
		ReadTimeout:  200 * time.Millisecond,
		WriteTimeout: 200 * time.Millisecond,
		PoolSize:     5,
	}
}

func TestNewRedisClientRejectsInvalidConfiguration(t *testing.T) {
	tests := []struct {
		name   string
		mutate func(*config.RedisConfig)
	}{
		{
			name: "empty address",
			mutate: func(cfg *config.RedisConfig) {
				cfg.Address = ""
			},
		},
		{
			name: "whitespace address",
			mutate: func(cfg *config.RedisConfig) {
				cfg.Address = "   "
			},
		},
		{
			name: "negative database index",
			mutate: func(cfg *config.RedisConfig) {
				cfg.DB = -1
			},
		},
		{
			name: "zero dial timeout",
			mutate: func(cfg *config.RedisConfig) {
				cfg.DialTimeout = 0
			},
		},
		{
			name: "zero read timeout",
			mutate: func(cfg *config.RedisConfig) {
				cfg.ReadTimeout = 0
			},
		},
		{
			name: "zero write timeout",
			mutate: func(cfg *config.RedisConfig) {
				cfg.WriteTimeout = 0
			},
		},
		{
			name: "zero pool size",
			mutate: func(cfg *config.RedisConfig) {
				cfg.PoolSize = 0
			},
		},
	}

	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			cfg := validRedisConfig()
			tt.mutate(&cfg)

			client, err := NewRedisClient(context.Background(), cfg)
			if err == nil {
				if client != nil {
					_ = client.Close()
				}
				t.Fatal("expected invalid configuration to fail")
			}

			if client != nil {
				_ = client.Close()
				t.Fatal("expected nil client on initialization failure")
			}
		})
	}
}

func TestNewRedisClientRejectsNilContext(t *testing.T) {
	var ctx context.Context

	client, err := NewRedisClient(ctx, validRedisConfig())
	if err == nil {
		if client != nil {
			_ = client.Close()
		}
		t.Fatal("expected nil context to fail")
	}

	if client != nil {
		_ = client.Close()
		t.Fatal("expected nil client")
	}
}

func TestNewRedisClientFailsWhenServerUnavailable(t *testing.T) {
	cfg := validRedisConfig()
	cfg.Address = "127.0.0.1:0"
	cfg.DialTimeout = 100 * time.Millisecond
	cfg.ReadTimeout = 100 * time.Millisecond
	cfg.WriteTimeout = 100 * time.Millisecond

	ctx, cancel := context.WithTimeout(
		context.Background(),
		500*time.Millisecond,
	)
	defer cancel()

	client, err := NewRedisClient(ctx, cfg)
	if err == nil {
		if client != nil {
			_ = client.Close()
		}
		t.Fatal("expected connectivity verification to fail")
	}

	if client != nil {
		_ = client.Close()
		t.Fatal("expected nil client when connectivity verification fails")
	}
}

func TestPingRejectsNilContext(t *testing.T) {
	var ctx context.Context

	err := Ping(ctx, nil)
	if err == nil {
		t.Fatal("expected nil context to fail")
	}
}

func TestPingRejectsNilClient(t *testing.T) {
	err := Ping(context.Background(), nil)
	if err == nil {
		t.Fatal("expected nil client to fail")
	}
}

func TestCloseAcceptsNilClient(t *testing.T) {
	var client *redis.Client

	if err := Close(client); err != nil {
		t.Fatalf("Close(nil) returned unexpected error: %v", err)
	}
}

func TestCloseClosesClient(t *testing.T) {
	client := redis.NewClient(&redis.Options{
		Addr: "127.0.0.1:6379",
	})

	if err := Close(client); err != nil {
		t.Fatalf("Close() error = %v", err)
	}
}
```
