package httpx

import (
	"net/http"
	"net/http/httptest"

	"github.com/gin-gonic/gin"

	"github.com/kdxlabs/kdxlabsllm/internal/config"

func TestCORSAllowsConfiguredOrigin(t *testing.T) {
	router := gin.New()
	router.Use(CORS(config.SecurityConfig{
		CORSOrigins: []string{"https://app.example.com"},
	}))

	router.GET("/test", func(c *gin.Context) {
		c.Status(http.StatusNoContent)
	})

	recorder := httptest.NewRecorder()
	request := httptest.NewRequest(http.MethodGet, "/test", nil)
	request.Header.Set("Origin", "https://app.example.com")

	router.ServeHTTP(recorder, request)

	if recorder.Code != http.StatusNoContent {
		t.Fatalf("status = %d, want %d", recorder.Code, http.StatusNoContent)
	}

	if got := recorder.Header().Get("Access-Control-Allow-Origin"); got != "https://app.example.com" {
		t.Fatalf("allow origin = %q", got)
	}

	if got := recorder.Header().Get("Vary"); got != "Origin" {
		t.Fatalf("Vary = %q, want Origin", got)
	}
}

func TestCORSHandlesPreflight(t *testing.T) {
	router := gin.New()
	router.Use(CORS(config.SecurityConfig{
		CORSOrigins: []string{"https://app.example.com"},
	}))

	handlerCalled := false

	router.OPTIONS("/test", func(c *gin.Context) {
		handlerCalled = true
		c.Status(http.StatusOK)
	})

	recorder := httptest.NewRecorder()
	request := httptest.NewRequest(http.MethodOptions, "/test", nil)
	request.Header.Set("Origin", "https://app.example.com")
	request.Header.Set("Access-Control-Request-Method", "POST")

	router.ServeHTTP(recorder, request)

	if recorder.Code != http.StatusNoContent {
		t.Fatalf("status = %d, want %d", recorder.Code, http.StatusNoContent)
	}

	if handlerCalled {
		t.Fatal("preflight request should be handled by CORS middleware")
	}

	if got := recorder.Header().Get("Access-Control-Allow-Methods"); got == "" {
		t.Fatal("expected allowed methods header")
	}

	if got := recorder.Header().Get("Access-Control-Allow-Headers"); got == "" {
		t.Fatal("expected allowed headers")
	}
}

func TestCORSRejectsMalformedOrigins(t *testing.T) {
	tests := []string{
		"javascript:alert(1)",
		"https://app.example.com/path",
		"https://app.example.com?x=1",
		"https://user:pass@app.example.com",
		"https://app.example.com#fragment",
		"not-an-origin",
	}

	for _, origin := range tests {
		t.Run(origin, func(t *testing.T) {
			router := gin.New()
			router.Use(CORS(config.SecurityConfig{
				CORSOrigins: []string{"https://app.example.com"},
			}))

			router.GET("/test", func(c *gin.Context) {
				c.Status(http.StatusNoContent)
			})

			recorder := httptest.NewRecorder()
			request := httptest.NewRequest(http.MethodGet, "/test", nil)
			request.Header.Set("Origin", origin)

			router.ServeHTTP(recorder, request)

			if recorder.Code != http.StatusForbidden {
				t.Fatalf("status = %d, want %d", recorder.Code, http.StatusForbidden)
			}
		})
	}
}

func TestCORSWildcardDoesNotEnableCredentials(t *testing.T) {
	router := gin.New()
	router.Use(CORS(config.SecurityConfig{
		CORSOrigins: []string{"*"},
	}))

	router.GET("/test", func(c *gin.Context) {
		c.Status(http.StatusNoContent)
	})

	recorder := httptest.NewRecorder()
	request := httptest.NewRequest(http.MethodGet, "/test", nil)
	request.Header.Set("Origin", "https://unknown.example")

	router.ServeHTTP(recorder, request)

	if got := recorder.Header().Get("Access-Control-Allow-Origin"); got != "*" {
		t.Fatalf("allow origin = %q, want *", got)
	}

	if got := recorder.Header().Get("Access-Control-Allow-Credentials"); got != "" {
		t.Fatalf("credentials header should be absent, got %q", got)
	}
}

func TestNormalizeOrigin(t *testing.T) {
	tests := []struct {
		name    string
		input   string
		want    string
		wantOK  bool
	}{
		{
			name:   "HTTPS origin",
			input:  "https://app.example.com",
			want:   "https://app.example.com",
			wantOK: true,
		},
		{
			name:   "HTTP origin",
			input:  "http://localhost:3000",
			want:   "http://localhost:3000",
			wantOK: true,
		},
		{
			name:   "path rejected",
			input:  "https://app.example.com/path",
			wantOK: false,
		},
		{
			name:   "unsupported scheme",
			input:  "ftp://app.example.com",
			wantOK: false,
		},
		{
			name:   "empty host",
			input:  "https://",
			wantOK: false,
		},
	}

	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			got, ok := normalizeOrigin(tt.input)

			if ok != tt.wantOK {
				t.Fatalf("normalizeOrigin() ok = %v, want %v", ok, tt.wantOK)
			}

			if got != tt.want {
				t.Fatalf("normalizeOrigin() = %q, want %q", got, tt.want)
			}
		})
	}
}
