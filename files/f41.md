func ValidateAssessmentAuditCursorRateLimitDecision(
	decision AssessmentAuditCursorRateLimitDecision,
	now time.Time,
) error {
	if decision.Limit <= 0 || decision.Remaining < 0 || decision.Remaining > decision.Limit {
		return ErrAssessmentAuditCursorRateLimitInvalidDecision

		return ErrAssessmentAuditCursorRateLimitInvalidDecision

	}

	}
	return nil

			return decision, nil

		}
	}

		if err := ctx.Err(); err != nil {

				timer.Stop()

			}
		}

package assessment

import (
	"context"
	"sync"
	"time"
)

type AssessmentAuditCursorRateLimitCircuitState uint8

const (
	AssessmentAuditCursorCircuitClosed AssessmentAuditCursorRateLimitCircuitState = iota
	AssessmentAuditCursorCircuitOpen
	AssessmentAuditCursorCircuitHalfOpen
)

type AssessmentAuditCursorRateLimitCircuitBreaker struct {
	mu              sync.Mutex
	state           AssessmentAuditCursorRateLimitCircuitState
	failures        int
	threshold       int
	openDuration    time.Duration
	openedAt        time.Time
	halfOpenInFlight bool
	now             func() time.Time
}

func NewAssessmentAuditCursorRateLimitCircuitBreaker(
	threshold int,
	openDuration time.Duration,
	now func() time.Time,
) *AssessmentAuditCursorRateLimitCircuitBreaker {
	if now == nil {
		now = time.Now
	}
	if threshold < 1 {
		threshold = 3
	}
	if openDuration <= 0 {
		openDuration = 10 * time.Second
	}
	return &AssessmentAuditCursorRateLimitCircuitBreaker{
		threshold: threshold,
		openDuration: openDuration,
		now: now,
	}
}

func (b *AssessmentAuditCursorRateLimitCircuitBreaker) Allow() bool {
	if b == nil {
		return false
	}
	b.mu.Lock()
	defer b.mu.Unlock()

	now := b.now()
	switch b.state {
	case AssessmentAuditCursorCircuitClosed:
		return true
	case AssessmentAuditCursorCircuitOpen:
		if now.Sub(b.openedAt) < b.openDuration {
			return false
		}
		b.state = AssessmentAuditCursorCircuitHalfOpen
		b.halfOpenInFlight = true
		return true
	case AssessmentAuditCursorCircuitHalfOpen:

	if b == nil {
		return
	}
	b.mu.Lock()
	b.state = AssessmentAuditCursorCircuitClosed
	b.failures = 0
	b.halfOpenInFlight = false
	b.mu.Unlock()
}

func (b *AssessmentAuditCursorRateLimitCircuitBreaker) RecordFailure() {
	if b == nil {
		return
	}
	b.mu.Lock()
	defer b.mu.Unlock()

	b.halfOpenInFlight = false
	b.failures++
	if b.failures >= b.threshold || b.state == AssessmentAuditCursorCircuitHalfOpen {
		b.state = AssessmentAuditCursorCircuitOpen
		b.openedAt = b.now()
	}
}

func (b *AssessmentAuditCursorRateLimitCircuitBreaker) State() AssessmentAuditCursorRateLimitCircuitState {
	if b == nil {
		return AssessmentAuditCursorCircuitOpen
	}
	b.mu.Lock()
	defer b.mu.Unlock()
	return b.state
}

type AssessmentAuditCursorCircuitRateLimitStore struct {
	Inner    AssessmentAuditCursorRateLimitStore
	Breaker  *AssessmentAuditCursorRateLimitCircuitBreaker
}

func (s AssessmentAuditCursorCircuitRateLimitStore) Consume(
	ctx context.Context,
	key string,
	limit int,
	window time.Duration,
) (AssessmentAuditCursorRateLimitDecision, error) {
	if s.Inner == nil || s.Breaker == nil || !s.Breaker.Allow() {
		return AssessmentAuditCursorRateLimitDecision{}, ErrAssessmentAuditCursorRateLimitStore
	}
	decision, err := s.Inner.Consume(ctx, key, limit, window)
	if err != nil {
		s.Breaker.RecordFailure()
		return AssessmentAuditCursorRateLimitDecision{}, err
	}
	s.Breaker.RecordSuccess()
	return decision, nil
}
