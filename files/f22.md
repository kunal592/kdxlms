package httpx

import (
	"encoding/json"
	"net/http"
	"net/http/httptest"
	"strings"
	"testing"

	"github.com/gin-gonic/gin"
	"go.uber.org/zap"
	"go.uber.org/zap/zapcore"
)

func TestRecoveryReturnsInternalServerError(t *testing.T) {
	router := gin.New()
	router.Use(RequestID())
	router.Use(Recovery(zap.NewNop()))

	router.GET("/panic", func(c *gin.Context) {
		panic("sensitive internal detail")
	})

	recorder := httptest.NewRecorder()
	request := httptest.NewRequest(http.MethodGet, "/panic", nil)
	request.Header.Set(RequestIDHeader, "panic-test-123")

	router.ServeHTTP(recorder, request)

	if recorder.Code != http.StatusInternalServerError {
		t.Fatalf(
			"status = %d, want %d",
			recorder.Code,
			http.StatusInternalServerError,
		)
	}

	var response Envelope
	if err := json.Unmarshal(recorder.Body.Bytes(), &response); err != nil {
		t.Fatalf("decode response: %v", err)
	}

	if response.Error == nil {
		t.Fatal("expected error details")
	}

	if response.Error.Code != "INTERNAL_ERROR" {
		t.Errorf("error code = %q, want INTERNAL_ERROR", response.Error.Code)
	}

	if response.RequestID != "panic-test-123" {
		t.Errorf("request ID = %q, want panic-test-123", response.RequestID)
	}

	if strings.Contains(recorder.Body.String(), "sensitive internal detail") {
		t.Fatal("response exposed panic details")
	}
}

func TestRecoveryLogsPanic(t *testing.T) {
	var output strings.Builder

	core := zapcore.NewCore(
		zapcore.NewJSONEncoder(zap.NewProductionEncoderConfig()),
		zapcore.AddSync(&output),
		zapcore.ErrorLevel,
	)
	logger := zap.New(core)

	router := gin.New()
	router.Use(RequestID())
	router.Use(Recovery(logger))

	router.GET("/panic", func(c *gin.Context) {
		panic("test panic")
	})

	recorder := httptest.NewRecorder()
	request := httptest.NewRequest(http.MethodGet, "/panic", nil)
	request.Header.Set(RequestIDHeader, "recovery-log-test")

	router.ServeHTTP(recorder, request)

	if !strings.Contains(output.String(), "panic recovered") {
		t.Fatal("expected panic recovery log")
	}

	if !strings.Contains(output.String(), "recovery-log-test") {
		t.Fatal("expected request ID in recovery log")
	}
}

func TestRecoveryHandlesNilLogger(t *testing.T) {
	router := gin.New()
	router.Use(Recovery(nil))

	router.GET("/panic", func(c *gin.Context) {
		panic("test panic")
	})

	recorder := httptest.NewRecorder()
	request := httptest.NewRequest(http.MethodGet, "/panic", nil)

	router.ServeHTTP(recorder, request)

	if recorder.Code != http.StatusInternalServerError {
		t.Fatalf(
			"status = %d, want %d",
			recorder.Code,
			http.StatusInternalServerError,
		)
	}
}

func TestRecoveryDoesNotReplaceWrittenResponse(t *testing.T) {
	router := gin.New()
	router.Use(Recovery(zap.NewNop()))

	router.GET("/partial", func(c *gin.Context) {
		c.Writer.WriteHeader(http.StatusOK)
		_, _ = c.Writer.Write([]byte("partial"))
		panic("panic after response started")
	})

	recorder := httptest.NewRecorder()
	request := httptest.NewRequest(http.MethodGet, "/partial", nil)

	router.ServeHTTP(recorder, request)

	if recorder.Code != http.StatusOK {
		t.Fatalf("status = %d, want %d", recorder.Code, http.StatusOK)
	}

	if recorder.Body.String() != "partial" {
		t.Fatalf("body = %q, want partial", recorder.Body.String())
	}
}
