## File 7 — `internal/platform/database/postgres.go`

Purpose: Production PostgreSQL connection pool for KDXLabsLLM using `pgxpool`. This file establishes database connectivity, applies pool limits, verifies the database connection, and handles initialization failures safely.

```

package database

import (
	"context"
	"errors"
	"fmt"
	"time"

	"github.com/jackc/pgx/v5/pgxpool"

	"github.com/kdxlabs/kdxlabsllm/internal/config"
)

// NewPostgresPool creates and verifies a PostgreSQL connection pool.
//
// The caller owns the returned pool and must call Close during application
// shutdown. The provided context controls initialization and connectivity
// verification.
func NewPostgresPool(
	ctx context.Context,
	cfg config.DatabaseConfig,
) (*pgxpool.Pool, error) {
	if ctx == nil {
		return nil, errors.New("database initialization context must not be nil")
	}

	if cfg.DSN == "" {
		return nil, errors.New("database DSN must not be empty")
	}

	if cfg.MaxOpenConns < 1 {
		return nil, errors.New("database maximum open connections must be positive")
	}

	if cfg.MaxIdleConns < 0 || cfg.MaxIdleConns > cfg.MaxOpenConns {
		return nil, errors.New("invalid database idle connection limit")
	}

	if cfg.ConnectTimeout <= 0 {
		return nil, errors.New("database connect timeout must be positive")
	}

	if cfg.ConnMaxLifetime <= 0 || cfg.ConnMaxIdleTime <= 0 {
		return nil, errors.New("database connection lifetimes must be positive")
	}

	poolConfig, err := pgxpool.ParseConfig(cfg.DSN)
	if err != nil {
		return nil, fmt.Errorf("parse PostgreSQL configuration: %w", err)
	}

	poolConfig.MaxConns = int32(cfg.MaxOpenConns)
	poolConfig.MinConns = 0
	poolConfig.MaxConnLifetime = cfg.ConnMaxLifetime
	poolConfig.MaxConnIdleTime = cfg.ConnMaxIdleTime
	poolConfig.ConnConfig.ConnectTimeout = cfg.ConnectTimeout

	pool, err := pgxpool.NewWithConfig(ctx, poolConfig)
	if err != nil {
		return nil, fmt.Errorf("create PostgreSQL connection pool: %w", err)
	}

	pingCtx, cancel := context.WithTimeout(ctx, cfg.ConnectTimeout)
	defer cancel()

	if err := pool.Ping(pingCtx); err != nil {
		pool.Close()
		return nil, fmt.Errorf("verify PostgreSQL connectivity: %w", err)
	}

	return pool, nil
}

// Ping verifies that PostgreSQL is reachable within the supplied context.
// Callers should use a bounded context for health checks.
func Ping(ctx context.Context, pool *pgxpool.Pool) error {
	if ctx == nil {
		return errors.New("database ping context must not be nil")
	}

	if pool == nil {
		return errors.New("database pool must not be nil")
	}

	if err := pool.Ping(ctx); err != nil {
		return fmt.Errorf("PostgreSQL ping failed: %w", err)
	}

	return nil
}

// Close releases resources held by the connection pool.
func Close(pool *pgxpool.Pool) {
	if pool != nil {
		pool.Close()
	}
}

// HealthCheckTimeout returns a bounded context for database health checks.
func HealthCheckTimeout(
	parent context.Context,
	timeout time.Duration,
) (context.Context, context.CancelFunc, error) {
	if parent == nil {
		return nil, nil, errors.New("health check context must not be nil")
	}

	if timeout <= 0 {
		return nil, nil, errors.New("health check timeout must be positive")
	}

	ctx, cancel := context.WithTimeout(parent, timeout)
	return ctx, cancel, nil
}
```
