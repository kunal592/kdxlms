## File 8 — `internal/platform/database/postgres_test.go`

Purpose: Test PostgreSQL pool configuration validation, nil handling, context handling, and health-check timeout behavior without requiring a live database for every test.

```

package database

import (
	"context"
	"testing"
	"time"

	"github.com/jackc/pgx/v5/pgxpool"

	"github.com/kdxlabs/kdxlabsllm/internal/config"
)

func validDatabaseConfig() config.DatabaseConfig {
	return config.DatabaseConfig{
		DSN:             "postgres://user:password@localhost:5432/kdxlabsllm?sslmode=disable",
		MaxOpenConns:    10,
		MaxIdleConns:    5,
		ConnMaxLifetime: 30 * time.Minute,
		ConnMaxIdleTime: 5 * time.Minute,
		ConnectTimeout:  2 * time.Second,
	}
}

func TestNewPostgresPoolRejectsInvalidConfiguration(t *testing.T) {
	tests := []struct {
		name   string
		mutate func(*config.DatabaseConfig)
	}{
		{
			name: "empty DSN",
			mutate: func(cfg *config.DatabaseConfig) {
				cfg.DSN = ""
			},
		},
		{
			name: "zero maximum connections",
			mutate: func(cfg *config.DatabaseConfig) {
				cfg.MaxOpenConns = 0
			},
		},
		{
			name: "negative idle connections",
			mutate: func(cfg *config.DatabaseConfig) {
				cfg.MaxIdleConns = -1
			},
		},
		{
			name: "idle connections exceed maximum",
			mutate: func(cfg *config.DatabaseConfig) {
				cfg.MaxIdleConns = cfg.MaxOpenConns + 1
			},
		},
		{
			name: "zero connect timeout",
			mutate: func(cfg *config.DatabaseConfig) {
				cfg.ConnectTimeout = 0
			},
		},
		{
			name: "zero connection lifetime",
			mutate: func(cfg *config.DatabaseConfig) {
				cfg.ConnMaxLifetime = 0
			},
		},
		{
			name: "zero idle lifetime",
			mutate: func(cfg *config.DatabaseConfig) {
				cfg.ConnMaxIdleTime = 0
			},
		},
		{
			name: "invalid DSN",
			mutate: func(cfg *config.DatabaseConfig) {
				cfg.DSN = "://invalid"
			},
		},
	}

	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			cfg := validDatabaseConfig()
			tt.mutate(&cfg)

			pool, err := NewPostgresPool(context.Background(), cfg)
			if err == nil {
				if pool != nil {
					pool.Close()
				}
				t.Fatal("expected invalid configuration to fail")
			}

			if pool != nil {
				pool.Close()
				t.Fatal("expected nil pool on initialization failure")
			}
		})
	}
}

func TestNewPostgresPoolRejectsNilContext(t *testing.T) {
	var ctx context.Context

	if err == nil {

		}

	if pool != nil {
		pool.Close()
		t.Fatal("expected nil pool")
	}
}

func TestHealthCheckTimeout(t *testing.T) {
	ctx, cancel, err := HealthCheckTimeout(
		context.Background(),
		2*time.Second,
	)
	if err != nil {
		t.Fatalf("HealthCheckTimeout() error = %v", err)
	}
	defer cancel()

	deadline, ok := ctx.Deadline()
	if !ok {
		t.Fatal("expected health check context to have a deadline")
	}

	if time.Until(deadline) <= 0 {
		t.Fatal("expected health check deadline to be in the future")
	}
}

func TestHealthCheckTimeoutRejectsInvalidArguments(t *testing.T) {
	tests := []struct {
		name    string
		parent  context.Context
		timeout time.Duration
	}{
		{
			name:    "nil context",
			parent:  nil,
			timeout: time.Second,
		},
		{
			name:    "zero timeout",
			parent:  context.Background(),
			timeout: 0,
		},
		{
			name:    "negative timeout",
			parent:  context.Background(),
			timeout: -time.Second,
		},
	}

	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			ctx, cancel, err := HealthCheckTimeout(tt.parent, tt.timeout)

			if err == nil {
				if cancel != nil {
					cancel()
				}
				t.Fatal("expected invalid arguments to fail")
			}

			if ctx != nil {
				t.Fatal("expected nil context on error")
			}

			if cancel != nil {
				t.Fatal("expected nil cancel function on error")
			}
		})
	}
}
```
