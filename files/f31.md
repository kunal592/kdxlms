package enrollment

import (
	"errors"
	"strings"
)

type ProgressSummary struct {
	TenantID         string  `json:"tenant_id"`
	CourseID         string  `json:"course_id"`
	EnrollmentID     string  `json:"enrollment_id"`
	PublishedLessons int64   `json:"published_lessons"`
	CompletedLessons int64   `json:"completed_lessons"`
	CompletionRate   float64 `json:"completion_rate"`
}

func (s ProgressSummary) Validate() error {
	if strings.TrimSpace(s.TenantID) == "" ||
		strings.TrimSpace(s.CourseID) == "" ||
		strings.TrimSpace(s.EnrollmentID) == "" {
		return ErrInvalidProgress
	}
	if s.PublishedLessons < 0 ||
		s.CompletedLessons < 0 ||
		s.CompletedLessons > s.PublishedLessons {
		return errors.New("progress summary counts are invalid")
	}

	expected := 0.0
	if s.PublishedLessons > 0 {
		expected = float64(s.CompletedLessons) * 100 / float64(s.PublishedLessons)
	}
	if s.CompletionRate < 0 || s.CompletionRate > 100 ||
		(s.CompletionRate-expected > 0.0001) ||
		(expected-s.CompletionRate > 0.0001) {
		return errors.New("progress summary completion rate is inconsistent")
	}
	return nil
}

func NewProgressSummary(
	tenantID, courseID, enrollmentID string,
	published, completed int64,
) (ProgressSummary, error) {
	summary := ProgressSummary{
		TenantID: tenantID,
		CourseID: courseID,
		EnrollmentID: enrollmentID,
		PublishedLessons: published,
		CompletedLessons: completed,
	}
	if published > 0 {
		summary.CompletionRate = float64(completed) * 100 / float64(published)
	}
	if err := summary.Validate(); err != nil {
		return ProgressSummary{}, err
	}
	return summary, nil
}

type PostgresProgressSummaryRepository struct {
	db *pgxpool.Pool
}

func NewPostgresProgressSummaryRepository(
	db *pgxpool.Pool,
) (*PostgresProgressSummaryRepository, error) {
	if db == nil {
		return nil, errors.New("progress summary repository: database pool is required")
	}
	return &PostgresProgressSummaryRepository{db: db}, nil
}

func (r *PostgresProgressSummaryRepository) GetProgressSummary(
	ctx context.Context,
	tenantID, courseID, enrollmentID string,
) (ProgressSummary, error) {
	if ctx == nil {
		return ProgressSummary{}, errors.New("progress summary repository: context is required")
	}
	if tenantID == "" || courseID == "" || enrollmentID == "" {
		return ProgressSummary{}, ErrInvalidProgress
	}

	var published, completed int64
	err := r.db.QueryRow(ctx, `
		SELECT
			COUNT(l.id) FILTER (WHERE l.status = 'published') AS published_lessons,
			COUNT(p.lesson_id) FILTER (
				WHERE l.status = 'published' AND p.completed = TRUE
			) AS completed_lessons
		FROM course_enrollments e
		LEFT JOIN course_lessons l
			ON l.tenant_id = e.tenant_id
			AND l.course_id = e.course_id
		LEFT JOIN lesson_progress p
			ON p.tenant_id = e.tenant_id
			AND p.course_id = e.course_id
			AND p.enrollment_id = e.id
			AND p.lesson_id = l.id
		WHERE e.tenant_id = $1
			AND e.course_id = $2
			AND e.id = $3
		GROUP BY e.id
	`, tenantID, courseID, enrollmentID).Scan(&published, &completed)

	if errors.Is(err, pgx.ErrNoRows) {
		return ProgressSummary{}, ErrEnrollmentNotFound
	}
	if err != nil {
		return ProgressSummary{}, fmt.Errorf("get progress summary: %w", err)
	}

	summary, err := NewProgressSummary(
		tenantID, courseID, enrollmentID, published, completed,
	)
	if err != nil {
		return ProgressSummary{}, fmt.Errorf("build progress summary: %w", err)
	}
	return summary, nil
}

	ctx context.Context,
	tenantID, courseID, certificateID string,
) (Certificate, error) {
	if ctx == nil {
		return Certificate{}, errors.New("certificate repository: context is required")
	}
	return scanCertificate(r.db.QueryRow(ctx, `
		SELECT id, tenant_id, course_id, enrollment_id,
		       user_id, serial_number, issued_at
		FROM course_certificates
		WHERE tenant_id = $1 AND course_id = $2 AND id = $3
	`, tenantID, courseID, certificateID))
}

func (r *PostgresCertificateRepository) GetCertificateByEnrollment(
	ctx context.Context,
	tenantID, courseID, enrollmentID string,
) (Certificate, error) {
	if ctx == nil {
		return Certificate{}, errors.New("certificate repository: context is required")
	}
	return scanCertificate(r.db.QueryRow(ctx, `
		SELECT id, tenant_id, course_id, enrollment_id,
		       user_id, serial_number, issued_at
		FROM course_certificates
		WHERE tenant_id = $1 AND course_id = $2 AND enrollment_id = $3
	`, tenantID, courseID, enrollmentID))
}

func scanCertificate(row pgx.Row) (Certificate, error) {
	var c Certificate
	err := row.Scan(
		&c.ID, &c.TenantID, &c.CourseID, &c.EnrollmentID,
		&c.UserID, &c.SerialNumber, &c.IssuedAt,
	)
	if errors.Is(err, pgx.ErrNoRows) {
		return Certificate{}, ErrCertificateNotFound
	}
	if err != nil {
		return Certificate{}, fmt.Errorf("scan certificate: %w", err)
	}
	if err := c.Validate(); err != nil {
		return Certificate{}, fmt.Errorf("invalid stored certificate: %w", err)
	}
	return c, nil
}

func mapCertificateWriteError(err error) error {
	var pgErr *pgconn.PgError
	if !errors.As(err, &pgErr) {
		return fmt.Errorf("write certificate: %w", err)
	}

	switch {
	case pgErr.Code == "23505" &&
		(pgErr.ConstraintName == "course_certificates_enrollment_unique" ||
			pgErr.ConstraintName == "course_certificates_serial_number_key"):
		return ErrCertificateExists
	case pgErr.Code == "23503" &&
		pgErr.ConstraintName == "course_certificates_enrollment_fk":
		return ErrEnrollmentNotFound
	case pgErr.Code == "23503" &&
		pgErr.ConstraintName == "course_certificates_user_fk":
		return ErrUserNotFound
	default:
		return fmt.Errorf("write certificate: %w", err)
	}
}

package enrollment

import (
	"context"
	"errors"
	"strings"
	"time"
)

type CertificateClock interface {
	Now() time.Time
}

type CertificateIDGenerator func(prefix string) string
type CertificateSerialGenerator func() (string, error)

type CertificateService struct {
	certificates CertificateRepository
	enrollments  Repository
	summaries    ProgressSummaryRepository
	clock        CertificateClock
	newID        CertificateIDGenerator
	newSerial    CertificateSerialGenerator
}

func NewCertificateService(
	certificates CertificateRepository,
	enrollments Repository,
	summaries ProgressSummaryRepository,
	clock CertificateClock,
	newID CertificateIDGenerator,
	newSerial CertificateSerialGenerator,
) (*CertificateService, error) {
	if certificates == nil || enrollments == nil || summaries == nil ||
		clock == nil || newID == nil || newSerial == nil {
		return nil, errors.New("certificate service: all dependencies are required")
	}

	return &CertificateService{
		certificates: certificates,
		enrollments:  enrollments,
		summaries:    summaries,
		clock:        clock,
		newID:        newID,
		newSerial:    newSerial,
	}, nil
}

// IssueForUser issues a certificate only after every currently published
// lesson in the course has been completed by the specified enrolled user.
func (s *CertificateService) IssueForUser(
	ctx context.Context,
	tenantID, courseID, enrollmentID, userID string,
) (Certificate, error) {
	if ctx == nil {
		return Certificate{}, errors.New("certificate service: context is required")
	}
	tenantID = strings.TrimSpace(tenantID)
	courseID = strings.TrimSpace(courseID)
	enrollmentID = strings.TrimSpace(enrollmentID)
	userID = strings.TrimSpace(userID)
	if tenantID == "" || courseID == "" || enrollmentID == "" || userID == "" {
		return Certificate{}, ErrInvalidCertificate
	}

	enrollmentRecord, err := s.enrollments.GetByID(
		ctx, tenantID, courseID, enrollmentID,
	)
	if err != nil {
		if errors.Is(err, ErrNotFound) {
			return Certificate{}, ErrEnrollmentNotFound
		}
		return Certificate{}, err
	}
	if enrollmentRecord.UserID != userID {
		// Avoid disclosing another user's enrollment.
		return Certificate{}, ErrEnrollmentNotFound
	}
	if enrollmentRecord.Status != StatusActive && enrollmentRecord.Status != StatusCompleted {
		return Certificate{}, ErrInvalidTransition
	}

	// Return an existing certificate rather than issuing a second one.
	existing, err := s.certificates.GetCertificateByEnrollment(
		ctx, tenantID, courseID, enrollmentID,
	)
	if err == nil {
		if existing.UserID != userID {
			return Certificate{}, ErrCertificateNotFound
		}
		return existing, nil
	}
	if !errors.Is(err, ErrCertificateNotFound) {
		return Certificate{}, err
	}

	summary, err := s.summaries.GetProgressSummary(
		ctx, tenantID, courseID, enrollmentID,
	)
	if err != nil {
		return Certificate{}, err
	}
	if err := summary.Validate(); err != nil {
		return Certificate{}, err
	}
	if summary.PublishedLessons == 0 ||
		summary.CompletedLessons != summary.PublishedLessons {
		return Certificate{}, ErrCertificateNotEligible
	}

	serial, err := s.newSerial()
	if err != nil {
		return Certificate{}, errors.New("certificate serial generation failed")
	}
	serial = strings.TrimSpace(serial)
	if serial == "" {
		return Certificate{}, errors.New("certificate serial generator returned an empty serial")
	}

	certificate := Certificate{
		ID:           s.newID("cert"),
		TenantID:     tenantID,
		CourseID:     courseID,
		EnrollmentID: enrollmentID,
		UserID:       userID,
		SerialNumber: serial,
		IssuedAt:     s.clock.Now().UTC(),
	}
	if err := certificate.Validate(); err != nil {
		return Certificate{}, err
	}

	if err := s.certificates.CreateCertificate(ctx, certificate); err != nil {
		if errors.Is(err, ErrCertificateExists) {
			// A concurrent request may have issued it after our initial read.
			concurrent, readErr := s.certificates.GetCertificateByEnrollment(
				ctx, tenantID, courseID, enrollmentID,
			)
			if readErr == nil && concurrent.UserID == userID {
				return concurrent, nil
			}
		}
		return Certificate{}, err
	}
	return certificate, nil
}

func (s *CertificateService) GetForUser(
	ctx context.Context,
	tenantID, courseID, enrollmentID, userID string,
) (Certificate, error) {
	if ctx == nil {
		return Certificate{}, errors.New("certificate service: context is required")
	}
	if strings.TrimSpace(userID) == "" {
		return Certificate{}, ErrCertificateNotFound
	}

	enrollmentRecord, err := s.enrollments.GetByID(
		ctx, tenantID, courseID, enrollmentID,
	)
	if err != nil {
		if errors.Is(err, ErrNotFound) {
			return Certificate{}, ErrCertificateNotFound
		}
		return Certificate{}, err
	}
	if enrollmentRecord.UserID != userID {
		return Certificate{}, ErrCertificateNotFound
	}

	certificate, err := s.certificates.GetCertificateByEnrollment(
		ctx, tenantID, courseID, enrollmentID,
	)
	if err != nil {
		return Certificate{}, err
	}
	if certificate.UserID != userID {
		return Certificate{}, ErrCertificateNotFound
	}
	return certificate, nil
}

package enrollment

import (
	"context"
	"errors"
	"strings"

type ProgressSummaryService struct {
	enrollments Repository
	summaries   ProgressSummaryRepository
}

func NewProgressSummaryService(
	enrollments Repository,
	summaries ProgressSummaryRepository,
) (*ProgressSummaryService, error) {
	if enrollments == nil || summaries == nil {
		return nil, errors.New("progress summary service: repositories are required")
	}
	return &ProgressSummaryService{
		enrollments: enrollments,
		summaries:   summaries,
	}, nil
}

func (s *ProgressSummaryService) GetForUser(
	ctx context.Context,
	tenantID, courseID, enrollmentID, userID string,
) (ProgressSummary, error) {
	if ctx == nil {
		return ProgressSummary{}, errors.New("progress summary service: context is required")
	}
	if strings.TrimSpace(tenantID) == "" ||
		strings.TrimSpace(courseID) == "" ||
		strings.TrimSpace(enrollmentID) == "" ||
		strings.TrimSpace(userID) == "" {
		return ProgressSummary{}, ErrInvalidProgress
	}

	record, err := s.enrollments.GetByID(ctx, tenantID, courseID, enrollmentID)
	if err != nil {
		if errors.Is(err, ErrNotFound) {
			return ProgressSummary{}, ErrEnrollmentNotFound
		}
		return ProgressSummary{}, err
	}
	if record.UserID != userID {
		// Avoid exposing another student's enrollment.
		return ProgressSummary{}, ErrEnrollmentNotFound
	}
	if record.Status != StatusActive && record.Status != StatusCompleted {
		return ProgressSummary{}, ErrInvalidTransition
	}

	summary, err := s.summaries.GetProgressSummary(
		ctx, tenantID, courseID, enrollmentID,
	)
	if err != nil {
		return ProgressSummary{}, err
	}
	if err := summary.Validate(); err != nil {
		return ProgressSummary{}, err
	}
	return summary, nil
}

// NewCertificateSerial generates a cryptographically random serial number.
// It does not contain user IDs, course IDs, or other personally identifying data.
func NewCertificateSerial() (string, error) {
	random := make([]byte, 16)
	if _, err := rand.Read(random); err != nil {
		return "", errors.New("generate certificate serial: secure randomness unavailable")
	}
	return "KDX-" + strings.ToUpper(hex.EncodeToString(random)), nil
}

	tenantID, courseID string,

		return nil, ErrInvalidPagination

		SELECT id, tenant_id, course_id, certificate_id,
		       revoked_by, reason, revoked_at
		FROM certificate_revocations
		WHERE tenant_id = $1 AND course_id = $2
		ORDER BY revoked_at DESC, id DESC
		LIMIT $3 OFFSET $4

		return nil, fmt.Errorf("list certificate revocations: %w", err)

		if scanErr != nil {
			return nil, fmt.Errorf("scan certificate revocation: %w", scanErr)
		}
		records = append(records, record)
	}

		pgErr.ConstraintName == "certificate_revocations_certificate_fk":
		return ErrCertificateNotFound
	default:
		return fmt.Errorf("write certificate revocation: %w", err)
	}
}

	revocations CertificateRevocationRepository,
	clock CertificateRevocationClock,
	newID CertificateRevocationIDGenerator,
) (*CertificateRevocationService, error) {
	if certificates == nil || revocations == nil || clock == nil || newID == nil {
		return nil, errors.New("certificate revocation service: all dependencies are required")

	return &CertificateRevocationService{
		certificates: certificates,
		revocations: revocations,
		clock: clock,
		newID: newID,
	}, nil

func (s *CertificateRevocationService) Revoke(
	ctx context.Context,
	tenantID, courseID, certificateID, actorID, reason string,
) (CertificateRevocation, error) {
	if ctx == nil {
		return CertificateRevocation{}, errors.New("certificate revocation service: context is required")
	}

	tenantID = strings.TrimSpace(tenantID)
	courseID = strings.TrimSpace(courseID)
	certificateID = strings.TrimSpace(certificateID)
	actorID = strings.TrimSpace(actorID)
	reason = strings.TrimSpace(reason)

	if tenantID == "" || courseID == "" || certificateID == "" || actorID == "" {
		return CertificateRevocation{}, ErrInvalidRevocation
	}

	// Confirm that the certificate exists in this tenant and course.
	_, err := s.certificates.GetCertificateByID(ctx, tenantID, courseID, certificateID)
	if err != nil {
		if errors.Is(err, ErrCertificateNotFound) {
			return CertificateRevocation{}, ErrCertificateNotFound
		}
		return CertificateRevocation{}, err
	}

	existing, err := s.revocations.GetRevocation(ctx, tenantID, courseID, certificateID)
	if err == nil {
		return existing, ErrCertificateAlreadyRevoked
	}
	if !errors.Is(err, ErrRevocationNotFound) {
		return CertificateRevocation{}, err
	}

	record := CertificateRevocation{
		ID: s.newID("crev"),
		TenantID: tenantID,
		CourseID: courseID,
		CertificateID: certificateID,
		RevokedBy: actorID,
		Reason: reason,
		RevokedAt: s.clock.Now().UTC(),
	}
	if err := record.Validate(); err != nil {
		return CertificateRevocation{}, err
	}
	if err := s.revocations.CreateRevocation(ctx, record); err != nil {
		if errors.Is(err, ErrCertificateAlreadyRevoked) {
			// A concurrent revocation won. Read the persisted record for a
			// consistent response; still report that this request did not win.
			_, _ = s.revocations.GetRevocation(ctx, tenantID, courseID, certificateID)
		}
		return CertificateRevocation{}, err
	}
	return record, nil
}

func (s *CertificateRevocationService) GetStatus(
	ctx context.Context,
	tenantID, courseID, certificateID string,
) (bool, *CertificateRevocation, error) {
	if ctx == nil {
		return false, nil, errors.New("certificate revocation service: context is required")
	}

	_, err := s.certificates.GetCertificateByID(ctx, tenantID, courseID, certificateID)
	if err != nil {
		if errors.Is(err, ErrCertificateNotFound) {
			return false, nil, ErrCertificateNotFound
		}
		return false, nil, err
	}

	record, err := s.revocations.GetRevocation(ctx, tenantID, courseID, certificateID)
	if errors.Is(err, ErrRevocationNotFound) {
		return false, nil, nil
	}
	if err != nil {
		return false, nil, err
	}
	return true, &record, nil
}

import (
	"context"
	"errors"
	"strings"

type CertificateVerification struct {
	Valid       bool      `json:"valid"`
	Certificate *CertificateResponse `json:"certificate,omitempty"`
	Revoked     bool      `json:"revoked"`
	Reason      string    `json:"reason,omitempty"`
}

type CertificateVerificationService struct {
	certificates CertificateRepository
	revocations  CertificateRevocationRepository
}

func NewCertificateVerificationService(
	certificates CertificateRepository,
	revocations CertificateRevocationRepository,
) (*CertificateVerificationService, error) {
	if certificates == nil || revocations == nil {
		return nil, errors.New("certificate verification service: repositories are required")
	}
	return &CertificateVerificationService{
		certificates: certificates,
		revocations: revocations,
	}, nil
}

// Verify checks a certificate within its tenant and course. A certificate is
// valid only if it exists and has no revocation record.
func (s *CertificateVerificationService) Verify(
	ctx context.Context,
	tenantID, courseID, certificateID string,
) (CertificateVerification, error) {
	if ctx == nil {
		return CertificateVerification{}, errors.New("certificate verification service: context is required")
	}
	tenantID = strings.TrimSpace(tenantID)
	courseID = strings.TrimSpace(courseID)
	certificateID = strings.TrimSpace(certificateID)
	if tenantID == "" || courseID == "" || certificateID == "" {
		return CertificateVerification{}, ErrInvalidCertificate
	}

	certificate, err := s.certificates.GetCertificateByID(
		ctx, tenantID, courseID, certificateID,
	)
	if err != nil {
		if errors.Is(err, ErrCertificateNotFound) {
			return CertificateVerification{Valid: false}, nil
		}
		return CertificateVerification{}, err
	}

	revocation, err := s.revocations.GetRevocation(
		ctx, tenantID, courseID, certificateID,
	)
	if errors.Is(err, ErrRevocationNotFound) {
		response := NewCertificateResponse(certificate)
		return CertificateVerification{
			Valid: true,
			Certificate: &response,
			Revoked: false,
		}, nil
	}
	if err != nil {
		return CertificateVerification{}, err
	}

	response := NewCertificateResponse(certificate)
	return CertificateVerification{
		Valid: false,
		Certificate: &response,
		Revoked: true,
		Reason: revocation.Reason,
	}, nil
}

		c.Param("serial_number"),
	)

			c.JSON(http.StatusBadRequest, gin.H{"error": "INVALID_REQUEST"})
			return
		}
		c.JSON(http.StatusInternalServerError, gin.H{"error": "INTERNAL_ERROR"})
		return
	}
	c.JSON(http.StatusOK, gin.H{"data": result})
}

package enrollment

import (
	"errors"
	"strings"
	"unicode"
)

// NormalizeCertificateSerial canonicalizes serial numbers for user input.
// Serial numbers are identifiers, not secrets; callers must still scope
// lookups by tenant and course.
func NormalizeCertificateSerial(value string) (string, error) {
	value = strings.TrimSpace(value)
	if value == "" || len(value) > 100 {
		return "", ErrInvalidCertificate
	}

	var normalized strings.Builder
	normalized.Grow(len(value))
	for _, r := range value {
		if unicode.IsSpace(r) {
			continue
		}
		if !(r >= 'a' && r <= 'z') &&
			!(r >= 'A' && r <= 'Z') &&
			!(r >= '0' && r <= '9') &&
			r != '-' {
			return "", ErrInvalidCertificate
		}
		normalized.WriteRune(unicode.ToUpper(r))
	}

	result := normalized.String()
	if result == "" {
		return "", ErrInvalidCertificate
	}
	if !strings.HasPrefix(result, "KDX-") {
		return "", errors.New("certificate serial has an unsupported prefix")
	}
	return result, nil
}
