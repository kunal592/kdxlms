package email

import (
	"context"
	"errors"
	"net/mail"
	"strings"
)

type Message struct {
	To      string
	Subject string
	Text    string
	HTML    string
}

type Sender interface {
	Send(context.Context, Message) error
}

func ValidateMessage(message Message) error {
	if strings.TrimSpace(message.To) == "" {
		return errors.New("email recipient is required")
	}
	if strings.TrimSpace(message.Subject) == "" {
		return errors.New("email subject is required")
	}
	if strings.TrimSpace(message.Text) == "" && strings.TrimSpace(message.HTML) == "" {
		return errors.New("email body is required")
	}

	address, err := mail.ParseAddress(message.To)
	if err != nil || address.Address != message.To {
		return errors.New("email recipient is invalid")
	}

	if strings.ContainsAny(message.Subject, "\r\n") {
		return errors.New("email subject contains invalid line breaks")
	}

	return nil
}

package identity

import (
	"context"
	"errors"
	"fmt"
	"net/url"
	"strings"
	"time"

	"github.com/kdxlabs/kdxlabsllm/internal/platform/email"
)

type EmailPasswordResetNotifier struct {
	sender   email.Sender
	resetURL string
}

func NewEmailPasswordResetNotifier(
	sender email.Sender,
	resetURL string,
) (*EmailPasswordResetNotifier, error) {
	if sender == nil {
		return nil, errors.New("password reset notifier: email sender is required")
	}

	parsed, err := url.Parse(strings.TrimSpace(resetURL))
	if err != nil || parsed.Host == "" ||
		(parsed.Scheme != "https" && parsed.Scheme != "http") {
		return nil, errors.New("password reset notifier: a valid HTTP(S) reset URL is required")
	}
	if parsed.User != nil {
		return nil, errors.New("password reset notifier: reset URL must not contain credentials")
	}
	if parsed.RawQuery != "" || parsed.Fragment != "" {
		return nil, errors.New("password reset notifier: base URL must not include query or fragment")
	}
	if parsed.Scheme != "https" &&
		parsed.Hostname() != "localhost" &&
		parsed.Hostname() != "127.0.0.1" {
		return nil, errors.New("password reset notifier: HTTPS is required outside localhost")
	}

	return &EmailPasswordResetNotifier{
		sender: sender,
		resetURL: strings.TrimRight(parsed.String(), "/"),
	}, nil
}

func (n *EmailPasswordResetNotifier) SendPasswordReset(
	ctx context.Context,
	to string,
	token string,
	expiresAt time.Time,
) error {
	if ctx == nil {
		return errors.New("password reset notifier: context is required")
	}
	if token == "" {
		return errors.New("password reset notifier: token is required")
	}

	link := n.resetURL + "?token=" + url.QueryEscape(token)
	message := email.Message{
		To: to,
		Subject: "Reset your KDXLabsLLM password",
		Text: fmt.Sprintf(
			"Use this link to reset your KDXLabsLLM password:\n%s\n\n"+
				"This link expires at %s. If you did not request a password reset, ignore this email.",
			link,
			expiresAt.UTC().Format(time.RFC3339),
		),
	}

	if err := email.ValidateMessage(message); err != nil {
		return fmt.Errorf("validate password reset email: %w", err)
	}
	if err := n.sender.Send(ctx, message); err != nil {
		return fmt.Errorf("send password reset email: %w", err)
	}
	return nil
}

	"net/http"

	"github.com/gin-gonic/gin"
	"github.com/kdxlabs/kdxlabsllm/internal/platform/httpx"
)

type LogoutServiceAPI interface {
	Logout(context.Context, string) error
}

type AccountHandler struct {
	logoutService LogoutServiceAPI
}

func NewAccountHandler(logoutService LogoutServiceAPI) *AccountHandler {
	return &AccountHandler{logoutService: logoutService}
}

// Me must be registered behind RequireAuthentication middleware.
func (h *AccountHandler) Me(c *gin.Context) {
	user, ok := AuthenticatedUser(c)
	if !ok {
		httpx.Unauthorized(
			c,
			"AUTHENTICATION_REQUIRED",
			"Authentication is required.",
		)
		return
	}

	httpx.JSON(c, http.StatusOK, gin.H{"user": user})
}

// Logout revokes the bearer token supplied by the client.
func (h *AccountHandler) Logout(c *gin.Context) {
	token, ok := bearerToken(c.GetHeader("Authorization"))
	if !ok {
		httpx.Unauthorized(
			c,
			"AUTHENTICATION_REQUIRED",
			"A valid bearer token is required.",
		)
		return
	}

	if err := h.logoutService.Logout(c.Request.Context(), token); err != nil {
		httpx.InternalServerError(c)
		return
	}

	httpx.JSON(c, http.StatusOK, gin.H{
		"message": "Signed out successfully.",
	})
}

import (
	"context"
	"errors"
	"strings"

	"github.com/kdxlabs/kdxlabsllm/internal/platform/security"

var ErrCurrentPasswordIncorrect = errors.New("current password is incorrect")

func (s *AuthService) ChangePassword(
	ctx context.Context,
	userID string,
	currentPassword string,
	newPassword string,
) error {
	if ctx == nil {
		return errors.New("change password: context is required")
	}
	if strings.TrimSpace(userID) == "" {
		return ErrInvalidUserID
	}
	if err := ValidatePassword(newPassword); err != nil {
		return err
	}
	if currentPassword == "" {
		return ErrCurrentPasswordIncorrect
	}

	user, err := s.users.GetByID(ctx, userID)
	if err != nil {
		return err
	}
	if user.Status != UserStatusActive {
		return ErrAccountDisabled
	}

	matches, err := security.VerifyPassword(currentPassword, user.PasswordHash)
	if err != nil {
		return err
	}
	if !matches {
		return ErrCurrentPasswordIncorrect
	}

	matches, err = security.VerifyPassword(newPassword, user.PasswordHash)
	if err == nil && matches {
		return ErrPasswordReuse
	}

	newHash, err := security.HashPassword(newPassword)
	if err != nil {
		return err
	}

	now := s.clock.Now()
	if err := s.users.UpdatePasswordHash(ctx, userID, newHash, now); err != nil {
		return err
	}

	// Revoke all sessions, including the current session, after a password change.
	return s.sessions.RevokeAllUserSessions(ctx, userID, now)
}

			httpx.Unauthorized(c, "AUTHENTICATION_REQUIRED", "Authentication is required.")
		default:
			httpx.InternalServerError(c)

		return

type LogoutService struct {
	sessions SessionRepository
	clock    Clock
}

func NewLogoutService(sessions SessionRepository, clock Clock) (*LogoutService, error) {
	if sessions == nil || clock == nil {
		return nil, errors.New("logout service: sessions and clock are required")
	}
	return &LogoutService{sessions: sessions, clock: clock}, nil
}

func (s *LogoutService) Logout(ctx context.Context, rawToken string) error {
	if ctx == nil {
		return errors.New("logout service: context is required")
	}
	if strings.TrimSpace(rawToken) == "" {
		return ErrInvalidCredentials
	}

	digest, err := security.DigestToken(rawToken)
	if err != nil {
		return ErrInvalidCredentials
	}

	session, err := s.sessions.GetSessionByTokenDigest(ctx, digest)
	if err != nil {
		if errors.Is(err, ErrSessionNotFound) {
			// Repeated logout is idempotent.
			return nil
		}
		return err
	}

	if session.RevokedAt != nil {
		return nil
	}

	return s.sessions.RevokeSession(ctx, session.ID, s.clock.Now())
}

func (s *LogoutService) LogoutAll(ctx context.Context, userID string) error {
	if ctx == nil {
		return errors.New("logout service: context is required")
	}
	if strings.TrimSpace(userID) == "" {
		return ErrInvalidUserID
	}

	return s.sessions.RevokeAllUserSessions(ctx, userID, s.clock.Now())
}

func sessionRevocationTime(clock Clock) time.Time {
	return clock.Now().UTC()

package identity

import (
	"context"
	"errors"
	"time"
)

var (
	ErrVerificationTokenNotFound = errors.New("verification token not found")
	ErrInvalidVerificationToken  = errors.New("invalid verification token")
	ErrVerificationTokenExpired  = errors.New("verification token expired")
	ErrVerificationTokenUsed     = errors.New("verification token already used")
)

type EmailVerificationToken struct {
	ID          string
	UserID      string
	TokenDigest string
	CreatedAt   time.Time
	ExpiresAt   time.Time
	ConsumedAt  *time.Time
}

func (t EmailVerificationToken) Validate() error {
	if t.ID == "" || t.UserID == "" {
		return errors.New("verification token ID and user ID are required")
	}
	if len(t.TokenDigest) != 64 {
		return errors.New("verification token digest must be 64 characters")
	}
	if t.CreatedAt.IsZero() || !t.ExpiresAt.After(t.CreatedAt) {
		return errors.New("verification token expiry must follow creation")
	}
	if t.ConsumedAt != nil && t.ConsumedAt.Before(t.CreatedAt) {
		return errors.New("verification token consumption precedes creation")
	}
	return nil
}

type VerificationTokenRepository interface {
	CreateVerificationToken(context.Context, EmailVerificationToken) error
	GetVerificationTokenByDigest(context.Context, string) (EmailVerificationToken, error)
	ConsumeVerificationToken(context.Context, string, time.Time) error
	InvalidateUserVerificationTokens(context.Context, string, time.Time) error
}

package identity

import (
	"context"
	"errors"
	"fmt"
	"time"

	"github.com/jackc/pgx/v5"
	"github.com/jackc/pgx/v5/pgxpool"
)

type PostgresVerificationTokenRepository struct {
	pool *pgxpool.Pool
}

var _ VerificationTokenRepository = (*PostgresVerificationTokenRepository)(nil)

func NewPostgresVerificationTokenRepository(
	pool *pgxpool.Pool,
) (*PostgresVerificationTokenRepository, error) {
	if pool == nil {
		return nil, errors.New("verification repository: pool is required")
	}
	return &PostgresVerificationTokenRepository{pool: pool}, nil
}

func (r *PostgresVerificationTokenRepository) CreateVerificationToken(
	ctx context.Context,
	token EmailVerificationToken,
) error {
	if ctx == nil {
		return errors.New("verification repository: context is required")
	}
	if err := token.Validate(); err != nil {
		return fmt.Errorf("validate verification token: %w", err)
	}

	_, err := r.pool.Exec(ctx, `
		INSERT INTO email_verification_tokens (
			id, user_id, token_digest, created_at, expires_at, consumed_at
		)
		VALUES ($1, $2, $3, $4, $5, $6)
	`,
		token.ID,
		token.UserID,
		token.TokenDigest,
		token.CreatedAt,
		token.ExpiresAt,
		token.ConsumedAt,
	)
	return err
}

func (r *PostgresVerificationTokenRepository) GetVerificationTokenByDigest(
	ctx context.Context,
	digest string,
) (EmailVerificationToken, error) {
	if ctx == nil {
		return EmailVerificationToken{}, errors.New("verification repository: context is required")
	}

	var token EmailVerificationToken
	err := r.pool.QueryRow(ctx, `
		SELECT id, user_id, token_digest, created_at, expires_at, consumed_at
		FROM email_verification_tokens
		WHERE token_digest = $1
	`, digest).Scan(
		&token.ID,
		&token.UserID,
		&token.TokenDigest,
		&token.CreatedAt,
		&token.ExpiresAt,
		&token.ConsumedAt,
	)
	if errors.Is(err, pgx.ErrNoRows) {
		return EmailVerificationToken{}, ErrVerificationTokenNotFound
	}
	return token, err
}

func (r *PostgresVerificationTokenRepository) ConsumeVerificationToken(
	ctx context.Context,
	digest string,
	consumedAt time.Time,
) error {
	if ctx == nil {
		return errors.New("verification repository: context is required")
	}

	tag, err := r.pool.Exec(ctx, `
		UPDATE email_verification_tokens
		SET consumed_at = $2
		WHERE token_digest = $1
		  AND consumed_at IS NULL
		  AND expires_at > $2
	`, digest, consumedAt)
	if err != nil {
		return err
	}
	if tag.RowsAffected() == 0 {
		return ErrInvalidVerificationToken
	}
	return nil
}

func (r *PostgresVerificationTokenRepository) InvalidateUserVerificationTokens(
	ctx context.Context,
	userID string,
	invalidatedAt time.Time,
) error {
	if ctx == nil {
		return errors.New("verification repository: context is required")
	}

	_, err := r.pool.Exec(ctx, `
		UPDATE email_verification_tokens
		SET consumed_at = $2
		WHERE user_id = $1
		  AND consumed_at IS NULL
	`, userID, invalidatedAt)
	return err
}
