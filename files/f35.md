import (
	"context"
	"errors"
	"fmt"
	"strings"

	"github.com/jackc/pgx/v5"
)

func (r *PostgresAssessmentAuditDeadLetterRepository) RequeueDeadLetter(
	ctx context.Context,
	tenantID string,
	eventID string,
) error {
	if r == nil || r.db == nil {
		return errors.New("dead-letter repository is not initialized")
	}

	tenantID = strings.TrimSpace(tenantID)
	eventID = strings.TrimSpace(eventID)
	if tenantID == "" {
		return ErrDeadLetterTenantRequired
	}
	if err := ValidateAuditEventID(eventID); err != nil {
		return err
	}

	tx, err := r.db.BeginTx(ctx, pgx.TxOptions{})
	if err != nil {
		return fmt.Errorf("begin dead-letter requeue transaction: %w", err)
	}
	defer func() { _ = tx.Rollback(ctx) }()

	const selectQuery = `
		SELECT event_id, tenant_id, actor_user_id, course_id,
		       assessment_id, action, occurred_at, created_at
		FROM assessment_audit_dead_letters
		WHERE tenant_id = $1 AND event_id = $2
		FOR UPDATE
	`

	var event AssessmentAuditOutboxEvent
	err = tx.QueryRow(ctx, selectQuery, tenantID, eventID).Scan(
		&event.EventID,
		&event.TenantID,
		&event.ActorUserID,
		&event.CourseID,
		&event.AssessmentID,
		&event.Action,
		&event.OccurredAt,
		&event.CreatedAt,
	)
	if errors.Is(err, pgx.ErrNoRows) {
		return ErrDeadLetterNotFound
	}
	if err != nil {
		return fmt.Errorf("read dead-letter event for requeue: %w", err)
	}

	const insertQuery = `
		INSERT INTO assessment_audit_outbox (
			event_id, tenant_id, actor_user_id, course_id,
			assessment_id, action, occurred_at, created_at,
			attempts, processed_at, claimed_until, claim_token, last_error
		)
		VALUES ($1, $2, $3, $4, $5, $6, $7, $8, 0, NULL, NULL, NULL, NULL)
		ON CONFLICT (event_id) DO NOTHING
	`

	tag, err := tx.Exec(
		ctx, insertQuery,
		event.EventID, event.TenantID, event.ActorUserID,
		event.CourseID, event.AssessmentID, event.Action,
		event.OccurredAt, event.CreatedAt,
	)
	if err != nil {
		return fmt.Errorf("insert requeued audit event: %w", err)
	}
	if tag.RowsAffected() != 1 {
		return ErrDeadLetterConflict
	}

	const deleteQuery := `
		DELETE FROM assessment_audit_dead_letters
		WHERE tenant_id = $1 AND event_id = $2
	`
	tag, err = tx.Exec(ctx, deleteQuery, tenantID, eventID)
	if err != nil {
		return fmt.Errorf("remove requeued dead-letter event: %w", err)
	}
	if tag.RowsAffected() != 1 {
		return ErrDeadLetterNotFound
	}

	if err := tx.Commit(ctx); err != nil {
		return fmt.Errorf("commit dead-letter requeue: %w", err)
	}
	return nil
}

package assessment

import "strings"

const maxDeadLetterErrorLength = 500

func RedactAssessmentAuditDeadLetterError(message string) string {
	message = strings.TrimSpace(message)
	if message == "" {
		return ""
	}

	// Do not expose multiline diagnostic dumps through administrative APIs.
	message = strings.ReplaceAll(message, "\r", " ")
	message = strings.ReplaceAll(message, "\n", " ")
	message = strings.Join(strings.Fields(message), " ")

	lower := strings.ToLower(message)
	sensitiveMarkers := []string{
		"password=",
		"token=",
		"authorization:",
		"bearer ",
		"secret=",
		"api_key=",
		"api-key=",
	}
	for _, marker := range sensitiveMarkers {
		if index := strings.Index(lower, marker); index >= 0 {
			message = message[:index] + "[redacted]"
			break
		}
	}

	if len(message) > maxDeadLetterErrorLength {
		message = message[:maxDeadLetterErrorLength]
	}
	return message
}

	"net/http"

	"time"

type deadLetterRateLimitEntry struct {
	windowStart time.Time
	count       int
	lastSeen    time.Time
}

type AssessmentAuditDeadLetterRateLimiter struct {
	mu         sync.Mutex
	entries    map[string]deadLetterRateLimitEntry
	limit      int
	window     time.Duration
	maxEntries int
	now        func() time.Time
}

func NewAssessmentAuditDeadLetterRateLimiter(
	limit int,
	window time.Duration,
	maxEntries int,
) *AssessmentAuditDeadLetterRateLimiter {
	if limit < 1 {
		limit = 30
	}
	if window <= 0 {
		window = time.Minute
	}
	if maxEntries < 100 {
		maxEntries = 1000
	}

	return &AssessmentAuditDeadLetterRateLimiter{
		entries:    make(map[string]deadLetterRateLimitEntry),
		limit:      limit,
		window:     window,
		maxEntries: maxEntries,
		now:        time.Now,
	}
}

func (l *AssessmentAuditDeadLetterRateLimiter) Allow(key string) bool {
	if l == nil {
		return false
	}
	key = strings.TrimSpace(key)
	if key == "" {
		return false
	}

	l.mu.Lock()
	defer l.mu.Unlock()

	now := l.now()
	entry, exists := l.entries[key]

	if !exists && len(l.entries) >= l.maxEntries {
		l.evictOldestLocked()
	}

	if !exists || now.Sub(entry.windowStart) >= l.window || now.Before(entry.windowStart) {
		l.entries[key] = deadLetterRateLimitEntry{
			windowStart: now,
			count:       1,
			lastSeen:    now,
		}
		return true
	}

	entry.lastSeen = now
	if entry.count >= l.limit {
		l.entries[key] = entry
		return false
	}
	entry.count++
	l.entries[key] = entry
	return true
}

func (l *AssessmentAuditDeadLetterRateLimiter) evictOldestLocked() {
	var oldestKey string
	var oldestTime time.Time

	for key, entry := range l.entries {
		if oldestKey == "" || entry.lastSeen.Before(oldestTime) {
			oldestKey = key
			oldestTime = entry.lastSeen
		}
	}
	if oldestKey != "" {
		delete(l.entries, oldestKey)
	}
}

func AssessmentAuditDeadLetterRateLimitKey(r *http.Request) string {
	if r == nil {
		return ""
	}
	// Prefer a trusted authenticated identity. Never trust forwarded IP headers
	// unless a separately configured trusted-proxy layer validates them.
	if principal, err := AssessmentAuditDeadLetterPrincipalFromContext(r.Context()); err == nil {
		return principal.TenantID + ":" + principal.ActorID
	}

	host, _, err := net.SplitHostPort(r.RemoteAddr)
	if err == nil {
		return host
	}
	return strings.TrimSpace(r.RemoteAddr)
}

func (l *AssessmentAuditDeadLetterRateLimiter) Middleware(
	next http.Handler,
) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if !l.Allow(AssessmentAuditDeadLetterRateLimitKey(r)) {
			w.Header().Set("Retry-After", "60")
			writeDeadLetterHTTPError(w, http.StatusTooManyRequests, "rate limit exceeded")
			return
		}
		next.ServeHTTP(w, r)
	})
}

		return nil, err
	}

		return nil, errors.New("request body too large")
	}
	return body, nil

import (
	"context"
	"errors"
	"time"
)

type AssessmentAuditOutboxBatchProcessor interface {
	ProcessBatch(ctx context.Context) (int, error)
}

type AssessmentAuditOutboxRunnerConfig struct {
	Interval     time.Duration
	InitialDelay time.Duration
}

type AssessmentAuditOutboxRunner struct {
	processor AssessmentAuditOutboxBatchProcessor
	config    AssessmentAuditOutboxRunnerConfig
}

func NewAssessmentAuditOutboxRunner(
	processor AssessmentAuditOutboxBatchProcessor,
	config AssessmentAuditOutboxRunnerConfig,
) (*AssessmentAuditOutboxRunner, error) {
	if processor == nil {
		return nil, errors.New("outbox batch processor is required")
	}
	if config.Interval <= 0 {
		return nil, errors.New("outbox runner interval must be positive")
	}
	if config.InitialDelay < 0 {
		return nil, errors.New("outbox initial delay cannot be negative")
	}
	return &AssessmentAuditOutboxRunner{
		processor: processor,
		config:    config,
	}, nil
}

func (r *AssessmentAuditOutboxRunner) Run(ctx context.Context) error {
	if r == nil || r.processor == nil {
		return errors.New("outbox runner is not initialized")
	}

	if r.config.InitialDelay > 0 {
		timer := time.NewTimer(r.config.InitialDelay)
		select {
		case <-ctx.Done():
			if !timer.Stop() {
				select {
				case <-timer.C:
				default:
				}
			}
			return ctx.Err()
		case <-timer.C:
		}
	}

	ticker := time.NewTicker(r.config.Interval)
	defer ticker.Stop()

	for {
		if _, err := r.processor.ProcessBatch(ctx); err != nil {
			if ctx.Err() != nil {
				return ctx.Err()
			}
			// A transient batch error must not permanently stop the worker.
			// The next scheduled tick retries pending work.
		}

		select {
		case <-ctx.Done():
			return ctx.Err()
		case <-ticker.C:
		}
	}
}

import (
	"context"
	"errors"
	"sync"
	"time"
)

var ErrAssessmentAuditOutboxUnhealthy = errors.New("assessment audit outbox worker is unhealthy")

type AssessmentAuditOutboxHealthSnapshot struct {
	StartedAt      time.Time
	LastAttemptAt  time.Time
	LastSuccessAt  time.Time
	LastErrorAt    time.Time
	LastError      string
	ConsecutiveErr int
	TotalBatches   uint64
	TotalProcessed uint64
	Ready          bool
}

type AssessmentAuditOutboxHealth struct {
	mu    sync.RWMutex
	state AssessmentAuditOutboxHealthSnapshot
	clock func() time.Time
}

func NewAssessmentAuditOutboxHealth() *AssessmentAuditOutboxHealth {
	return &AssessmentAuditOutboxHealth{clock: time.Now}
}

func (h *AssessmentAuditOutboxHealth) MarkStarted() {
	if h == nil {
		return
	}
	h.mu.Lock()
	defer h.mu.Unlock()

	now := h.clock().UTC()
	if h.state.StartedAt.IsZero() {
		h.state.StartedAt = now
	}
	h.state.Ready = true
}

func (h *AssessmentAuditOutboxHealth) RecordBatch(processed int, err error) {
	if h == nil {
		return
	}
	h.mu.Lock()
	defer h.mu.Unlock()

	now := h.clock().UTC()
	h.state.LastAttemptAt = now
	h.state.TotalBatches++
	if processed > 0 {
		h.state.TotalProcessed += uint64(processed)
	}

	if err != nil {
		h.state.LastErrorAt = now
		h.state.LastError = err.Error()
		h.state.ConsecutiveErr++
		return
	}

	h.state.LastSuccessAt = now
	h.state.LastError = ""
	h.state.ConsecutiveErr = 0
	h.state.Ready = true
}

func (h *AssessmentAuditOutboxHealth) MarkStopped() {
	if h == nil {
		return
	}
	h.mu.Lock()
	h.state.Ready = false
	h.mu.Unlock()
}

func (h *AssessmentAuditOutboxHealth) Snapshot() AssessmentAuditOutboxHealthSnapshot {
	if h == nil {
		return AssessmentAuditOutboxHealthSnapshot{}
	}
	h.mu.RLock()
	defer h.mu.RUnlock()
	return h.state
}

func (h *AssessmentAuditOutboxHealth) Check(ctx context.Context) error {
	if err := ctx.Err(); err != nil {
		return err
	}
	snapshot := h.Snapshot()
	if !snapshot.Ready {
		return ErrAssessmentAuditOutboxUnhealthy
	}
	return nil
}

func (h *AssessmentAuditOutboxHealthHandler) ServeHTTP(
	w http.ResponseWriter,
	r *http.Request,
) {
	if r.Method != http.MethodGet {
		w.Header().Set("Allow", http.MethodGet)
		http.Error(w, "method not allowed", http.StatusMethodNotAllowed)
		return
	}
	if h == nil || h.health == nil {
		http.Error(w, "health unavailable", http.StatusServiceUnavailable)
		return
	}

	snapshot := h.health.Snapshot()
	w.Header().Set("Content-Type", "application/json; charset=utf-8")
	w.Header().Set("Cache-Control", "no-store")
	w.Header().Set("X-Content-Type-Options", "nosniff")

	status := http.StatusOK
	if !snapshot.Ready {
		status = http.StatusServiceUnavailable
	}
	w.WriteHeader(status)
	_ = json.NewEncoder(w).Encode(snapshot)
}

package assessment

import (
	"context"
	"errors"
	"time"
)

func ShutdownAssessmentAuditOutbox(
	cancel context.CancelFunc,
	done <-chan error,
	timeout time.Duration,
) error {
	if cancel == nil || done == nil || timeout <= 0 {
		return errors.New("invalid outbox shutdown configuration")
	}

	cancel()

	timer := time.NewTimer(timeout)
	defer timer.Stop()

	select {
	case err, ok := <-done:
		if !ok {
			return nil
		}
		if errors.Is(err, context.Canceled) {
			return nil
		}
		return err
	case <-timer.C:
		return errors.New("assessment audit outbox worker shutdown timed out")
	}
}
