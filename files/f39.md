import (
	"net/http"

func AssessmentAuditQueryLimitsMiddleware(
	next http.Handler,
	limits AssessmentAuditQueryLimits,
) http.Handler {
	if next == nil {
		next = http.NotFoundHandler()
	}
	if err := limits.Validate(); err != nil {
		return http.HandlerFunc(func(w http.ResponseWriter, _ *http.Request) {
			http.Error(w, "service configuration error", http.StatusInternalServerError)
		})
	}

	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		query := r.URL.Query()

		for key, values := range query {
			if len(key) > limits.MaxFilterBytes {
				http.Error(w, "query parameter name too long", http.StatusBadRequest)
				return
			}
			for _, value := range values {
				if err := limits.ValidateFilterText(value); err != nil {
					http.Error(w, "invalid query parameter", http.StatusBadRequest)
					return
				}
			}
		}

		next.ServeHTTP(w, r)
	})
}

import (
	"encoding/base64"
	"errors"
	"strings"

var ErrAssessmentAuditCursorInvalid = errors.New("assessment audit cursor is invalid")

type AssessmentAuditCursor struct {
	CreatedAt string
	EventID   string
}

func EncodeAssessmentAuditCursor(cursor AssessmentAuditCursor) (string, error) {
	if err := cursor.Validate(); err != nil {
		return "", err
	}

	payload := cursor.CreatedAt + "\n" + cursor.EventID
	return base64.RawURLEncoding.EncodeToString([]byte(payload)), nil
}

func DecodeAssessmentAuditCursor(value string) (AssessmentAuditCursor, error) {
	if value == "" || len(value) > 1024 {
		return AssessmentAuditCursor{}, ErrAssessmentAuditCursorInvalid
	}

	decoded, err := base64.RawURLEncoding.DecodeString(value)
	if err != nil {
		return AssessmentAuditCursor{}, ErrAssessmentAuditCursorInvalid
	}

	parts := strings.Split(string(decoded), "\n")
	if len(parts) != 2 {
		return AssessmentAuditCursor{}, ErrAssessmentAuditCursorInvalid
	}

	cursor := AssessmentAuditCursor{
		CreatedAt: parts[0],
		EventID:   parts[1],
	}
	if err := cursor.Validate(); err != nil {
		return AssessmentAuditCursor{}, err
	}
	return cursor, nil
}

func (c AssessmentAuditCursor) Validate() error {
	if strings.TrimSpace(c.CreatedAt) == "" || strings.TrimSpace(c.EventID) == "" {
		return ErrAssessmentAuditCursorInvalid
	}
	if len(c.CreatedAt) > 64 || len(c.EventID) > 128 {
		return ErrAssessmentAuditCursorInvalid
	}
	if strings.ContainsAny(c.CreatedAt+c.EventID, "\r\n\x00") {
		return ErrAssessmentAuditCursorInvalid
	}
	return nil
}

package assessment

import (
	"crypto/hmac"
	"crypto/sha256"
	"crypto/subtle"
	"encoding/base64"
	"errors"
)

var ErrAssessmentAuditCursorSignatureInvalid = errors.New("assessment audit cursor signature is invalid")

func SignAssessmentAuditCursor(payload []byte, secret []byte) (string, error) {
	if len(payload) == 0 || len(payload) > 768 || len(secret) < 32 {
		return "", ErrAssessmentAuditCursorSignatureInvalid
	}

	mac := hmac.New(sha256.New, secret)
	_, _ = mac.Write(payload)
	signature := mac.Sum(nil)

	return base64.RawURLEncoding.EncodeToString(payload) + "." +
		base64.RawURLEncoding.EncodeToString(signature), nil
}

func VerifyAssessmentAuditCursor(token string, secret []byte) ([]byte, error) {
	if len(secret) < 32 || len(token) == 0 || len(token) > 2048 {
		return nil, ErrAssessmentAuditCursorSignatureInvalid
	}

	var payloadPart, signaturePart string
	dot := -1
	for i := 0; i < len(token); i++ {
		if token[i] == '.' {
			if dot >= 0 {
				return nil, ErrAssessmentAuditCursorSignatureInvalid
			}
			dot = i
		}
	}
	if dot <= 0 || dot == len(token)-1 {
		return nil, ErrAssessmentAuditCursorSignatureInvalid
	}

	payloadPart, signaturePart = token[:dot], token[dot+1:]
	payload, err := base64.RawURLEncoding.DecodeString(payloadPart)
	if err != nil || len(payload) == 0 || len(payload) > 768 {
		return nil, ErrAssessmentAuditCursorSignatureInvalid
	}
	signature, err := base64.RawURLEncoding.DecodeString(signaturePart)
	if err != nil {
		return nil, ErrAssessmentAuditCursorSignatureInvalid
	}

	mac := hmac.New(sha256.New, secret)
	_, _ = mac.Write(payload)
	expected := mac.Sum(nil)

	if len(signature) != len(expected) ||
		subtle.ConstantTimeCompare(signature, expected) != 1 {
		return nil, ErrAssessmentAuditCursorSignatureInvalid
	}
	return payload, nil
}

const AssessmentAuditCursorKeyRingEnv = "ASSESSMENT_AUDIT_CURSOR_KEYRING"
const AssessmentAuditCursorActiveKeyEnv = "ASSESSMENT_AUDIT_CURSOR_ACTIVE_KEY"

var ErrAssessmentAuditCursorKeyRingConfig = errors.New("assessment audit cursor key ring configuration is invalid")

// Key-ring format: key-id=base64-key,key-id=base64-key.
// Keys must contain at least 32 bytes after decoding.
func LoadAssessmentAuditCursorKeyRingFromEnv() (AssessmentAuditCursorKeyRing, error) {
	raw := strings.TrimSpace(os.Getenv(AssessmentAuditCursorKeyRingEnv))
	activeID := strings.TrimSpace(os.Getenv(AssessmentAuditCursorActiveKeyEnv))
	if raw == "" || activeID == "" {
		return AssessmentAuditCursorKeyRing{}, ErrAssessmentAuditCursorKeyRingConfig
	}

	keys := make(map[string][]byte)
	for _, item := range strings.Split(raw, ",") {
		keyID, encoded, ok := strings.Cut(strings.TrimSpace(item), "=")
		if !ok || keyID == "" || encoded == "" {
			return AssessmentAuditCursorKeyRing{}, ErrAssessmentAuditCursorKeyRingConfig
		}
		if _, exists := keys[keyID]; exists {
			return AssessmentAuditCursorKeyRing{}, ErrAssessmentAuditCursorKeyRingConfig
		}

		key, err := base64.StdEncoding.DecodeString(encoded)
		if err != nil {
			key, err = base64.RawStdEncoding.DecodeString(encoded)
		}
		if err != nil || len(key) < 32 {
			return AssessmentAuditCursorKeyRing{}, ErrAssessmentAuditCursorKeyRingConfig
		}
		keys[keyID] = key
	}

	ring := AssessmentAuditCursorKeyRing{
		ActiveKeyID: activeID,
		Keys:        keys,
	}
	if err := ring.Validate(); err != nil {
		return AssessmentAuditCursorKeyRing{}, ErrAssessmentAuditCursorKeyRingConfig
	}
	return ring, nil
}

		return nil, ErrAssessmentAuditCursorProviderServiceUnavailable
	}

		return nil, ErrAssessmentAuditCursorProviderServiceUnavailable
	}

		return nil, ErrAssessmentAuditCursorProviderServiceUnavailable
	}
	if now == nil {
		now = time.Now
	}

		Keys:         keys,
		ExpiryPolicy: expiry,
		Now:          now,
	}, nil
}

func (s *AssessmentAuditCursorProviderService) Encode(cursor AssessmentAuditScopedCursor) (string, error) {
	if s == nil || s.Keys == nil || s.Now == nil {
		return "", ErrAssessmentAuditCursorProviderServiceUnavailable
	}
	if err := cursor.Validate(); err != nil {
		return "", err
	}
	if err := s.ExpiryPolicy.Check(cursor.CreatedAt, s.Now()); err != nil {
		return "", err
	}

	ring, err := s.Keys.Snapshot()
	if err != nil {
		return "", ErrAssessmentAuditCursorProviderServiceUnavailable
	}

	tokenService, err := NewAssessmentAuditCursorTokenService(ring)
	if err != nil {
		return "", ErrAssessmentAuditCursorProviderServiceUnavailable
	}

	payload, err := EncodeAssessmentAuditScopedCursor(cursor)
	if err != nil {
		return "", err

	token, tenantID, audience string,

	if err != nil {
		return AssessmentAuditScopedCursor{}, ErrAssessmentAuditCursorProviderServiceUnavailable

	if err != nil {
		return AssessmentAuditScopedCursor{}, ErrAssessmentAuditCursorProviderServiceUnavailable
	}

	if err != nil {
		return AssessmentAuditScopedCursor{}, err
