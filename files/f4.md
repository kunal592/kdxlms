## File 4 — `internal/config/config_test.go`

Purpose: Automated tests for configuration loading, environment-variable parsing, production security requirements, and invalid settings.

Production configuration must be tested before it is used to start the application. These tests isolate environment variables so results don't depend on your machine's existing configuration.

```

package config

import (
	"os"
	"strings"
	"testing"
	"time"
)

func clearConfigEnv(t *testing.T) {
	t.Helper()

	keys := []string{
		"APP_NAME",
		"APP_ENV",
		"APP_VERSION",
		"HTTP_ADDR",
		"HTTP_READ_TIMEOUT",
		"HTTP_READ_HEADER_TIMEOUT",
		"HTTP_WRITE_TIMEOUT",
		"HTTP_IDLE_TIMEOUT",
		"HTTP_SHUTDOWN_TIMEOUT",
		"DATABASE_URL",
		"DATABASE_MAX_OPEN_CONNS",
		"DATABASE_MAX_IDLE_CONNS",
		"DATABASE_CONN_MAX_LIFETIME",
		"DATABASE_CONN_MAX_IDLE_TIME",
		"DATABASE_CONNECT_TIMEOUT",
		"REDIS_ADDR",
		"REDIS_USERNAME",
		"REDIS_PASSWORD",
		"REDIS_DB",
		"REDIS_TLS_ENABLED",
		"REDIS_POOL_SIZE",
		"LOG_LEVEL",
		"APP_SECRET",
		"TRUSTED_PROXIES",
		"CORS_ALLOWED_ORIGINS",
	}

	for _, key := range keys {
		if err := os.Unsetenv(key); err != nil {
			t.Fatalf("unset %s: %v", key, err)
		}
	}
}

func TestLoadDefaults(t *testing.T) {
	clearConfigEnv(t)

	cfg, err := Load()
	if err != nil {
		t.Fatalf("Load() returned unexpected error: %v", err)
	}

	if cfg.App.Name != "KDXLabsLLM" {
		t.Errorf("App.Name = %q, want %q", cfg.App.Name, "KDXLabsLLM")
	}

	if cfg.App.Environment != "development" {
		t.Errorf(
			"App.Environment = %q, want %q",
			cfg.App.Environment,
			"development",
		)
	}

	if cfg.HTTP.Address != ":8080" {
		t.Errorf("HTTP.Address = %q, want %q", cfg.HTTP.Address, ":8080")
	}

	if cfg.Database.MaxOpenConns != 25 {
		t.Errorf(
			"Database.MaxOpenConns = %d, want 25",
			cfg.Database.MaxOpenConns,
		)
	}

	if cfg.Redis.Address != "localhost:6379" {
		t.Errorf(
			"Redis.Address = %q, want %q",
			cfg.Redis.Address,
			"localhost:6379",
		)
	}
}

func TestLoadEnvironmentOverrides(t *testing.T) {
	clearConfigEnv(t)

	t.Setenv("APP_NAME", "KDXLabsLLM-Test")
	t.Setenv("APP_ENV", "test")
	t.Setenv("HTTP_ADDR", ":9090")
	t.Setenv("HTTP_READ_TIMEOUT", "20s")
	t.Setenv("DATABASE_MAX_OPEN_CONNS", "40")
	t.Setenv("REDIS_POOL_SIZE", "30")
	t.Setenv("LOG_LEVEL", "debug")
	t.Setenv("CORS_ALLOWED_ORIGINS", "https://example.com, https://app.example.com")

	cfg, err := Load()
	if err != nil {
		t.Fatalf("Load() returned unexpected error: %v", err)
	}

	if cfg.App.Name != "KDXLabsLLM-Test" {
		t.Errorf("unexpected app name: %q", cfg.App.Name)
	}

	if cfg.HTTP.Address != ":9090" {
		t.Errorf("unexpected HTTP address: %q", cfg.HTTP.Address)
	}

	if cfg.HTTP.ReadTimeout != 20*time.Second {
		t.Errorf("unexpected read timeout: %v", cfg.HTTP.ReadTimeout)
	}

	if cfg.Database.MaxOpenConns != 40 {
		t.Errorf("unexpected database pool size: %d", cfg.Database.MaxOpenConns)
	}

	if cfg.Redis.PoolSize != 30 {
		t.Errorf("unexpected Redis pool size: %d", cfg.Redis.PoolSize)
	}

	if len(cfg.Security.AllowedOrigins) != 2 {
		t.Fatalf(
			"allowed origins count = %d, want 2",
			len(cfg.Security.AllowedOrigins),
		)
	}

	if cfg.Security.AllowedOrigins[1] != "https://app.example.com" {
		t.Errorf(
			"unexpected second origin: %q",
			cfg.Security.AllowedOrigins[1],
		)
	}
}

func TestProductionRejectsWildcardCORS(t *testing.T) {
	clearConfigEnv(t)

	t.Setenv("APP_ENV", "production")
	t.Setenv("APP_SECRET", strings.Repeat("a", 32))
	t.Setenv("DATABASE_URL", "postgres://user:password@localhost:5432/kdxlabsllm")
	t.Setenv("REDIS_TLS_ENABLED", "true")
	t.Setenv("CORS_ALLOWED_ORIGINS", "*")

	_, err := Load()
	if err == nil {
		t.Fatal("Load() accepted wildcard production CORS")
	}

	if !strings.Contains(err.Error(), "wildcard CORS") {
		t.Fatalf("unexpected validation error: %v", err)
	}
}

func TestValidateRejectsInvalidDatabasePool(t *testing.T) {
	clearConfigEnv(t)

	cfg, err := Load()
	if err != nil {
		t.Fatalf("Load() returned unexpected error: %v", err)
	}

	cfg.Database.MaxIdleConns = cfg.Database.MaxOpenConns + 1

	if err := cfg.Validate(); err == nil {
		t.Fatal("Validate() accepted idle connections above the pool limit")
	}
}

func TestValidateRejectsNegativeRedisDatabase(t *testing.T) {
	clearConfigEnv(t)

	cfg, err := Load()
	if err != nil {
		t.Fatalf("Load() returned unexpected error: %v", err)
	}

	cfg.Redis.DB = -1

	if err := cfg.Validate(); err == nil {
		t.Fatal("Validate() accepted a negative Redis database index")
	}
}
```
