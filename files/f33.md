type AssessmentLister interface {
	List(ctx context.Context, tenantID, courseID string, limit, offset int) ([]Assessment, error)
}

type AssessmentListService struct {
	repository Repository
}

func NewAssessmentListService(repository Repository) (*AssessmentListService, error) {
	if repository == nil {
		return nil, errors.New("assessment repository is required")
	}
	return &AssessmentListService{repository: repository}, nil
}

func (s *AssessmentListService) List(
	ctx context.Context,
	tenantID, courseID string,
	limit, offset int,
) ([]Assessment, error) {
	tenantID = strings.TrimSpace(tenantID)
	courseID = strings.TrimSpace(courseID)

	if tenantID == "" || courseID == "" {
		return nil, ErrInvalidAssessment
	}
	if limit < 1 || limit > 100 || offset < 0 {
		return nil, ErrInvalidPagination
	}

	items, err := s.repository.List(ctx, tenantID, courseID, limit, offset)
	if err != nil {
		return nil, err
	}

	visible := make([]Assessment, 0, len(items))
	for _, item := range items {
		if item.Status == StatusPublished {
			visible = append(visible, item)
		}
	}
	return visible, nil
}

		writeAssessmentError(w, http.StatusMethodNotAllowed, "method not allowed")
		return
	}

	tenantID := strings.TrimSpace(r.Header.Get("X-Tenant-ID"))

		writeAssessmentError(w, http.StatusBadRequest, "missing request context")
		return
	}

	limit := 20
	offset := 0

	if raw := r.URL.Query().Get("limit"); raw != "" {
		value, err := strconv.Atoi(raw)
		if err != nil || value < 1 || value > 100 {
			writeAssessmentError(w, http.StatusBadRequest, "limit must be between 1 and 100")
			return
		}
		limit = value
	}
	if raw := r.URL.Query().Get("offset"); raw != "" {
		value, err := strconv.Atoi(raw)

			return
		}
		offset = value
	}

// List returns learner-visible questions only for a published assessment.
// QuestionResponse deliberately omits answer keys and explanations.
func (s *PublishedQuestionService) List(
	ctx context.Context,
	tenantID, courseID, assessmentID string,
	limit, offset int,
) ([]QuestionResponse, error) {
	if strings.TrimSpace(tenantID) == "" ||
		strings.TrimSpace(courseID) == "" ||
		strings.TrimSpace(assessmentID) == "" {
		return nil, ErrInvalidAssessment
	}
	if limit < 1 || limit > 100 || offset < 0 {
		return nil, ErrInvalidPagination
	}

	assessment, err := s.assessments.GetByID(
		ctx, tenantID, courseID, assessmentID,
	)
	if err != nil {
		return nil, err
	}
	if assessment.Status != StatusPublished {
		return nil, ErrAssessmentUnavailable
	}

	questions, err := s.questions.ListQuestions(
		ctx, tenantID, courseID, assessmentID, limit, offset,
	)
	if err != nil {
		return nil, err
	}

	result := make([]QuestionResponse, 0, len(questions))
	for _, question := range questions {
		result = append(result, NewQuestionResponse(question))
	}
	return result, nil
}

var ErrAttemptOwnershipMismatch = errors.New("attempt ownership mismatch")

type AttemptOwnershipChecker struct {
	attempts AttemptRepository
}

func NewAttemptOwnershipChecker(
	attempts AttemptRepository,
) (*AttemptOwnershipChecker, error) {
	if attempts == nil {
		return nil, errors.New("attempt repository is required")
	}
	return &AttemptOwnershipChecker{attempts: attempts}, nil
}

func (c *AttemptOwnershipChecker) GetOwnedAttempt(
	ctx context.Context,
	tenantID, userID, attemptID string,
) (Attempt, error) {
	tenantID = strings.TrimSpace(tenantID)
	userID = strings.TrimSpace(userID)
	attemptID = strings.TrimSpace(attemptID)

	if tenantID == "" || userID == "" || attemptID == "" {
		return Attempt{}, ErrAttemptOwnershipMismatch
	}

	attempt, err := c.attempts.GetByID(ctx, tenantID, attemptID)
	if err != nil {
		return Attempt{}, err
	}
	if attempt.UserID != userID {
		// Do not return the attempt to callers when ownership does not match.
		return Attempt{}, ErrAttemptOwnershipMismatch
	}

	return attempt, nil
}

		errors.Is(err, ErrAssessmentTimedOut),

	}

	TotalQuestions int

func CalculateScore(correct, total int, passingScore float64) (ScoreResult, error) {
	if total <= 0 || correct < 0 || correct > total ||
		math.IsNaN(passingScore) || math.IsInf(passingScore, 0) ||
		passingScore < 0 || passingScore > 100 {

	}

	percentage := math.Round((float64(correct)/float64(total))*10000) / 100

	return ScoreResult{
		CorrectAnswers: correct,
		TotalQuestions: total,
		Percentage:     percentage,
		Passed:         percentage >= passingScore,
	}, nil
}

import (
	"crypto/rand"
	"errors"
	"math/big"

var ErrQuestionShuffleFailed = errors.New("question shuffle failed")

// ShuffleQuestionOrder uses crypto/rand so the permutation is not based on a
// predictable pseudo-random seed. The input slice is copied, not mutated.
func ShuffleQuestionOrder(questions []Question) ([]Question, error) {
	result := append([]Question(nil), questions...)

	for i := len(result) - 1; i > 0; i-- {
		n, err := rand.Int(rand.Reader, big.NewInt(int64(i+1)))
		if err != nil {
			return nil, ErrQuestionShuffleFailed
		}

		j := int(n.Int64())
		result[i], result[j] = result[j], result[i]
	}

	return result, nil
}

import (
	"context"
	"errors"
	"fmt"
	"strings"
	"time"
)

type AssessmentPublisherRepository interface {
	GetByID(ctx context.Context, tenantID, courseID, assessmentID string) (Assessment, error)
	Update(ctx context.Context, assessment Assessment) error
}

type AssessmentPublishClock interface {
	Now() time.Time
}

type AssessmentPublishService struct {
	assessments AssessmentPublisherRepository
	questions   QuestionRepository
	clock       AssessmentPublishClock
}

func NewAssessmentPublishService(
	assessments AssessmentPublisherRepository,
	questions QuestionRepository,
	clock AssessmentPublishClock,
) (*AssessmentPublishService, error) {
	if assessments == nil || questions == nil || clock == nil {
		return nil, errors.New("publish service dependencies are required")
	}
	return &AssessmentPublishService{
		assessments: assessments,
		questions:   questions,
		clock:       clock,
	}, nil
}

func (s *AssessmentPublishService) Publish(
	ctx context.Context,
	tenantID, courseID, assessmentID string,
) (Assessment, error) {
	if strings.TrimSpace(tenantID) == "" ||
		strings.TrimSpace(courseID) == "" ||
		strings.TrimSpace(assessmentID) == "" {
		return Assessment{}, ErrInvalidAssessment
	}

	item, err := s.assessments.GetByID(ctx, tenantID, courseID, assessmentID)
	if err != nil {
		return Assessment{}, err
	}
	if item.Status == StatusPublished {
		return item, nil
	}

	questions, err := s.questions.ListQuestions(
		ctx, tenantID, courseID, assessmentID, MaxQuestionsPerAssessment+1, 0,
	)
	if err != nil {
		return Assessment{}, err
	}
	if err := ValidateQuestionCount(len(questions)); err != nil {
		return Assessment{}, err
	}
	if err := ValidateAssessmentForPublishing(item, questions); err != nil {
		return Assessment{}, err
	}
	if err := ValidateAttemptLimits(item.AttemptLimit, item.TimeLimitSeconds); err != nil {
		return Assessment{}, fmt.Errorf("%w: %v", ErrAssessmentCannotPublish, err)
	}

	now := s.clock.Now().UTC()
	item.Status = StatusPublished
	item.PublishedAt = &now
	item.UpdatedAt = now

	if err := s.assessments.Update(ctx, item); err != nil {
		return Assessment{}, err
	}
	return item, nil
}

	"errors"
	"net/http"
	"strings"

type AssessmentPublisher interface {
	Publish(
		ctx context.Context,
		tenantID, courseID, assessmentID string,
	) (Assessment, error)
}

type AssessmentPublishHandler struct {
	service AssessmentPublisher
}

func NewAssessmentPublishHandler(service AssessmentPublisher) *AssessmentPublishHandler {
	return &AssessmentPublishHandler{service: service}
}

func (h *AssessmentPublishHandler) Publish(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodPost {
		writeAssessmentError(w, http.StatusMethodNotAllowed, "method not allowed")
		return
	}

	tenantID := strings.TrimSpace(r.Header.Get("X-Tenant-ID"))
	courseID := strings.TrimSpace(r.PathValue("courseID"))
	assessmentID := strings.TrimSpace(r.PathValue("assessmentID"))

	if tenantID == "" || courseID == "" || assessmentID == "" {
		writeAssessmentError(w, http.StatusBadRequest, "missing request context")
		return
	}

	item, err := h.service.Publish(r.Context(), tenantID, courseID, assessmentID)
	if err != nil {
		switch {
		case errors.Is(err, ErrNotFound):
			writeAssessmentError(w, http.StatusNotFound, "assessment not found")
		case errors.Is(err, ErrAssessmentCannotPublish),
			errors.Is(err, ErrInvalidAttemptLimits):
			writeAssessmentError(w, http.StatusUnprocessableEntity, "assessment does not meet publishing requirements")
		default:
			writeAssessmentError(w, http.StatusInternalServerError, "could not publish assessment")
		}
		return
	}

	w.Header().Set("Content-Type", "application/json")
	_ = json.NewEncoder(w).Encode(NewAssessmentResponse(item))
}

	"context"
	"errors"
	"time"

type AuditClock interface {
	Now() time.Time
}

type AuditedAssessmentPublisher struct {
	publisher AssessmentPublisher
	audit     AssessmentAuditWriter
	clock     AuditClock
}

func NewAuditedAssessmentPublisher(
	publisher AssessmentPublisher,
	audit AssessmentAuditWriter,
	clock AuditClock,
) (*AuditedAssessmentPublisher, error) {
	if publisher == nil || audit == nil || clock == nil {
		return nil, errors.New("publisher, audit writer and clock are required")
	}
	return &AuditedAssessmentPublisher{
		publisher: publisher,
		audit:     audit,
		clock:     clock,
	}, nil
}

// PublishForActor publishes an assessment and records the actor and action.
// The caller must authorize the actor before invoking this method.
func (p *AuditedAssessmentPublisher) PublishForActor(
	ctx context.Context,
	tenantID, actorUserID, courseID, assessmentID string,
) (Assessment, error) {
	item, err := p.publisher.Publish(ctx, tenantID, courseID, assessmentID)
	if err != nil {
		return Assessment{}, err
	}

	err = WriteAssessmentAudit(ctx, p.audit, AssessmentAuditEvent{
		TenantID:     tenantID,
		ActorUserID:  actorUserID,
		CourseID:     courseID,
		AssessmentID: assessmentID,
		Action:       "assessment.published",
		OccurredAt:   p.clock.Now(),
	})
	if err != nil {
		return Assessment{}, errors.Join(
			errors.New("assessment published but audit write failed"),
			err,
		)
	}

	return item, nil
}

package assessment

import (
	"context"
	"errors"
	"fmt"
	"strings"

	"github.com/jackc/pgx/v5/pgxpool"
)

type PostgresAssessmentAuditWriter struct {
	db *pgxpool.Pool
}

func NewPostgresAssessmentAuditWriter(
	db *pgxpool.Pool,
) (*PostgresAssessmentAuditWriter, error) {
	if db == nil {
		return nil, errors.New("database pool is required")
	}
	return &PostgresAssessmentAuditWriter{db: db}, nil
}

func (w *PostgresAssessmentAuditWriter) WriteAssessmentAudit(
	ctx context.Context,
	event AssessmentAuditEvent,
) error {
	if w == nil || w.db == nil {
		return errors.New("assessment audit writer is not initialized")
	}

	event.TenantID = strings.TrimSpace(event.TenantID)
	event.ActorUserID = strings.TrimSpace(event.ActorUserID)
	event.CourseID = strings.TrimSpace(event.CourseID)
	event.AssessmentID = strings.TrimSpace(event.AssessmentID)
	event.Action = strings.TrimSpace(event.Action)

	if event.TenantID == "" || event.ActorUserID == "" ||
		event.CourseID == "" || event.AssessmentID == "" ||
		event.Action == "" || event.OccurredAt.IsZero() {
		return errors.New("invalid assessment audit event")
	}

	const query = `
		INSERT INTO assessment_audit_events (
			tenant_id,
			actor_user_id,
			course_id,
			assessment_id,
			action,
			occurred_at
		)
		VALUES ($1, $2, $3, $4, $5, $6)
	`

	_, err := w.db.Exec(
		ctx,
		query,
		event.TenantID,
		event.ActorUserID,
		event.CourseID,
		event.AssessmentID,
		event.Action,
		event.OccurredAt.UTC(),
	)
	if err != nil {
		return fmt.Errorf("insert assessment audit event: %w", err)
	}

	return nil
}
