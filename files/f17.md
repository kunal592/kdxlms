## File 17 — `internal/platform/httpx/cors.go`

This middleware controls which browser origins can call the KDXLabsLLM API. It uses the existing `SecurityConfig.CORSOrigins` configuration and supports preflight requests, credentialed requests, and explicit origin matching.

```

package httpx

import (
	"net/http"
	"net/url"
	"strings"

	"github.com/gin-gonic/gin"

	"github.com/kdxlabs/kdxlabsllm/internal/config"
)

const (
	corsAllowMethods = "GET, POST, PUT, PATCH, DELETE, OPTIONS"
	corsAllowHeaders = "Authorization, Content-Type, X-Request-ID, X-CSRF-Token"
	corsMaxAge       = "600"
)

// CORS returns middleware that enforces the configured browser origins.
//
// Origins must be explicit origins, such as https://app.example.com,
// without paths, query strings, or fragments. Wildcard origins are supported
// only for non-credentialed access. Production configuration should use
// explicit origins.
func CORS(cfg config.SecurityConfig) gin.HandlerFunc {
	allowedOrigins := make(map[string]struct{}, len(cfg.CORSOrigins))
	allowWildcard := false

	for _, rawOrigin := range cfg.CORSOrigins {
		origin := strings.TrimSpace(rawOrigin)
		if origin == "" {
			continue
		}

		if origin == "*" {
			allowWildcard = true
			continue
		}

		if normalized, ok := normalizeOrigin(origin); ok {
			allowedOrigins[normalized] = struct{}{}
		}
	}

	return func(c *gin.Context) {
		requestOrigin := strings.TrimSpace(c.GetHeader("Origin"))

		// Non-browser requests may omit Origin. CORS is a browser policy,
		// not an authentication or authorization mechanism.
		if requestOrigin == "" {
			c.Next()
			return
		}

		origin, validOrigin := normalizeOrigin(requestOrigin)
		if !validOrigin {
			rejectCORS(c)
			return
		}

		_, explicitlyAllowed := allowedOrigins[origin]
		if !explicitlyAllowed && !allowWildcard {
			rejectCORS(c)
			return
		}

		if allowWildcard && !explicitlyAllowed {
			c.Header("Access-Control-Allow-Origin", "*")
		} else {
			c.Header("Access-Control-Allow-Origin", origin)
			c.Header("Vary", "Origin")
		}

		c.Header("Access-Control-Allow-Methods", corsAllowMethods)
		c.Header("Access-Control-Allow-Headers", corsAllowHeaders)
		c.Header("Access-Control-Max-Age", corsMaxAge)

		// Do not enable credentialed cross-origin requests by default.
		// Cookie-based authentication requires an explicit CSRF strategy.
		if c.Request.Method == http.MethodOptions {
			c.AbortWithStatus(http.StatusNoContent)
			return
		}

		c.Next()
	}
}

func rejectCORS(c *gin.Context) {
	c.AbortWithStatusJSON(http.StatusForbidden, gin.H{
		"error": gin.H{
			"code":    "ORIGIN_NOT_ALLOWED",
			"message": "This request origin is not allowed.",
		},
	})
}

func normalizeOrigin(raw string) (string, bool) {
	parsed, err := url.Parse(raw)
	if err != nil {
		return "", false
	}

	if parsed.Scheme != "http" && parsed.Scheme != "https" {
		return "", false
	}

	if parsed.Host == "" ||
		parsed.User != nil ||
		parsed.RawQuery != "" ||
		parsed.Fragment != "" ||
		parsed.Path != "" ||
		parsed.Opaque != "" {
		return "", false
	}

	// Reject values that url.Parse may accept but that are unsuitable as
	// serialized browser origins.
	if strings.ContainsAny(raw, " \t\r\n") {
		return "", false
	}

	return parsed.Scheme + "://" + parsed.Host, true
}
```
