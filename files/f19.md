## File 19 — `internal/platform/httpx/access_log.go`

This middleware provides structured HTTP access logs using Zap. Each request log includes the request ID, method, path, response status, latency, client IP, and response size.

Sensitive query parameters and request bodies are deliberately excluded from logs.

```

package httpx

import (
	"time"

	"github.com/gin-gonic/gin"
	"go.uber.org/zap"
)

// AccessLog returns middleware that records one structured log per request.
//
// Register RequestID before AccessLog so each log can include the request ID.
// Avoid logging request bodies, authorization headers, cookies, or query
// strings because they may contain credentials or personal information.
func AccessLog(logger *zap.Logger) gin.HandlerFunc {
	if logger == nil {
		logger = zap.NewNop()
	}

	return func(c *gin.Context) {
		startedAt := time.Now()

		// Capture only the URL path, never the query string.
		path := c.Request.URL.Path
		if path == "" {
			path = "/"
		}

		method := c.Request.Method

		c.Next()

		latency := time.Since(startedAt)
		status := c.Writer.Status()

		fields := []zap.Field{
			zap.String("request_id", GetRequestID(c)),
			zap.String("method", method),
			zap.String("path", path),
			zap.Int("status", status),
			zap.Duration("latency", latency),
			zap.Int("response_size", c.Writer.Size()),
			zap.String("client_ip", c.ClientIP()),
		}

		if len(c.Errors) > 0 {
			// Gin errors may contain internal details. Keep them out of
			// access logs; application handlers should log sanitized
			// diagnostic information at the appropriate layer.
			fields = append(fields, zap.Int("gin_error_count", len(c.Errors)))
		}

		switch {
		case status >= 500:
			logger.Error("HTTP request completed", fields...)
		case status >= 400:
			logger.Warn("HTTP request completed", fields...)
		default:
			logger.Info("HTTP request completed", fields...)
		}
	}
}
```
