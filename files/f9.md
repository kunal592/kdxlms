## File 9 — `internal/platform/cache/redis.go`

Purpose: Production Redis client initialization for KDXLabsLLM. Redis will support session management, rate limiting, verification codes, caching, and other backend workloads.

This implementation includes connection validation, configurable timeouts, optional TLS, startup connectivity verification, and safe cleanup.

```

package cache

import (
	"context"
	"crypto/tls"
	"errors"
	"fmt"
	"strings"

	"github.com/redis/go-redis/v9"

	"github.com/kdxlabs/kdxlabsllm/internal/config"
)

// NewRedisClient creates a Redis client and verifies connectivity.
//
// The caller owns the returned client and must close it during application
// shutdown. The initialization context controls the connectivity check.
func NewRedisClient(
	ctx context.Context,
	cfg config.RedisConfig,
) (*redis.Client, error) {
	if ctx == nil {
		return nil, errors.New("Redis initialization context must not be nil")
	}

	if strings.TrimSpace(cfg.Address) == "" {
		return nil, errors.New("Redis address must not be empty")
	}

	if cfg.DB < 0 {
		return nil, errors.New("Redis database index must not be negative")
	}

	if cfg.DialTimeout <= 0 {
		return nil, errors.New("Redis dial timeout must be positive")
	}

	if cfg.ReadTimeout <= 0 || cfg.WriteTimeout <= 0 {
		return nil, errors.New("Redis read and write timeouts must be positive")
	}

	if cfg.PoolSize < 1 {
		return nil, errors.New("Redis pool size must be positive")
	}

	options := &redis.Options{
		Addr:         cfg.Address,
		Username:     cfg.Username,
		Password:     cfg.Password,
		DB:           cfg.DB,
		DialTimeout:  cfg.DialTimeout,
		ReadTimeout:  cfg.ReadTimeout,
		WriteTimeout: cfg.WriteTimeout,
		PoolSize:     cfg.PoolSize,
	}

	if cfg.TLS {
		options.TLSConfig = &tls.Config{
			MinVersion: tls.VersionTLS12,
		}
	}

	client := redis.NewClient(options)

	if err := client.Ping(ctx).Err(); err != nil {
		if closeErr := client.Close(); closeErr != nil {
			return nil, fmt.Errorf(
				"verify Redis connectivity: %w (close client: %v)",
				err,
				closeErr,
			)
		}

		return nil, fmt.Errorf("verify Redis connectivity: %w", err)
	}

	return client, nil
}

// Ping verifies that Redis is reachable using the supplied context.
// Use a bounded context when calling this function from health endpoints.
func Ping(ctx context.Context, client *redis.Client) error {
	if ctx == nil {
		return errors.New("Redis ping context must not be nil")
	}

	if client == nil {
		return errors.New("Redis client must not be nil")
	}

	if err := client.Ping(ctx).Err(); err != nil {
		return fmt.Errorf("Redis ping failed: %w", err)
	}

	return nil
}

// Close safely releases the Redis client's connections.
func Close(client *redis.Client) error {
	if client == nil {
		return nil
	}

	if err := client.Close(); err != nil {
		return fmt.Errorf("close Redis client: %w", err)
	}

	return nil
}
```
